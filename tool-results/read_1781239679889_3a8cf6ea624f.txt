    40→async function isGatewayOnlineCached(): Promise<boolean> {
    41→  const now = Date.now()
    42→  if (now - gatewayCacheTime < GATEWAY_CACHE_TTL) return gatewayOnlineCached
    43→  gatewayCacheTime = now
    44→  const result = await isGatewayOnline()
    45→  gatewayOnlineCached = result.online
    46→  return gatewayOnlineCached

    53→function isTrivialExchange(userMsg: string, assistantMsg: string): boolean {
    54→  // Skip memory extraction for trivial exchanges (greetings, short confirmations)
    55→  if (userMsg.trim().length < 15 && assistantMsg.trim().length < 50) return true

    65→/**
    66→ * Run a ReAct (Reason-Act-Observe) loop for an agent with tool execution.
    67→ *
    68→ * This enables standalone agents like Omega to actually USE their declared tools
    69→ * (opencode, knowledge_search, tavily, etc.) instead of just having them in prompts.
    70→ *
    71→ * Flow:
    72→ * 1. Call LLM with system prompt + user message
    73→ * 2. Parse tool_call patterns from LLM output
    74→ * 3. If tool calls found → execute → feed results back → repeat
    75→ * 4. If no tool calls → return final answer
    76→ * 5. Max 5 iterations to prevent infinite loops
    77→ */
    78→async function runAgentReActLoop(params: {
    79→  systemPrompt: string
    80→  userMessage: string
    81→  chatHistory: Array<{ role: string; content: string }>
    82→  agentProvider: string
    83→  agentModel: string
    84→  agentId: string
    85→  agentName: string
    86→  agentTemperature?: number
    87→  agentMaxTokens?: number
    88→  agentTools?: string[]
    89→  maxIterations?: number
    90→}): Promise<{ content: string; iterations: number; toolCallsExecuted: number }> {
    91→  const {
    92→    systemPrompt,
    93→    userMessage,
    94→    chatHistory,
    95→    agentProvider,
    96→    agentModel,
    97→    agentId,
    98→    agentName,
    99→    agentTemperature,
   100→    agentMaxTokens,
   101→    agentTools = [],
   102→    maxIterations = 5,
   103→  } = params
   104→
   105→  // Build messages array for multi-turn conversation
   106→  const messages: Array<{
   107→    role: 'system' | 'user' | 'assistant' | 'tool'
   108→    content: string
   109→    tool_call_id?: string
   110→    tool_calls?: Array<{ id: string; type: 'function'; function: { name: string; arguments: string } }>
   111→  }> = [
   112→    { role: 'system', content: systemPrompt },
   113→    // Include last 6 messages from chat history for context
   114→    ...chatHistory.slice(-6).map(m => ({
   115→      role: m.role as 'user' | 'assistant',
   116→      content: m.content,
   117→    })),
   118→    { role: 'user', content: userMessage },
   119→  ]
   120→
   121→  let finalContent = ''
   122→  let iterations = 0
   123→  let totalToolCalls = 0
   124→
   125→  while (iterations < maxIterations) {
   126→    iterations++
   127→
   128→    // Format messages for LLM (since callLLMForAgent takes a single prompt string)
   129→    const formattedPrompt = formatMessagesForLLM(messages)
   130→
   131→    // Get tool definitions for this agent (for structured tool calling)
   132→    const toolDefs = agentTools.length > 0 ? getToolDefinitionsForLLM(agentTools) : []
   133→
   134→    // Call LLM with tools parameter (P0.3 added tools support)
   135→    const result = await callLLMForAgent(
   136→      formattedPrompt,
   137→      { provider: agentProvider, model: agentModel },
   138→      systemPrompt,
   139→      {
   140→        agentId,
   141→        agentName,
   142→        temperature: agentTemperature,
   143→        maxTokens: agentMaxTokens,
   144→        tools: toolDefs.length > 0 ? toolDefs : undefined,
   145→      }
   146→    )
   147→
   148→    if (!result || (result.error && !result.content)) {
   149→      finalContent = result?.error || 'LLM call failed'
   150→      break
   151→    }
   152→
   153→    finalContent = result.content
   154→
   155→    // Parse tool calls — prefer structured tool_calls from LLM, fallback to regex
   156→    let toolCalls: Array<{ id: string; name: string; args: Record<string, unknown> }> = []
   157→
   158→    if (result.toolCalls && result.toolCalls.length > 0) {
   159→      // ✅ Structured tool_calls from LLM (preferred — P0.4)
   160→      toolCalls = result.toolCalls.map(tc => ({
   161→        id: tc.id,
   162→        name: tc.function.name,
   163→        args: (() => { try { return JSON.parse(tc.function.arguments) } catch { return {} } })(),
   164→      }))
   165→      console.log(`[ReAct] ${agentName} structured tool_calls: ${toolCalls.map(tc => tc.name).join(', ')}`)
   166→    } else {
   167→      // ❌ Fallback regex parsing (for models that don't support tool calling)
   168→      toolCalls = parseToolCallsFromOutput(result.content)
   169→    }
   170→
   171→    if (toolCalls.length === 0) {
   172→      // No tool calls → agent is done
   173→      break
   174→    }
   175→
   176→    // Execute tool calls
   177→    console.log(`[ReAct] ${agentName} calling ${toolCalls.length} tools (iteration ${iterations})`)
   178→
   179→    // Add assistant message with tool calls to conversation
   180→    messages.push({
   181→      role: 'assistant',
   182→      content: result.content,
   183→      ...(result.toolCalls && result.toolCalls.length > 0 ? {
   184→        tool_calls: result.toolCalls,
   185→      } : {}),
   186→    })
   187→
   188→    for (const tc of toolCalls) {
   189→      totalToolCalls++
   190→
   191→      // Only execute tools that the agent has permission for
   192→      if (agentTools.length > 0 && !agentTools.includes(tc.name)) {
   193→        messages.push({
   194→          role: 'tool',
   195→          content: JSON.stringify({ tool: tc.name, success: false, result: `Tool "${tc.name}" not available for this agent. Available tools: ${agentTools.join(', ')}` }),
   196→          tool_call_id: tc.id,
   197→        })
   198→        continue
   199→      }
   200→
   201→      console.log(`[ReAct] Executing: ${tc.name}(${JSON.stringify(tc.args).slice(0, 200)})`)
   202→
   203→      const toolResult = await executeTool(tc.name, tc.args)
   204→
   205→      // Add tool result to conversation (with tool_call_id for OpenAI format)
   206→      messages.push({
   207→        role: 'tool',
   208→        content: JSON.stringify({
   209→          tool: tc.name,
   210→          success: toolResult.success,
   211→          result: toolResult.result,
   212→        }),
   213→        tool_call_id: tc.id,
   214→      })
   215→    }
   216→
   217→    // If max iterations reached, break
   218→    if (iterations >= maxIterations) {
   219→      finalContent += `\n\n⚠️ Đạt max ${maxIterations} vòng iteration.`
   220→      break
   221→    }
   222→  }
   223→
   224→  console.log(`[ReAct] ${agentName} done: ${iterations} iterations, ${totalToolCalls} tool calls`)
   225→  return { content: finalContent, iterations, toolCallsExecuted: totalToolCalls }
   226→}

   430→async function routeToOpenCode(
   431→  userMessage: string,
   432→  model?: string,
   433→  sessionId?: string,
   434→  messages?: Array<{ role: string; content: string }>,
   435→  agentProfileId?: string,
   436→  teamMode?: string,
   437→  teamName?: string,
   438→): Promise<NextResponse | null> {
   439→  try {
   440→    const online = await isOpenCodeOnline()
   441→    if (!online) return null
   442→
   443→    // OC-3.1: Enrich with Knowledge Base context before creating session
   444→    const codeConfidence = detectCodeConfidence(userMessage)
   445→    const enrichment = await enrichCodeContext(userMessage)
   446→    
   447→    // OC-3.5: Cross-Session Learning — incorporate past corrections and insights
   448→    let entityTypeCount = 0
   449→    let documentCount = 0  
   450→    let correctionCount = 0
   451→    let insightCount = 0
   452→    let filesInWorkspace = 0
   453→    try {
   454→      entityTypeCount = await db.localEntity.count()
   455→      correctionCount = await db.agentCorrection.count({ where: { applied: true } })
   456→      insightCount = await db.agentInsight.count()
   457→      documentCount = await db.localEntity.count({ where: { domain: 'document' } })
   458→    } catch {}
   459→
   460→    // Count workspace files (cross-platform: uses Node.js fs instead of `find` command)
   461→    try {
   462→      const { readdirSync, statSync } = await import('fs')
   463→      const { join } = await import('path')
   464→      const workspaceDir = process.env.OPENCODE_WORKSPACE || process.cwd()
   465→      const skipDirs = new Set(['.next', 'node_modules', '.git', 'qdrant-storage', 'db', 'upload', 'skills'])
   466→      function countFiles(dir: string): number {
   467→        let count = 0
   468→        try {
   469→          for (const entry of readdirSync(dir)) {
   470→            const full = join(dir, entry)
   471→            try {
   472→              const stat = statSync(full)
   473→              if (stat.isDirectory()) {
   474→                if (!skipDirs.has(entry)) count += countFiles(full)
   475→              } else {
   476→                count++
   477→              }
   478→            } catch {}
   479→          }
   480→        } catch {}
   481→        return count
   482→      }
   483→      filesInWorkspace = countFiles(workspaceDir)
   484→    } catch {}
   485→
   486→    // OC-3.7: Generate enriched system prompt with KB context
   487→    const enrichedSystemPrompt = generateOpenCodeSystemPrompt({
   488→      entityTypeCount,
   489→      documentCount,
   490→      correctionCount,
   491→      insightCount,
   492→      filesInWorkspace,
   493→      modelList: [],
   494→      kbEnabled: true,
   495→      mcpToolsEnabled: ['knowledge_search', 'knowledge_graph', 'knowledge_write', 'web_search'],
   496→      recentCorrections: enrichment.corrections,
   497→      recentInsights: enrichment.insights.slice(0, 5),
   498→      relatedEntities: enrichment.entities,
   499→    })
   500→
   501→    // Try to create an OpenCode session with the prompt + enriched context
   502→    const session = await createOpenCodeSession({
   503→      prompt: userMessage,
   504→      model: model || undefined,
   505→    })
   506→
   507→    if (!session) return null
   508→
   509→    // Track in SQLite
   510→    if (sessionId) {
   511→      try {
   512→        await db.agentSession.upsert({
   513→          where: { sessionId },
   514→          create: {
   515→            sessionId,
   516→            model: model || 'opencode',
   517→            provider: 'opencode',
   518→            title: userMessage.slice(0, 50) || 'Code Session',
   519→            messageCount: 1,
   520→            agentProfileId: agentProfileId || null,
   521→            teamMode: teamMode || null,
   522→            teamName: teamName || null,
   523→          },
   524→          update: {
   525→            messageCount: { increment: 1 },
   526→            model: model || undefined,
   527→          },
   528→        })
   529→      } catch {}
   530→    }
   531→
   532→    // Also track as OpenCodeSession
   533→    try {
   534→      await db.openCodeSession.create({
   535→        data: {
   536→          sessionId: session.sessionId || session.id || `oc-${Date.now()}`,
   537→          model: model || null,
   538→          provider: 'opencode',
   539→          prompt: userMessage,
   540→          status: 'active',
   541→        },
   542→      })
   543→    } catch {}
   544→
   545→    // Build enrichment info for the response
   546→    const enrichmentInfo = enrichment.enrichmentScore > 0
   547→      ? `\n\n📊 **KB Context Enrichment** (score: ${(enrichment.enrichmentScore * 100).toFixed(0)}%):\n` +
   548→        (enrichment.entities.length > 0 ? `• Entities: ${enrichment.entities.length} found (${enrichment.entities.slice(0, 3).map(e => e.name).join(', ')}${enrichment.entities.length > 3 ? '...' : ''})\n` : '') +

   583→/**
   584→ * Route chat through OpenClaw Gateway with model override per agent.
   585→ * When Gateway is online, it provides 37 built-in tools + agent loop.
   586→ * Uses x-openclaw-model header to specify which model the agent should use.
   587→ */
   588→async function routeThroughGateway(params: {
   589→  messages: Array<{ role: string; content: string }>
   590→  agentProvider: string
   591→  agentModel: string
   592→  sessionId?: string
   593→  temperature?: number
   594→  maxTokens?: number
   595→  agentProfileId?: string
   596→  agentProfileName?: string
   597→  teamMode?: string
   598→  teamName?: string
   599→}): Promise<NextResponse | null> {
   600→  const { messages, agentProvider, agentModel, sessionId, temperature, maxTokens,
   601→          agentProfileId, agentProfileName, teamMode, teamName } = params
   602→
   603→  try {
   604→    // Map agent model to x-openclaw-model header format
   605→    // Gateway uses: nvidia/qwen/qwen3.5-397b-a17b format
   606→    const gatewayModel = `${agentProvider}/${agentModel}`
   607→
   608→    const res = await gatewayFetch('/v1/chat/completions', {
   609→      method: 'POST',
   610→      headers: {
   611→        'Content-Type': 'application/json',
   612→        'x-openclaw-model': gatewayModel,
   613→      },
   614→      body: JSON.stringify({
   615→        model: 'openclaw/default',
   616→        messages,
   617→        stream: false,
   618→        ...(temperature !== undefined ? { temperature } : {}),
   619→        ...(maxTokens !== undefined ? { max_tokens: maxTokens } : {}),
   620→      }),
   621→      timeout: 120000,
   622→    })
   623→
   624→    if (!res.ok) {
   625→      const errorText = await res.text().catch(() => '')
   626→      console.warn(`[Gateway] Chat error ${res.status}: ${errorText.slice(0, 200)}`)
   627→      return null
   628→    }
   629→
   630→    const data = await res.json()
   631→    const content = data.choices?.[0]?.message?.content || 'No response'
   632→
   633→    // Track token usage
   634→    const usage = data.usage as { prompt_tokens?: number; completion_tokens?: number; total_tokens?: number } | undefined
   635→    const totalTokens = usage?.total_tokens ?? ((usage?.prompt_tokens ?? 0) + (usage?.completion_tokens ?? 0))
   636→    if (totalTokens > 0) {
   637→      addTokensUsed(totalTokens, agentProvider.toLowerCase(), undefined, agentModel)
   638→      if (agentProfileId && agentProfileName) {
   639→        const inputTokens = usage?.prompt_tokens ?? 0
   640→        const outputTokens = usage?.completion_tokens ?? 0
   641→        addTokensUsedByAgent(totalTokens, inputTokens, outputTokens, agentProvider.toLowerCase(), agentModel, agentProfileId, agentProfileName)
   642→      }
   643→    }
   644→
   645→    // Track session
   646→    if (sessionId) {
   647→      await db.agentSession.upsert({
   648→        where: { sessionId },
   649→        create: {
   650→          sessionId,
   651→          model: agentModel,
   652→          provider: agentProvider,
   653→          title: messages[0]?.content?.slice(0, 50) || 'New Chat',
   654→          messageCount: messages.length + 1,
   655→          agentProfileId: agentProfileId || null,
   656→          teamMode: teamMode || null,
   657→          teamName: teamName || null,
   658→        },
   659→        update: { messageCount: { increment: 1 } },
   660→      }).catch(() => {})
   661→    }
   662→
   663→    return NextResponse.json({
   664→      content,
   665→      model: agentModel,
   666→      provider: agentProvider,
   667→      sessionId: sessionId || `session-${Date.now()}`,
   668→      agentProfileId: agentProfileId || undefined,
   669→      sources: [],
   670→      confidence: 0.7,
   671→      kbResults: 'gateway-react',
   672→      gatewayToolsAvailable: true,
   673→      gatewayModel: data.model,
   674→    })
   675→  } catch (err) {
   676→    console.warn('[Gateway] Error:', err instanceof Error ? err.message : String(err))
   677→    return null
   678→  }
   679→}

  1068→    //   - Agent with configured provider/model when ReAct loop fails (fallback)
  1069→    //   - Gateway offline fallback (no agent config)
  1070→    if (hasAgentProviderModel) {
  1071→      console.log(`[OpenClaw Chat] ReAct loop failed, falling back to /api/query for agent (${agentProvider}/${agentModel})`)
  1072→    }
  1073→    try {
  1074→      const baseUrl = process.env.NEXT_PUBLIC_BASE_URL || 'http://localhost:3000'
  1075→      const queryRes = await fetch(`${baseUrl}/api/query`, {
  1076→        method: 'POST',
  1077→        headers: { 'Content-Type': 'application/json' },
  1078→        body: JSON.stringify({
  1079→          query: lastUserContent,
  1080→          options: {
  1081→            chatHistory: messages.filter((m: { role: string }) => m.role === 'user' || m.role === 'assistant').slice(-6).map((m: { role: string; content: string }) => ({ role: m.role, content: m.content })),
  1082→            // Skip meta-cognitive reasoning for chat flow — reduces latency by ~15s
  1083→            // KB results are already enriched with memories in the system prompt
  1084→            skipMetaCog: true,
  1085→            ...(agentTemperature !== undefined ? { temperature: agentTemperature } : {}),
  1086→            ...(agentMaxTokens !== undefined ? { maxTokens: agentMaxTokens } : {}),
  1087→            ...(agentProfileId ? { agentId: agentProfileId } : {}),
  1088→            ...(agentProfileName ? { agentName: agentProfileName } : {}),
  1089→            ...(agentProvider ? { agentProvider } : {}),
  1090→            ...(agentModel ? { agentModel } : {}),
  1091→          },
  1092→        }),
  1093→        signal: AbortSignal.timeout(60000),
  1094→      })
  1095→
  1096→      if (queryRes.ok) {
  1097→        const data = await queryRes.json()
  1098→        const content = data.answer || 'Không thể tạo câu trả lời.'
  1099→
  1100→        // If this was a code query but OpenCode was offline, add a note
  1101→        const codeNote = isCodeQuery(lastUserContent)
  1102→          ? '\n\n⚠️ **OpenCode Server đang offline** — Không thể thực hiện code editing. Kết quả trên chỉ dựa trên Knowledge Base. Hãy bật OpenCode Server để sử dụng đầy đủ tính năng code.'
  1103→          : ''
  1104→
  1105→        // Track session (include agentProfileId if from an Agent)
  1106→        // NOTE: agentTemperature/agentMaxTokens are now passed to the local fallback via /api/query
  1107→        if (sessionId) {
  1108→          try {
  1109→            await db.agentSession.upsert({
  1110→              where: { sessionId },
  1111→              create: {
  1112→                sessionId,
  1113→                model: model || 'local-rag',
  1114→                provider: 'local-fallback',
  1115→                title: messages[0]?.content?.slice(0, 50) || 'New Chat',
  1116→                messageCount: messages.length + 1,
  1117→                agentProfileId: agentProfileId || null,
  1118→                teamMode: teamMode || null,
  1119→                teamName: teamName || null,
  1120→              },
  1121→              update: {
  1122→                messageCount: { increment: 1 },
  1123→              },
  1124→            })
  1125→          } catch {}
  1126→        }
  1127→
  1128→        // Phase 4: Auto-Learn — if agent has provider/model and confidence is in learning zone,
  1129→        // trigger background learning so the knowledge is available for future queries.
  1130→        // Non-blocking: fire-and-forget, don't delay the response.
  1131→        const queryConfidence = data.confidence ?? 0
  1132→        const hasAgent = !!(agentProfileId && agentProvider && agentModel)
  1133→        if (shouldAutoLearn(queryConfidence, hasAgent)) {
  1134→          autoLearnFromAnswer({
  1135→            query: lastUserContent,
  1136→            answer: content,
  1137→            confidence: queryConfidence,
  1138→            agentId: agentProfileId!,
  1139→            agentName: agentProfileName || 'unknown',
  1140→            provider: agentProvider!,
  1141→            model: agentModel!,
  1142→            sources: data.sources,
  1143→          }).catch(err => console.error('[AutoLearn] Background error:', err instanceof Error ? err.message : String(err)))
  1144→        }
  1145→
  1146→        // Phase 5: Memory & Learning — save messages + extract memories in background
  1147→        // Non-blocking: fire-and-forget, don't delay the response.
  1148→        if (agentProfileId && sessionId) {
  1149→          // Save chat messages to DB for persistence
  1150→          saveChatMessages(sessionId, [
  1151→            { role: 'user', content: lastUserContent },
  1152→            { role: 'assistant', content, model: agentModel || model, provider: agentProvider || 'local-fallback', metadata: { confidence: queryConfidence, sources: data.sources?.length || 0 } },
  1153→          ]).catch(err => console.warn('[Memory] Failed to save chat messages:', err instanceof Error ? err.message : String(err)))
  1154→
  1155→          // Extract memories from this conversation (skip trivial exchanges to save LLM tokens)
  1156→          if (!isTrivialExchange(lastUserContent, content)) {
  1157→            extractMemoriesFromConversation({
  1158→              agentId: agentProfileId,
  1159→              agentName: agentProfileName || 'unknown',
  1160→              sessionId,
  1161→              userMessage: lastUserContent,
  1162→              assistantMessage: content,
  1163→            }).catch(err => console.warn('[Memory] Failed to extract memories:', err instanceof Error ? err.message : String(err)))
  1164→          }
  1165→
  1166→          // Periodic memory decay — every ~20 chat requests, decay old memories
  1167→          // This prevents stale memories from accumulating without requiring a cron scheduler
  1168→          memoryDecayCounter++
  1169→          if (memoryDecayCounter >= 20 && agentProfileId) {
  1170→            memoryDecayCounter = 0
  1171→            decayMemories(agentProfileId).catch(err =>
  1172→              console.warn('[Memory] Periodic decay error:', err instanceof Error ? err.message : String(err))
  1173→            )
  1174→          }
  1175→        }
  1176→
  1177→        return NextResponse.json({
  1178→          content: content + codeNote,
  1179→          model: model || 'local-rag',
  1180→          provider: 'local-fallback',
  1181→          sessionId: sessionId || `session-${Date.now()}`,
  1182→          agentProfileId: agentProfileId || undefined,
  1183→          sources: data.sources || [],
  1184→          confidence: data.confidence,
  1185→          kbResults: 'included',
  1186→          codeQuery: isCodeQuery(lastUserContent) || undefined,
  1187→          memoriesRecalled: recalledMemories.length || undefined,
  1188→        })
  1189→      }
  1190→    } catch (err) {
  1191→      console.warn('[OpenClaw Chat] Local fallback error:', err instanceof Error ? err.message : String(err))
  1192→    }
  1193→
  1194→    // Final fallback: mock response
  1195→    const userContent = lastUserContent || ''
  1196→    const mockContent = userContent.toLowerCase().includes('hello') || userContent.toLowerCase().includes('hi')
  1197→      ? `Xin chào! Tôi là AI Agent của The Magnum Opus. Gateway hiện đang offline nên tôi hoạt động ở chế độ mock. Hãy kết nối Gateway để nhận phản hồi AI thực sự.`
  1198→      : `Tôi đã nhận tin nhắn của bạn: "${userContent.slice(0, 100)}${userContent.length > 100 ? '...' : ''}"\n\nOpenClaw Gateway đang offline và Knowledge Base cũng không khả dụng. Tôi đang ở chế độ mock. Vui lòng kiểm tra kết nối.`
  1199→
  1200→    return NextResponse.json({
  1201→      content: mockContent,
  1202→      model: model || 'mock',
  1203→      provider: 'mock',
  1204→      sessionId: sessionId || `session-${Date.now()}`,
  1205→      kbResults: 'none',
  1206→    })
  1207→  } catch (err) {
  1208→    return NextResponse.json(
  1209→      { error: 'Invalid request body', details: err instanceof Error ? err.message : String(err) },
  1210→      { status: 400 }
  1211→    )
  1212→  }
