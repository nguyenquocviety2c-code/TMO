   497→/// Embedding cache — stores previously computed embeddings to avoid redundant API calls
   498→/// Survives server restarts (unlike in-memory cache)
   499→/// IMPORTANT: Same text gets DIFFERENT embeddings for "query" vs "passage" (NVIDIA input_type)
   500→/// So the unique key is (hash, inputType) — NOT just hash
   501→model EmbeddingCache {
   502→  id        String   @id @default(cuid())
   503→  hash     String   /// SHA-256 hash of the input text
   504→  inputType String  @default("passage") /// "query" | "passage" — different embeddings for same text
   505→  model    String   /// Model used to generate the embedding (e.g. "nvidia/llama-nemotron-embed-1b-v2")
   506→  vector   String   /// JSON array of the embedding vector (1536 floats)
   507→  hitCount Int      @default(1) /// Number of cache hits (for LRU eviction)
   508→  createdAt DateTime @default(now())
   509→  lastHitAt DateTime @default(now()) /// Last time this cache entry was used
   510→  expiresAt DateTime /// When this cache entry expires
   511→
   512→  @@unique([hash, inputType]) /// Composite key — same text has separate query & passage embeddings
   513→  @@index([expiresAt])
   514→  @@index([lastHitAt])
   515→}

   590→/// Smolab background tasks — tracks long-running agent/team work that continues
   591→/// even when user switches to another session. Each task is tied to one session
   592→/// and one agent (or team lead agent).
   593→model SmolabTask {
   594→  id              String   @id @default(cuid())
   595→  type            String   /// "agent_chat" | "team_workflow" | "agent_react" | "memory_extraction"
   596→  status          String   @default("pending") /// "pending" | "running" | "completed" | "failed" | "cancelled"
   597→  progress        Int      @default(0) /// 0-100 percentage
   598→
   599→  // Liên kết Session + Agent
   600→  sessionId       String   /// AgentSession.sessionId — phiên chat gốc
   601→  agentProfileId  String?  /// AgentProfile.id — agent đang xử lý (null nếu team mode)
   602→  teamName        String?  /// "code" | "research" — team đang xử lý (nếu team mode)
   603→
   604→  // Task input/output
   605→  inputSummary    String?  /// Tóm tắt yêu cầu (first 200 chars của user message)
   606→  result          String?  /// Kết quả khi hoàn thành (JSON string)
   607→  error           String?  /// Lỗi nếu thất bại
   608→
   609→  // Thời gian
   610→  createdAt       DateTime @default(now())
   611→  startedAt       DateTime?
   612→  completedAt     DateTime?
   613→  updatedAt       DateTime @updatedAt
   614→
   615→  // Relations — Cascade: khi session bị xóa, tất cả tasks của session đó cũng bị xóa
   616→  session         AgentSession  @relation(fields: [sessionId], references: [sessionId], onDelete: Cascade, onUpdate: Cascade)
   617→
   618→  @@index([sessionId])
   619→  @@index([agentProfileId, status])
   620→  @@index([teamName, status])
   621→  @@index([status])
   622→  @@index([createdAt])
   623→}

   625→/// Agent episodic memories — extracted from conversations, stored as vectors in Qdrant
   626→/// Each memory is a discrete piece of knowledge the agent learned from interaction
   627→model AgentMemory {
   628→  id             String   @id @default(cuid())
   629→  agentId        String   /// AgentProfile.id
   630→  agentName      String   /// Agent name (denormalized)
   631→  sessionId      String?  /// Source session
   632→  category       String   /// insight | fact | preference | correction | procedure | user_info
   633→  content        String   /// The actual memory content (text)
   634→  context        String?  /// Context: what triggered this memory (query, situation, etc.)
   635→  importance     Float    @default(0.5) /// 0.0-1.0 — how important this memory is
   636→  accessCount    Int      @default(0) /// How many times this memory was recalled
   637→  lastAccessedAt DateTime? /// Last time this memory was used in a response
   638→  qdrantPointId  String?  /// Qdrant point ID for vector search
   639→  embeddingModel String?  /// Embedding model used (e.g. nvidia/llama-nemotron-embed-1b-v2)
   640→  source         String   @default("auto") /// auto | manual | feedback | correction
   641→  tags           String?  /// JSON array of tags for categorization
   642→  isActive       Boolean  @default(true) /// Soft delete — inactive memories are not recalled
   643→  expiresAt      DateTime? /// Optional expiration for time-sensitive memories
   644→  createdAt      DateTime @default(now())
   645→  updatedAt      DateTime @updatedAt
   646→
   647→  @@index([agentId])
   648→  @@index([category])
   649→  @@index([importance])
   650→  @@index([accessCount])
   651→  @@index([isActive])
   652→  @@index([createdAt])
   653→  @@index([lastAccessedAt])
   654→}
