    66→/**
    67→ * Check if a tool is permitted for an agent based on ToolPermission records.
    68→ *
    69→ * Logic:
    70→ * - No explicit permission → default allow (tools are allowed unless explicitly denied)
    71→ * - permission = "deny" → blocked
    72→ * - permission = "ask" → blocked in auto mode (would need user interaction to approve)
    73→ * - permission = "allow" → check rate limit, update usage tracking
    74→ */
    75→async function checkToolPermission(agentId: string, toolName: string): Promise<{ allowed: boolean; reason?: string }> {
    76→  try {
    77→    const perm = await db.toolPermission.findUnique({
    78→      where: { agentId_toolName: { agentId, toolName } },
    79→    })
    80→
    81→    if (!perm) {
    82→      // No explicit permission → default allow (tools are allowed unless explicitly denied)
    83→      return { allowed: true }
    84→    }
    85→
    86→    if (perm.permission === 'deny') {
    87→      return { allowed: false, reason: `Tool "${toolName}" is denied for this agent` }
    88→    }
    89→
    90→    if (perm.permission === 'ask') {
    91→      // In auto mode, "ask" means deny (would need user interaction to approve)
    92→      return { allowed: false, reason: `Tool "${toolName}" requires user approval` }
    93→    }
    94→
    95→    // Check rate limit
    96→    if (perm.maxCallsPerHour && perm.lastUsedAt) {
    97→      const oneHourAgo = new Date(Date.now() - 3600000)
    98→      if (perm.lastUsedAt > oneHourAgo) {
    99→        // Get current hour call count (approximate)
   100→        if (perm.callCount >= perm.maxCallsPerHour) {
   101→          return { allowed: false, reason: `Tool "${toolName}" exceeded rate limit (${perm.maxCallsPerHour}/hour)` }
   102→        }
   103→      }
   104→    }
   105→
   106→    // Update usage tracking
   107→    await db.toolPermission.update({
   108→      where: { agentId_toolName: { agentId, toolName } },
   109→      data: {
   110→        lastUsedAt: new Date(),
   111→        callCount: { increment: 1 },
   112→      },
   113→    }).catch(() => {})
   114→
   115→    return { allowed: true }
   116→  } catch {
   117→    // Permission check failed → default allow
   118→    return { allowed: true }
   119→  }
   120→}

   126→/**
   127→ * Run a ReAct (Reason-Act-Observe) loop for an agent with tool execution.
   128→ *
   129→ * This enables standalone agents like Omega to actually USE their declared tools
   130→ * (opencode, knowledge_search, tavily, etc.) instead of just having them in prompts.
   131→ *
   132→ * Flow:
   133→ * 1. Call LLM with system prompt + user message
   134→ * 2. Parse tool_call patterns from LLM output
   135→ * 3. If tool calls found → execute → feed results back → repeat
   136→ * 4. If no tool calls → return final answer
   137→ * 5. Max 5 iterations to prevent infinite loops
   138→ */
   139→async function runAgentReActLoop(params: {
   140→  systemPrompt: string
   141→  userMessage: string
   142→  chatHistory: Array<{ role: string; content: string }>
   143→  agentProvider: string
   144→  agentModel: string
   145→  agentId: string
   146→  agentName: string
   147→  agentTemperature?: number
   148→  agentMaxTokens?: number
   149→  agentTools?: string[]
   150→  maxIterations?: number
   151→}): Promise<{ content: string; iterations: number; toolCallsExecuted: number }> {
   152→  const {
   153→    systemPrompt,
   154→    userMessage,
   155→    chatHistory,
   156→    agentProvider,
   157→    agentModel,
   158→    agentId,
   159→    agentName,
   160→    agentTemperature,
   161→    agentMaxTokens,
   162→    agentTools = [],
   163→    maxIterations = 5,
   164→  } = params
   165→
   166→  // Build messages array for multi-turn conversation using ChatMessageForLLM type
   167→  // This preserves structured tool_calls with tool_call_id for proper multi-turn tool calling
   168→  const messages: ChatMessageForLLM[] = [
   169→    { role: 'system', content: systemPrompt },
   170→    // Include last 6 messages from chat history for context
   171→    ...chatHistory.slice(-6).map(m => ({
   172→      role: m.role as 'user' | 'assistant',
   173→      content: m.content,
   174→    })),
   175→    { role: 'user', content: userMessage },
   176→  ]
   177→
   178→  let finalContent = ''
   179→  let iterations = 0
   180→  let totalToolCalls = 0
   181→
   182→  while (iterations < maxIterations) {
   183→    iterations++
   184→
   185→    // Get tool definitions for this agent (for structured tool calling)
   186→    const toolDefs = agentTools.length > 0 ? getToolDefinitionsForLLM(agentTools) : []
   187→
   188→    // Call LLM with full messages array — preserves tool_calls structure (CRITICAL FIX)
   189→    // Previously used formatMessagesForLLM() which flattened tool_calls to text, losing tool_call_id
   190→    const result = await callLLMForAgentWithMessages(
   191→      messages,
   192→      { provider: agentProvider, model: agentModel },
   193→      {
   194→        agentId,
   195→        agentName,
   196→        temperature: agentTemperature,
   197→        maxTokens: agentMaxTokens,
   198→        tools: toolDefs.length > 0 ? toolDefs : undefined,
   199→      }
   200→    )
   201→
   202→    if (!result || (result.error && !result.content && !result.toolCalls?.length)) {
   203→      finalContent = result?.error || 'LLM call failed'
   204→      break
   205→    }
   206→
   207→    finalContent = result.content || ''
   208→
   209→    // Parse tool calls — prefer structured tool_calls from LLM, fallback to regex
   210→    let toolCalls: Array<{ id: string; name: string; args: Record<string, unknown> }> = []
   211→
   212→    if (result.toolCalls && result.toolCalls.length > 0) {
   213→      // ✅ Structured tool_calls from LLM (preferred — P0.4)
   214→      toolCalls = result.toolCalls.map(tc => ({
   215→        id: tc.id,
   216→        name: tc.function.name,
   217→        args: (() => { try { return JSON.parse(tc.function.arguments) } catch { return {} } })(),
   218→      }))
   219→      console.log(`[ReAct] ${agentName} structured tool_calls: ${toolCalls.map(tc => tc.name).join(', ')}`)
   220→    } else {
   221→      // ❌ Fallback regex parsing (for models that don't support tool calling)
   222→      toolCalls = parseToolCallsFromOutput(result.content)
   223→    }
   224→
   225→    if (toolCalls.length === 0) {
   226→      // No tool calls → agent is done
   227→      break
   228→    }
   229→
   230→    // Execute tool calls
   231→    console.log(`[ReAct] ${agentName} calling ${toolCalls.length} tools (iteration ${iterations})`)
   232→
   233→    // Add assistant message with tool calls to conversation
   234→    messages.push({
   235→      role: 'assistant',
   236→      content: result.content,
   237→      ...(result.toolCalls && result.toolCalls.length > 0 ? {
   238→        tool_calls: result.toolCalls,
   239→      } : {}),
   240→    })
   241→
   242→    for (const tc of toolCalls) {
   243→      totalToolCalls++
   244→
   245→      // Only execute tools that the agent has permission for
   246→      if (agentTools.length > 0 && !agentTools.includes(tc.name)) {
   247→        messages.push({
   248→          role: 'tool',
   249→          content: JSON.stringify({ tool: tc.name, success: false, result: `Tool "${tc.name}" not available for this agent. Available tools: ${agentTools.join(', ')}` }),
   250→          tool_call_id: tc.id,
   251→        })
   252→        continue
   253→      }
   254→
   255→      // P2.2: Check tool permission before executing
   256→      const permCheck = await checkToolPermission(agentId || 'default', tc.name)
   257→      if (!permCheck.allowed) {
   258→        messages.push({
   259→          role: 'tool',
   260→          content: JSON.stringify({
   261→            tool: tc.name,
   262→            success: false,
   263→            result: permCheck.reason || `Tool "${tc.name}" is not permitted for this agent`,
   264→          }),
   265→          tool_call_id: tc.id,
   266→        })
   267→        continue
   268→      }
   269→
   270→      console.log(`[ReAct] Executing: ${tc.name}(${JSON.stringify(tc.args).slice(0, 200)})`)
   271→
   272→      const toolResult = await executeTool(tc.name, tc.args)
   273→
   274→      // Add tool result to conversation (with tool_call_id for OpenAI format)
   275→      messages.push({
   276→        role: 'tool',
   277→        content: JSON.stringify({
   278→          tool: tc.name,
   279→          success: toolResult.success,
   280→          result: toolResult.result,
   281→        }),
   282→        tool_call_id: tc.id,
   283→      })
   284→    }
   285→
   286→    // If max iterations reached, break
   287→    if (iterations >= maxIterations) {
   288→      finalContent += `\n\n⚠️ Đạt max ${maxIterations} vòng iteration.`
   289→      break
   290→    }
   291→  }
   292→
   293→  console.log(`[ReAct] ${agentName} done: ${iterations} iterations, ${totalToolCalls} tool calls`)
   294→  return { content: finalContent, iterations, toolCallsExecuted: totalToolCalls }
   295→}

   469→async function routeToOpenCode(
   470→  userMessage: string,

   474→  agentProfileId?: string,

   476→  teamName?: string,

   478→  try {
   479→    const online = await isOpenCodeOnline()
   480→    if (!online) return null
   481→
   482→    // OC-3.1: Enrich with Knowledge Base context before creating session
   483→    const codeConfidence = detectCodeConfidence(userMessage)
   484→    const enrichment = await enrichCodeContext(userMessage)
   485→    
   486→    // OC-3.5: Cross-Session Learning — incorporate past corrections and insights
   487→    let entityTypeCount = 0
   488→    let documentCount = 0  
   489→    let correctionCount = 0
   490→    let insightCount = 0
   491→    let filesInWorkspace = 0
   492→    try {
   493→      entityTypeCount = await db.localEntity.count()
   494→      correctionCount = await db.agentCorrection.count({ where: { applied: true } })
   495→      insightCount = await db.agentInsight.count()
   496→      documentCount = await db.localEntity.count({ where: { domain: 'document' } })
   497→    } catch {}
   498→
   499→    // Count workspace files (cross-platform: uses Node.js fs instead of `find` command)
   500→    try {
   501→      const { readdirSync, statSync } = await import('fs')
   502→      const { join } = await import('path')
   503→      const workspaceDir = process.env.OPENCODE_WORKSPACE || process.cwd()
   504→      const skipDirs = new Set(['.next', 'node_modules', '.git', 'qdrant-storage', 'db', 'upload', 'skills'])
   505→      function countFiles(dir: string): number {
   506→        let count = 0
   507→        try {
   508→          for (const entry of readdirSync(dir)) {
   509→            const full = join(dir, entry)
   510→            try {
   511→              const stat = statSync(full)
   512→              if (stat.isDirectory()) {
   513→                if (!skipDirs.has(entry)) count += countFiles(full)
   514→              } else {
   515→                count++
   516→              }
   517→            } catch {}
   518→          }
   519→        } catch {}
   520→        return count
   521→      }
   522→      filesInWorkspace = countFiles(workspaceDir)
   523→    } catch {}
   524→
   525→    // OC-3.7: Generate enriched system prompt with KB context
   526→    const enrichedSystemPrompt = generateOpenCodeSystemPrompt({
   527→      entityTypeCount,
   528→      documentCount,
   529→      correctionCount,
   530→      insightCount,
   531→      filesInWorkspace,
   532→      modelList: [],
   533→      kbEnabled: true,
   534→      mcpToolsEnabled: ['knowledge_search', 'knowledge_graph', 'knowledge_write', 'web_search'],
   535→      recentCorrections: enrichment.corrections,
   536→      recentInsights: enrichment.insights.slice(0, 5),
   537→      relatedEntities: enrichment.entities,
   538→    })
   539→

   622→/**
   623→ * Route chat through hybrid Gateway + local architecture.
   624→ *
   625→ * Architecture (Hybrid ReAct):
   626→ *   - LLM calls → NVIDIA NIM directly (callLLMForAgent)
   627→ *   - Tool execution → Gateway /tools/invoke when available, local fallback
   628→ *
   629→ * This replaces the old approach of routing ALL through Gateway /v1/chat/completions
   630→ * which returned 404 (Gateway uses WebSocket internally, not REST for chat).
   631→ */
   632→async function routeThroughGateway(params: {
   633→  messages: Array<{ role: string; content: string }>
   634→  agentProvider: string
   635→  agentModel: string
   636→  sessionId?: string
   637→  temperature?: number
   638→  maxTokens?: number
   639→  agentProfileId?: string
   640→  agentProfileName?: string
   641→  teamMode?: string
   642→  teamName?: string
   643→  agentTools?: string[]
   644→}): Promise<NextResponse | null> {
   645→  const { messages, agentProvider, agentModel, sessionId, temperature, maxTokens,
   646→          agentProfileId, agentProfileName, teamMode, teamName, agentTools = [] } = params
   647→
   648→  try {
   649→    // Hybrid approach: Use local ReAct loop with Gateway tool proxy
   650→    // This avoids the 404 from /v1/chat/completions while still using Gateway tools
   651→    const lastUserMsg = [...messages].reverse().find(m => m.role === 'user')
   652→    const userContent = typeof lastUserMsg?.content === 'string' ? lastUserMsg.content : ''
   653→    const systemMsg = messages.find(m => m.role === 'system')
   654→    const systemPrompt = systemMsg?.content || ''
   655→    const chatHistory = messages.filter(m => m.role === 'user' || m.role === 'assistant')
   656→
   657→    // Use the existing ReAct loop which already supports Gateway tool proxy
   658→    const reactResult = await runAgentReActLoop({
   659→      systemPrompt,
   660→      userMessage: userContent,
   661→      chatHistory,
   662→      agentProvider,
   663→      agentModel,
   664→      agentId: agentProfileId || 'unknown',
   665→      agentName: agentProfileName || 'Agent',
   666→      agentTemperature: temperature,
   667→      agentMaxTokens: maxTokens,
   668→      agentTools,
   669→      maxIterations: 5,
   670→    })
   671→
   672→    // Track session
   673→    if (sessionId) {
   674→      await db.agentSession.upsert({
   675→        where: { sessionId },
   676→        create: {
   677→          sessionId,
   678→          model: agentModel,
   679→          provider: agentProvider,
   680→          title: userContent.slice(0, 50) || 'New Chat',
   681→          messageCount: messages.length + 1,
   682→          agentProfileId: agentProfileId || null,
   683→          teamMode: teamMode || null,
   684→          teamName: teamName || null,
   685→        },
   686→        update: { messageCount: { increment: 1 } },
   687→      }).catch(() => {})
   688→    }
   689→
   690→    return NextResponse.json({
   691→      content: reactResult.content,
   692→      model: agentModel,
   693→      provider: agentProvider,
   694→      sessionId: sessionId || `session-${Date.now()}`,
   695→      agentProfileId: agentProfileId || undefined,
   696→      sources: [],
   697→      confidence: 0.7,
   698→      kbResults: 'hybrid-react',
   699→      gatewayToolsAvailable: true,
   700→      reactIterations: reactResult.iterations,
   701→      toolCallsExecuted: reactResult.toolCallsExecuted,
   702→    })
   703→  } catch (err) {
   704→    console.warn('[Gateway Hybrid] Error:', err instanceof Error ? err.message : String(err))
   705→    return null
   706→  }
   707→}

  1057→

  1080→
  1081→          return NextResponse.json({

  1091→        }
  1092→      } catch (err) {
  1093→        console.warn('[OpenClaw Chat] Gateway error:', err instanceof Error ? err.message : String(err))

  1095→    }
  1096→
  1097→    // /api/query fallback path — used for:
  1098→    //   - Agent with configured provider/model when ReAct loop fails (fallback)
  1099→    //   - Gateway offline fallback (no agent config)
  1100→    if (hasAgentProviderModel) {

  1102→    }
  1103→    try {
  1104→      const baseUrl = process.env.NEXT_PUBLIC_BASE_URL || 'http://localhost:3000'
  1105→      const queryRes = await fetch(`${baseUrl}/api/query`, {
  1106→        method: 'POST',
  1107→        headers: { 'Content-Type': 'application/json' },
  1108→        body: JSON.stringify({
  1109→          query: lastUserContent,
  1110→          options: {
  1111→            chatHistory: messages.filter((m: { role: string }) => m.role === 'user' || m.role === 'assistant').slice(-6).map((m: { role: string; content: string }) => ({ role: m.role, content: m.content })),
  1112→            // Skip meta-cognitive reasoning for chat flow — reduces latency by ~15s
  1113→            // KB results are already enriched with memories in the system prompt

  1118→            ...(agentProfileName ? { agentName: agentProfileName } : {}),
  1119→            ...(agentProvider ? { agentProvider } : {}),
  1120→            ...(agentModel ? { agentModel } : {}),
  1121→          },
  1122→        }),
  1123→        signal: AbortSignal.timeout(60000),
  1124→      })
  1125→
  1126→      if (queryRes.ok) {

  1128→        const content = data.answer || 'Không thể tạo câu trả lời.'
  1129→
  1130→        // If this was a code query but OpenCode was offline, add a note

  1133→          : ''

  1135→        // Track session (include agentProfileId if from an Agent)

  1139→            await db.agentSession.upsert({

  1143→                model: model || 'local-rag',

  1177→        // Non-blocking: fire-and-forget, don't delay the response.

  1204→          }
  1205→        }

  1213→          sources: data.sources || [],

  1217→          memoriesRecalled: recalledMemories.length || undefined,
  1218→        })

  1222→    }
  1223→

  1229→
  1230→    return NextResponse.json({
  1231→      content: mockContent,
  1232→      model: model || 'mock',
  1233→      provider: 'mock',
  1234→      sessionId: sessionId || `session-${Date.now()}`,
  1235→      kbResults: 'none',
  1236→    })
  1237→  } catch (err) {
  1238→    return NextResponse.json(
  1239→      { error: 'Invalid request body', details: err instanceof Error ? err.message : String(err) },
  1240→      { status: 400 }
