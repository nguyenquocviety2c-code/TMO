    17→    17→ *   Workflow Engine → callLLMForAgent() with tool definitions

   296→   296→/**
   297→   297→ * Execute a tool via Gateway HTTP /tools/invoke endpoint.
   298→   298→ * Only works for HTTP-allowed tools (web_search, web_fetch, etc.).
   299→   299→ * File operations and exec require session context and are NOT available.
   300→   300→ */
   301→   301→async function executeGatewayTool(name: string, args: Record<string, unknown>): Promise<unknown> {
   302→   302→  if (!GATEWAY_HTTP_ALLOWED_TOOLS.has(name)) {
   303→   303→    throw new Error(`Tool "${name}" requires Gateway session context — not available via HTTP proxy. Use /v1/chat/completions instead.`)
   304→   304→  }
   305→   305→
   306→   306→  const res = await fetch(`${GATEWAY_URL}/tools/invoke`, {
   307→   307→    method: 'POST',
   308→   308→    headers: { 'Content-Type': 'application/json' },
   309→   309→    body: JSON.stringify({ name, args }),
   310→   310→    signal: AbortSignal.timeout(30000),
   311→   311→  })
   312→   312→
   313→   313→  if (!res.ok) {
   314→   314→    const errorText = await res.text().catch(() => 'Unknown error')
   315→   315→    throw new Error(`Gateway tool error ${res.status}: ${errorText.slice(0, 200)}`)
   316→   316→  }
   317→   317→
   318→   318→  const data = await res.json()
   319→   319→  if (!data.ok && data.error) {
   320→   320→    throw new Error(`Tool error: ${data.error?.message || data.error || 'Unknown'}`)
   321→   321→  }
   322→   322→
   323→   323→  return data.result ?? data
   324→   324→}

   326→   326→// ==================== Z-AI-SDK LAZY INITIALIZATION ====================
   327→   327→//
   328→   328→// ⚠️  IMPORTANT: z-ai-web-dev-sdk is an OPTIONAL dependency!
   329→   329→//
   330→   330→// This SDK ONLY works inside the z.ai sandbox environment where backend
   331→   331→// services (web_search, page_reader) are available. If you clone this
   332→   332→// project and run locally, the SDK will not initialize — this is expected
   333→   333→// and NOT an error. Direct API keys (TAVILY_API_KEY, SERPER_API_KEY,
   334→   334→// JINA_API_KEY) in .env will be used instead.
   335→   335→//
   336→   336→// Fallback chain: Direct API key → z-ai-sdk → error with helpful hint
   337→   337→//
   338→   338→
   339→   339→let zaiInstance: any = null
   340→   340→let zaiInitPromise: Promise<any> | null = null
   341→   341→let zaiAvailable: boolean | null = null // null = not yet checked
   342→   342→
   343→   343→async function getZAI(): Promise<any> {
   344→   344→  if (zaiInstance) return zaiInstance
   345→   345→  if (zaiInitPromise) return zaiInitPromise
   346→   346→
   347→   347→  zaiInitPromise = (async () => {
   348→   348→    try {
   349→   349→      const ZAI = (await import('z-ai-web-dev-sdk')).default
   350→   350→      const instance = await ZAI.create()
   351→   351→      console.log('[ToolExecutor] ✅ z-ai-web-dev-sdk initialized — web_search + page_reader available as fallback')
   352→   352→      zaiInstance = instance
   353→   353→      zaiAvailable = true
   354→   354→      return instance
   355→   355→    } catch (err) {
   356→   356→      zaiAvailable = false
   357→   357→      console.warn(
   358→   358→        '[ToolExecutor] z-ai-web-dev-sdk not available (this is normal for local deployment — direct API keys will be used).',
   359→   359→        `Detail: ${err instanceof Error ? err.message : String(err)}`
   360→   360→      )
   361→   361→      return null
   362→   362→    }
   363→   363→  })()
   364→   364→
   365→   365→  return zaiInitPromise
   366→   366→}

   577→   577→ * @param toolName - Name of the tool to execute
   578→   578→ * @param args - Arguments from LLM's tool_call
   579→   579→ * @returns Execution result with success status
   580→   580→ */
   581→   581→export async function executeTool(
   582→   582→  toolName: string,
   583→   583→  args: Record<string, unknown>
   584→   584→): Promise<ToolExecutionResult> {
   585→   585→  const startTime = Date.now()
   586→   586→
   587→   587→  try {
   588→   588→    let result: unknown
   589→   589→
   590→   590→    switch (toolName) {
   591→   591→      case 'opencode':
   592→   592→        result = await executeOpenCode(args)

   594→   594→      case 'knowledge_search':
   595→   595→        result = await executeKnowledgeSearch(args)

   608→   608→        break

   610→   610→        result = await executeJina(args)
   611→   611→        break

   618→   618→          } catch (gwErr) {

   620→   620→              success: false,

   623→   623→            }
   624→   624→          }
   625→   625→        }

   627→   627→      }
   628→   628→    }
   629→   629→
   630→   630→    return {
   631→   631→      success: true,
   632→   632→      result,
   633→   633→      duration: Date.now() - startTime,

   639→   639→      success: false,
   640→   640→      result: `Tool error: ${errorMsg}`,

   648→   648→/**
   649→   649→ * OpenCode — Code operations (read, write, bash, mkdir)
   650→   650→ *
   651→   651→ * Uses the OpenCode API server (configured via OPENCODE_SERVER_URL env var) if available,
   652→   652→ * otherwise falls back to direct file system operations via the project API.
   653→   653→ */
   654→   654→async function executeOpenCode(args: Record<string, unknown>): Promise<unknown> {
   655→   655→  const action = args.action as string
   656→   656→  const path = args.path as string | undefined
   657→   657→  const content = args.content as string | undefined
   658→   658→  const command = args.command as string | undefined
   659→   659→
   660→   660→  // Try OpenCode server first
   661→   661→  try {
   662→   662→    const opencodeUrl = process.env.OPENCODE_SERVER_URL || 'http://localhost:18790'
   663→   663→    const res = await fetch(`${opencodeUrl}/api/execute`, {
   664→   664→      method: 'POST',
   665→   665→      headers: { 'Content-Type': 'application/json' },
   666→   666→      body: JSON.stringify({ action, path, content, command }),
   667→   667→      signal: AbortSignal.timeout(30000),
   668→   668→    })
   669→   669→
   670→   670→    if (res.ok) {
   671→   671→      const data = await res.json()
   672→   672→      return data
   673→   673→    }
   674→   674→  } catch {
   675→   675→    // OpenCode server not available, fall through to direct implementation
   676→   676→  }
   677→   677→
   678→   678→  // Fallback: Use the project's own API for file operations
   679→   679→  switch (action) {
   680→   680→    case 'read':
   681→   681→      return await executeFileRead(path || '')
   682→   682→    case 'write':
   683→   683→      return await executeFileWrite(path || '', content || '')
   684→   684→    case 'bash':
   685→   685→      return await executeBash(command || '')
   686→   686→    case 'mkdir':
   687→   687→      return { success: true, message: `mkdir not needed — write will create parent dirs` }
   688→   688→    default:
   689→   689→      return { success: false, error: `Unknown opencode action: ${action}` }
   690→   690→  }
   691→   691→}

   693→   693→/**
   694→   694→ * Read a file from the project directory.
   695→   695→ * Security: Only allows reading files within the project root.
   696→   696→ * Uses path.resolve + startsWith check to prevent path traversal.
   697→   697→ */
   698→   698→async function executeFileRead(filePath: string): Promise<{ success: boolean; content?: string; error?: string }> {
   699→   699→  if (!filePath) return { success: false, error: 'Path is required' }
   700→   700→
   701→   701→  try {
   702→   702→    const pathModule = await import('path')
   703→   703→    const projectRoot = process.cwd()
   704→   704→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   705→   705→
   706→   706→    // Security: Prevent path traversal — resolved path must be within project root
   707→   707→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   708→   708→      return { success: false, error: `Access denied: path escapes project root` }
   709→   709→    }
   710→   710→
   711→   711→    const fs = await import('fs/promises')
   712→   712→    const content = await fs.readFile(resolvedPath, 'utf-8')
   713→   713→
   714→   714→    // Limit output to prevent overwhelming LLM context
   715→   715→    const maxChars = 50000
   716→   716→    const truncated = content.length > maxChars
   717→   717→    const output = truncated ? content.slice(0, maxChars) + '\n\n... [TRUNCATED — file too large]' : content
   718→   718→
   719→   719→    return { success: true, content: output }
   720→   720→  } catch (err) {
   721→   721→    return {
   722→   722→      success: false,
   723→   723→      error: `Failed to read ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   724→   724→    }
   725→   725→  }
   726→   726→}
   727→   727→
   728→   728→/**
   729→   729→ * Write a file to the project directory.
   730→   730→ * Security: Only allows writing within the project root. Uses path.resolve to prevent traversal.
   731→   731→ * Auto-creates parent directories.
   732→   732→ */
   733→   733→async function executeFileWrite(filePath: string, content: string): Promise<{ success: boolean; message: string; error?: string }> {
   734→   734→  if (!filePath) return { success: false, message: '', error: 'Path is required' }
   735→   735→  if (!content) return { success: false, message: '', error: 'Content is required' }
   736→   736→
   737→   737→  try {
   738→   738→    const pathModule = await import('path')
   739→   739→    const projectRoot = process.cwd()
   740→   740→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   741→   741→
   742→   742→    // Security: Prevent path traversal — resolved path must be within project root
   743→   743→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   744→   744→      return { success: false, message: '', error: 'Access denied: path escapes project root' }
   745→   745→    }
   746→   746→
   747→   747→    const fs = await import('fs/promises')
   748→   748→
   749→   749→    // Auto-create parent directories
   750→   750→    const dir = pathModule.dirname(resolvedPath)
   751→   751→    await fs.mkdir(dir, { recursive: true })
   752→   752→
   753→   753→    // Write file
   754→   754→    await fs.writeFile(resolvedPath, content, 'utf-8')
   755→   755→
   756→   756→    return { success: true, message: `File written: ${filePath} (${content.length} chars)` }
   757→   757→  } catch (err) {
   758→   758→    return {
   759→   759→      success: false,
   760→   760→      message: '',
   761→   761→      error: `Failed to write ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   762→   762→    }
   763→   763→  }
   764→   764→}

   766→   766→/**
   767→   767→ * Execute a bash command safely.
   768→   768→ * Security: Only allows pre-approved commands.
   769→   769→ */
   770→   770→async function executeBash(command: string): Promise<{ success: boolean; output?: string; error?: string }> {
   771→   771→  if (!command) return { success: false, error: 'Command is required' }
   772→   772→
   773→   773→  // Security: Whitelist of allowed commands
   774→   774→  // NOTE: 'node' is excluded because `node -e "..."` can execute arbitrary code.
   775→   775→  //       'cat'/'head'/'tail'/'grep' are excluded — use opencode read instead.
   776→   776→  //       'npx'/'bunx' are excluded — can download+execute arbitrary npm packages.
   777→   777→  const allowedPrefixes = [
   778→   778→    'npm run',
   779→   779→    'npm test',
   780→   780→    'npm lint',
   781→   781→    'bun run',
   782→   782→    'bun test',
   783→   783→    'bun lint',
   784→   784→    'tsc',
   785→   785→    'eslint',
   786→   786→    'prettier',
   787→   787→    'echo',
   788→   788→    'ls',
   789→   789→    'wc',
   790→   790→    'rg',       // ripgrep — search only
   791→   791→    'which',
   792→   792→    'pwd',
   793→   793→    'prisma',
   794→   794→    'git status',
   795→   795→    'git diff',
   796→   796→    'git log',
   797→   797→  ]
   798→   798→
   799→   799→  // Block dangerous flags that could allow code execution or file access
   800→   800→  const dangerousFlags = ['-e ', '--eval ', '-r ', '--require ', '-p ', '--print ']
   801→   801→  const normalizedCmd = command.trim()
   802→   802→  const isAllowed = allowedPrefixes.some(prefix => normalizedCmd.startsWith(prefix))
   803→   803→  const usesDangerousFlag = dangerousFlags.some(flag => normalizedCmd.includes(flag))
   804→   804→
   805→   805→  // Security: Block shell metacharacters that enable command chaining/injection
   806→   806→  // These allow bypassing the whitelist by chaining additional commands:
   807→   807→  //   "npm run build; rm -rf /"   →  ;  chains commands
   808→   808→  //   "git status && cat /etc/p"  →  && chains on success
   809→   809→  //   "echo hi | bash"            →  |  pipes output
   810→   810→  //   "tsc || rm -rf /"           →  || chains on failure
   811→   811→  //   "echo $(cat /etc/passwd)"   →  $() command substitution
   812→   812→  //   "echo `cat /etc/passwd`"    →  backtick command substitution
   813→   813→  //   "echo hi > /etc/passwd"     →  >  redirect write
   814→   814→  //   "echo hi >> /etc/passwd"    →  >> redirect append
   815→   815→  const shellMetacharacters = /(;|&&|\|\||\||`|\$\(|\$\{|>|>>)/
   816→   816→  const usesShellMetacharacters = shellMetacharacters.test(normalizedCmd)
   817→   817→
   818→   818→  if (!isAllowed || usesDangerousFlag || usesShellMetacharacters) {
   819→   819→    return {
   820→   820→      success: false,
   821→   821→      error: `Command not allowed: "${normalizedCmd.slice(0, 100)}". ${!isAllowed ? 'Prefix not in whitelist.' : ''} ${usesDangerousFlag ? 'Dangerous flag detected.' : ''} ${usesShellMetacharacters ? 'Shell metacharacter (;, &&, ||, |, `, $(), >, >>) detected — command chaining is forbidden.' : ''}`,
   822→   822→    }
   823→   823→  }
   824→   824→
   825→   825→  try {
   826→   826→    const { exec } = await import('child_process')
   827→   827→    const { promisify } = await import('util')
   828→   828→    const execAsync = promisify(exec)
   829→   829→
   830→   830→    const { stdout, stderr } = await execAsync(normalizedCmd, {
   831→   831→      cwd: process.cwd(),
   832→   832→      timeout: 30000,
   833→   833→      maxBuffer: 1024 * 1024, // 1MB
   834→   834→    })
   835→   835→
   836→   836→    const output = [
   837→   837→      stdout ? `STDOUT:\n${stdout}` : '',
   838→   838→      stderr ? `STDERR:\n${stderr}` : '',
   839→   839→    ].filter(Boolean).join('\n')
   840→   840→
   841→   841→    return { success: true, output: output || '(no output)' }
   842→   842→  } catch (err) {
   843→   843→    const error = err as { stdout?: string; stderr?: string; message?: string }
   844→   844→    const output = [
   845→   845→      error.stdout ? `STDOUT:\n${error.stdout}` : '',
   846→   846→      error.stderr ? `STDERR:\n${error.stderr}` : '',
   847→   847→    ].filter(Boolean).join('\n')
   848→   848→
   849→   849→    return {
   850→   850→      success: false,
   851→   851→      output: output || undefined,
   852→   852→      error: `Command failed: ${error.message}`,
   853→   853→    }
   854→   854→  }
   855→   855→}

   881→   881→/**
   882→   882→ * Knowledge Graph — Neo4j Cypher query (read-only)
   883→   883→ * Enforces read-only at client level BEFORE sending to knowledge-bridge.
   884→   884→ * Only MATCH, RETURN, WHERE, WITH, ORDER BY, LIMIT, SKIP, DISTINCT are allowed.

   886→   886→async function executeKnowledgeGraph(args: Record<string, unknown>): Promise<unknown> {
   887→   887→  const query = args.query as string
   888→   888→
   889→   889→  if (!query) return { error: 'Cypher query is required' }
   890→   890→
   891→   891→  // Security: Enforce read-only at client level
   892→   892→  // Block any write/mutation Cypher keywords (case-insensitive)
   893→   893→  const writeKeywords = /\b(CREATE|DELETE|SET|MERGE|DROP|REMOVE|DETACH|CALL\s*\{.*\}\s*IN\s+TRANSACTION)\b/i
   894→   894→  if (writeKeywords.test(query)) {
   895→   895→    return { error: 'Only read-only Cypher queries (MATCH) are allowed. Write operations (CREATE, DELETE, SET, MERGE, DROP, REMOVE) are forbidden.' }
   896→   896→  }
   897→   897→
   898→   898→  try {
   899→   899→    const { agentGraphQuery } = await import('@/lib/knowledge-bridge')
   900→   900→    const result = await agentGraphQuery(query)
   901→   901→
   902→   902→    return {
   903→   903→      records: result.records,
   904→   904→      count: result.count,
   905→   905→      error: result.error,
   906→   906→    }
   907→   907→  } catch (err) {
   908→   908→    return { error: `Graph query failed: ${err instanceof Error ? err.message : String(err)}` }
   909→   909→  }
   910→   910→}

   912→   912→/**
   913→   913→ * Knowledge Write — Write entity/relationship to KB
   914→   914→ * Uses knowledge-bridge which writes to SQLite buffer (syncs to Neo4j later).
   915→   915→ * Only G3 (CATALYST) has this permission.
   916→   916→ */
   917→   917→async function executeKnowledgeWrite(args: Record<string, unknown>): Promise<unknown> {
   918→   918→  const entityName = args.entityName as string
   919→   919→  const entityType = args.entityType as string
   920→   920→  const description = args.description as string
   921→   921→  const category = args.category as string
   922→   922→  const relationshipType = args.relationshipType as string | undefined
   923→   923→  const targetEntityName = args.targetEntityName as string | undefined
   924→   924→
   925→   925→  if (!entityName || !entityType || !description || !category) {
   926→   926→    return { error: 'entityName, entityType, description, and category are required' }
   927→   927→  }
   928→   928→
   929→   929→  try {
   930→   930→    const { agentWriteEntity, agentWriteRelationship } = await import('@/lib/knowledge-bridge')
   931→   931→
   932→   932→    // Write entity
   933→   933→    const entity = await agentWriteEntity({
   934→   934→      entityName,
   935→   935→      entityType,
   936→   936→      description,
   937→   937→      domain: category,
   938→   938→    })
   939→   939→
   940→   940→    // Write relationship if specified
   941→   941→    let relationship: { id: string; type: string } | null = null
   942→   942→    if (relationshipType && targetEntityName) {
   943→   943→      const rel = await agentWriteRelationship({

  1433→  1433→// ==================== REACT LOOP HELPER ====================
  1434→  1434→
  1435→  1435→/**
  1436→  1436→ * Process a single LLM response — check for tool_calls and execute them.
  1437→  1437→ * Returns array of tool execution results for feeding back to LLM.
  1438→  1438→ *
  1439→  1439→ * This is the core of the ReAct loop:
  1440→  1440→ * 1. LLM returns response with tool_calls
  1441→  1441→ * 2. We execute each tool_call
  1442→  1442→ * 3. Results are fed back as tool messages
  1443→  1443→ * 4. LLM generates next response (may have more tool_calls or final text)
  1444→  1444→ */
  1445→  1445→export async function processToolCalls(
  1446→  1446→  toolCalls: Array<{ id: string; function: { name: string; arguments: string } }>,
  1447→  1447→): Promise<Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }>> {
  1448→  1448→  const results: Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }> = []
  1449→  1449→
  1450→  1450→  for (const tc of toolCalls) {
  1451→  1451→    let args: Record<string, unknown> = {}
  1452→  1452→    try {
  1453→  1453→      args = JSON.parse(tc.function.arguments)
  1454→  1454→    } catch {
  1455→  1455→      args = { _parseError: tc.function.arguments }
  1456→  1456→    }
  1457→  1457→
  1458→  1458→    console.log(`[ToolExecutor] Executing: ${tc.function.name}(${JSON.stringify(args).slice(0, 200)})`)
  1459→  1459→
  1460→  1460→    const result = await executeTool(tc.function.name, args)
  1461→  1461→
  1462→  1462→    results.push({
  1463→  1463→      toolCallId: tc.id,
  1464→  1464→      toolName: tc.function.name,
  1465→  1465→      result,
  1466→  1466→    })
  1467→  1467→
  1468→  1468→    console.log(`[ToolExecutor] Result: ${tc.function.name} → ${result.success ? 'OK' : 'FAIL'} (${result.duration}ms)`)
  1469→  1469→  }
  1470→  1470→
  1471→  1471→  return results
  1472→  1472→}
  1473→  1473→
  1474→  1474→/**
  1475→  1475→ * Format tool results into messages for the LLM.
  1476→  1476→ * Each tool result becomes a "tool" role message.
  1477→  1477→ * Note: Duration is excluded from LLM messages to save tokens — LLM doesn't need timing info.
  1478→  1478→ */
  1479→  1479→export function formatToolResultsAsMessages(
  1480→  1480→  results: Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }>,

  1482→  1482→  return results.map(r => ({

  1487→  1487→      success: r.result.success,
  1488→  1488→      result: r.result.result,
  1489→  1489→    }),
  1490→  1490→  }))
  1491→  1491→}
