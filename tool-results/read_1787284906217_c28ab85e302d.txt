   497→   497→/// Embedding cache — stores previously computed embeddings to avoid redundant API calls
   498→   498→/// Survives server restarts (unlike in-memory cache)
   499→   499→/// IMPORTANT: Same text gets DIFFERENT embeddings for "query" vs "passage" (NVIDIA input_type)
   500→   500→/// So the unique key is (hash, inputType) — NOT just hash
   501→   501→model EmbeddingCache {
   502→   502→  id        String   @id @default(cuid())
   503→   503→  hash     String   /// SHA-256 hash of the input text
   504→   504→  inputType String  @default("passage") /// "query" | "passage" — different embeddings for same text
   505→   505→  model    String   /// Model used to generate the embedding (e.g. "nvidia/llama-nemotron-embed-1b-v2")
   506→   506→  vector   String   /// JSON array of the embedding vector (1536 floats)
   507→   507→  hitCount Int      @default(1) /// Number of cache hits (for LRU eviction)
   508→   508→  createdAt DateTime @default(now())
   509→   509→  lastHitAt DateTime @default(now()) /// Last time this cache entry was used
   510→   510→  expiresAt DateTime /// When this cache entry expires
   511→   511→
   512→   512→  @@unique([hash, inputType]) /// Composite key — same text has separate query & passage embeddings
   513→   513→  @@index([expiresAt])
   514→   514→  @@index([lastHitAt])
   515→   515→}
