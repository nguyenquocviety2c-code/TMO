    41→  const now = Date.now()
    42→  if (now - gatewayCacheTime < GATEWAY_CACHE_TTL) return gatewayOnlineCached

    53→function isTrivialExchange(userMsg: string, assistantMsg: string): boolean {
    54→  // Skip memory extraction for trivial exchanges (greetings, short confirmations)
    55→  if (userMsg.trim().length < 15 && assistantMsg.trim().length < 50) return true

    65→/**
    66→ * Run a ReAct (Reason-Act-Observe) loop for an agent with tool execution.
    67→ *
    68→ * This enables standalone agents like Omega to actually USE their declared tools
    69→ * (opencode, knowledge_search, tavily, etc.) instead of just having them in prompts.
    70→ *
    71→ * Flow:
    72→ * 1. Call LLM with system prompt + user message
    73→ * 2. Parse tool_call patterns from LLM output
    74→ * 3. If tool calls found → execute → feed results back → repeat
    75→ * 4. If no tool calls → return final answer
    76→ * 5. Max 5 iterations to prevent infinite loops
    77→ */
    78→async function runAgentReActLoop(params: {
    79→  systemPrompt: string
    80→  userMessage: string
    81→  chatHistory: Array<{ role: string; content: string }>
    82→  agentProvider: string
    83→  agentModel: string
    84→  agentId: string
    85→  agentName: string
    86→  agentTemperature?: number
    87→  agentMaxTokens?: number
    88→  agentTools?: string[]
    89→  maxIterations?: number
    90→}): Promise<{ content: string; iterations: number; toolCallsExecuted: number }> {
    91→  const {
    92→    systemPrompt,
    93→    userMessage,
    94→    chatHistory,
    95→    agentProvider,
    96→    agentModel,
    97→    agentId,
    98→    agentName,
    99→    agentTemperature,
   100→    agentMaxTokens,
   101→    agentTools = [],
   102→    maxIterations = 5,
   103→  } = params
   104→
   105→  // Build messages array for multi-turn conversation using ChatMessageForLLM type
   106→  // This preserves structured tool_calls with tool_call_id for proper multi-turn tool calling
   107→  const messages: ChatMessageForLLM[] = [
   108→    { role: 'system', content: systemPrompt },
   109→    // Include last 6 messages from chat history for context
   110→    ...chatHistory.slice(-6).map(m => ({
   111→      role: m.role as 'user' | 'assistant',
   112→      content: m.content,
   113→    })),
   114→    { role: 'user', content: userMessage },
   115→  ]
   116→
   117→  let finalContent = ''
   118→  let iterations = 0
   119→  let totalToolCalls = 0
   120→
   121→  while (iterations < maxIterations) {
   122→    iterations++
   123→
   124→    // Get tool definitions for this agent (for structured tool calling)
   125→    const toolDefs = agentTools.length > 0 ? getToolDefinitionsForLLM(agentTools) : []
   126→
   127→    // Call LLM with full messages array — preserves tool_calls structure (CRITICAL FIX)
   128→    // Previously used formatMessagesForLLM() which flattened tool_calls to text, losing tool_call_id
   129→    const result = await callLLMForAgentWithMessages(
   130→      messages,
   131→      { provider: agentProvider, model: agentModel },
   132→      {
   133→        agentId,
   134→        agentName,
   135→        temperature: agentTemperature,
   136→        maxTokens: agentMaxTokens,
   137→        tools: toolDefs.length > 0 ? toolDefs : undefined,
   138→      }
   139→    )
   140→
   141→    if (!result || (result.error && !result.content && !result.toolCalls?.length)) {
   142→      finalContent = result?.error || 'LLM call failed'
   143→      break
   144→    }
   145→
   146→    finalContent = result.content || ''
   147→
   148→    // Parse tool calls — prefer structured tool_calls from LLM, fallback to regex
   149→    let toolCalls: Array<{ id: string; name: string; args: Record<string, unknown> }> = []
   150→
   151→    if (result.toolCalls && result.toolCalls.length > 0) {
   152→      // ✅ Structured tool_calls from LLM (preferred — P0.4)
   153→      toolCalls = result.toolCalls.map(tc => ({
   154→        id: tc.id,
   155→        name: tc.function.name,
   156→        args: (() => { try { return JSON.parse(tc.function.arguments) } catch { return {} } })(),
   157→      }))
   158→      console.log(`[ReAct] ${agentName} structured tool_calls: ${toolCalls.map(tc => tc.name).join(', ')}`)
   159→    } else {
   160→      // ❌ Fallback regex parsing (for models that don't support tool calling)
   161→      toolCalls = parseToolCallsFromOutput(result.content)
   162→    }
   163→
   164→    if (toolCalls.length === 0) {
   165→      // No tool calls → agent is done
   166→      break
   167→    }
   168→
   169→    // Execute tool calls
   170→    console.log(`[ReAct] ${agentName} calling ${toolCalls.length} tools (iteration ${iterations})`)
   171→
   172→    // Add assistant message with tool calls to conversation
   173→    messages.push({
   174→      role: 'assistant',
   175→      content: result.content,
   176→      ...(result.toolCalls && result.toolCalls.length > 0 ? {
   177→        tool_calls: result.toolCalls,
   178→      } : {}),
   179→    })
   180→
   181→    for (const tc of toolCalls) {
   182→      totalToolCalls++
   183→
   184→      // Only execute tools that the agent has permission for
   185→      if (agentTools.length > 0 && !agentTools.includes(tc.name)) {
   186→        messages.push({
   187→          role: 'tool',
   188→          content: JSON.stringify({ tool: tc.name, success: false, result: `Tool "${tc.name}" not available for this agent. Available tools: ${agentTools.join(', ')}` }),
   189→          tool_call_id: tc.id,
   190→        })
   191→        continue
   192→      }
   193→
   194→      console.log(`[ReAct] Executing: ${tc.name}(${JSON.stringify(tc.args).slice(0, 200)})`)
   195→
   196→      const toolResult = await executeTool(tc.name, tc.args)
   197→
   198→      // Add tool result to conversation (with tool_call_id for OpenAI format)
   199→      messages.push({
   200→        role: 'tool',
   201→        content: JSON.stringify({
   202→          tool: tc.name,
   203→          success: toolResult.success,
   204→          result: toolResult.result,
   205→        }),
   206→        tool_call_id: tc.id,
   207→      })
   208→    }
   209→
   210→    // If max iterations reached, break
   211→    if (iterations >= maxIterations) {
   212→      finalContent += `\n\n⚠️ Đạt max ${maxIterations} vòng iteration.`
   213→      break
   214→    }
   215→  }
   216→
   217→  console.log(`[ReAct] ${agentName} done: ${iterations} iterations, ${totalToolCalls} tool calls`)
   218→  return { content: finalContent, iterations, toolCallsExecuted: totalToolCalls }
   219→}

   493→async function routeToOpenCode(
   494→  userMessage: string,
   495→  model?: string,
   496→  sessionId?: string,
   497→  messages?: Array<{ role: string; content: string }>,
   498→  agentProfileId?: string,
   499→  teamMode?: string,
   500→  teamName?: string,
   501→): Promise<NextResponse | null> {
   502→  try {
   503→    const online = await isOpenCodeOnline()
   504→    if (!online) return null
   505→
   506→    // OC-3.1: Enrich with Knowledge Base context before creating session
   507→    const codeConfidence = detectCodeConfidence(userMessage)
   508→    const enrichment = await enrichCodeContext(userMessage)
   509→    
   510→    // OC-3.5: Cross-Session Learning — incorporate past corrections and insights
   511→    let entityTypeCount = 0
   512→    let documentCount = 0  
   513→    let correctionCount = 0
   514→    let insightCount = 0
   515→    let filesInWorkspace = 0
   516→    try {
   517→      entityTypeCount = await db.localEntity.count()
   518→      correctionCount = await db.agentCorrection.count({ where: { applied: true } })
   519→      insightCount = await db.agentInsight.count()
   520→      documentCount = await db.localEntity.count({ where: { domain: 'document' } })
   521→    } catch {}
   522→
   523→    // Count workspace files (cross-platform: uses Node.js fs instead of `find` command)
   524→    try {
   525→      const { readdirSync, statSync } = await import('fs')
   526→      const { join } = await import('path')
   527→      const workspaceDir = process.env.OPENCODE_WORKSPACE || process.cwd()
   528→      const skipDirs = new Set(['.next', 'node_modules', '.git', 'qdrant-storage', 'db', 'upload', 'skills'])
   529→      function countFiles(dir: string): number {
   530→        let count = 0
   531→        try {
   532→          for (const entry of readdirSync(dir)) {
   533→            const full = join(dir, entry)
   534→            try {
   535→              const stat = statSync(full)
   536→              if (stat.isDirectory()) {
   537→                if (!skipDirs.has(entry)) count += countFiles(full)
   538→              } else {
   539→                count++
   540→              }
   541→            } catch {}
   542→          }
   543→        } catch {}
   544→        return count
   545→      }
   546→      filesInWorkspace = countFiles(workspaceDir)
   547→    } catch {}
   548→
   549→    // OC-3.7: Generate enriched system prompt with KB context
   550→    const enrichedSystemPrompt = generateOpenCodeSystemPrompt({
   551→      entityTypeCount,
   552→      documentCount,
   553→      correctionCount,
   554→      insightCount,
   555→      filesInWorkspace,
   556→      modelList: [],
   557→      kbEnabled: true,
   558→      mcpToolsEnabled: ['knowledge_search', 'knowledge_graph', 'knowledge_write', 'web_search'],

   646→/**
   647→ * Route chat through hybrid Gateway + local architecture.
   648→ *
   649→ * Architecture (Hybrid ReAct):
   650→ *   - LLM calls → NVIDIA NIM directly (callLLMForAgent)
   651→ *   - Tool execution → Gateway /tools/invoke when available, local fallback
   652→ *
   653→ * This replaces the old approach of routing ALL through Gateway /v1/chat/completions
   654→ * which returned 404 (Gateway uses WebSocket internally, not REST for chat).
   655→ */
   656→async function routeThroughGateway(params: {
   657→  messages: Array<{ role: string; content: string }>
   658→  agentProvider: string
   659→  agentModel: string
   660→  sessionId?: string
   661→  temperature?: number
   662→  maxTokens?: number
   663→  agentProfileId?: string
   664→  agentProfileName?: string
   665→  teamMode?: string
   666→  teamName?: string
   667→  agentTools?: string[]
   668→}): Promise<NextResponse | null> {
   669→  const { messages, agentProvider, agentModel, sessionId, temperature, maxTokens,
   670→          agentProfileId, agentProfileName, teamMode, teamName, agentTools = [] } = params
   671→
   672→  try {
   673→    // Hybrid approach: Use local ReAct loop with Gateway tool proxy
   674→    // This avoids the 404 from /v1/chat/completions while still using Gateway tools
   675→    const lastUserMsg = [...messages].reverse().find(m => m.role === 'user')
   676→    const userContent = typeof lastUserMsg?.content === 'string' ? lastUserMsg.content : ''
   677→    const systemMsg = messages.find(m => m.role === 'system')
   678→    const systemPrompt = systemMsg?.content || ''
   679→    const chatHistory = messages.filter(m => m.role === 'user' || m.role === 'assistant')
   680→
   681→    // Use the existing ReAct loop which already supports Gateway tool proxy
   682→    const reactResult = await runAgentReActLoop({
   683→      systemPrompt,
   684→      userMessage: userContent,
   685→      chatHistory,
   686→      agentProvider,
   687→      agentModel,
   688→      agentId: agentProfileId || 'unknown',
   689→      agentName: agentProfileName || 'Agent',
   690→      agentTemperature: temperature,
   691→      agentMaxTokens: maxTokens,
   692→      agentTools,
   693→      maxIterations: 5,
   694→    })
   695→
   696→    // Track session
   697→    if (sessionId) {
   698→      await db.agentSession.upsert({
   699→        where: { sessionId },
   700→        create: {
   701→          sessionId,
   702→          model: agentModel,
   703→          provider: agentProvider,
   704→          title: userContent.slice(0, 50) || 'New Chat',
   705→          messageCount: messages.length + 1,
   706→          agentProfileId: agentProfileId || null,
   707→          teamMode: teamMode || null,
   708→          teamName: teamName || null,
   709→        },
   710→        update: { messageCount: { increment: 1 } },
   711→      }).catch(() => {})
   712→    }
   713→
   714→    return NextResponse.json({
   715→      content: reactResult.content,
   716→      model: agentModel,
   717→      provider: agentProvider,
   718→      sessionId: sessionId || `session-${Date.now()}`,
   719→      agentProfileId: agentProfileId || undefined,
   720→      sources: [],
   721→      confidence: 0.7,
   722→      kbResults: 'hybrid-react',
   723→      gatewayToolsAvailable: true,
   724→      reactIterations: reactResult.iterations,
   725→      toolCallsExecuted: reactResult.toolCallsExecuted,
   726→    })
   727→  } catch (err) {
   728→    console.warn('[Gateway Hybrid] Error:', err instanceof Error ? err.message : String(err))
   729→    return null
   730→  }
   731→}

  1120→

  1136→            // Skip meta-cognitive reasoning for chat flow — reduces latency by ~15s

  1150→      if (queryRes.ok) {

  1153→
  1154→        // If this was a code query but OpenCode was offline, add a note

  1254→    return NextResponse.json({
