    80→/**
    81→ * Check if a tool is permitted for an agent based on ToolPermission records.
    82→ *
    83→ * Logic:
    84→ * - No explicit permission → default allow (tools are allowed unless explicitly denied)
    85→ * - permission = "deny" → blocked
    86→ * - permission = "ask" → blocked in auto mode (would need user interaction to approve)
    87→ * - permission = "allow" → check rate limit, update usage tracking
    88→ */
    89→async function checkToolPermission(agentId: string, toolName: string): Promise<{ allowed: boolean; reason?: string }> {
    90→  try {
    91→    const perm = await db.toolPermission.findUnique({
    92→      where: { agentId_toolName: { agentId, toolName } },
    93→    })
    94→
    95→    if (!perm) {
    96→      // No explicit permission → default allow (tools are allowed unless explicitly denied)
    97→      return { allowed: true }
    98→    }
    99→
   100→    if (perm.permission === 'deny') {
   101→      return { allowed: false, reason: `Tool "${toolName}" is denied for this agent` }
   102→    }
   103→
   104→    if (perm.permission === 'ask') {
   105→      // In auto mode, "ask" means deny (would need user interaction to approve)
   106→      return { allowed: false, reason: `Tool "${toolName}" requires user approval` }
   107→    }
   108→
   109→    // Check rate limit — reset counter if last use was more than 1 hour ago
   110→    if (perm.maxCallsPerHour) {
   111→      const oneHourAgo = new Date(Date.now() - 3600000)
   112→      if (perm.lastUsedAt && perm.lastUsedAt < oneHourAgo) {
   113→        // Last use was over an hour ago — reset counter (new hour window)
   114→        await db.toolPermission.update({
   115→          where: { agentId_toolName: { agentId, toolName } },
   116→          data: { callCount: 0 },
   117→        }).catch(() => {})
   118→      } else if (perm.callCount >= perm.maxCallsPerHour) {
   119→        // Within the current hour window and exceeded limit
   120→        return { allowed: false, reason: `Tool "${toolName}" exceeded rate limit (${perm.maxCallsPerHour}/hour)` }
   121→      }
   122→    }
   123→
   124→    // Update usage tracking
   125→    await db.toolPermission.update({
   126→      where: { agentId_toolName: { agentId, toolName } },
   127→      data: {
   128→        lastUsedAt: new Date(),
   129→        callCount: { increment: 1 },
   130→      },
   131→    }).catch(() => {})
   132→
   133→    return { allowed: true }
   134→  } catch {
   135→    // Permission check failed → default allow
   136→    return { allowed: true }
   137→  }
   138→}

   144→/**
   145→ * Run a ReAct (Reason-Act-Observe) loop for an agent with tool execution.
   146→ *
   147→ * This enables standalone agents like Omega to actually USE their declared tools
   148→ * (opencode, knowledge_search, tavily, etc.) instead of just having them in prompts.
   149→ *
   150→ * Flow:
   151→ * 1. Call LLM with system prompt + user message
   152→ * 2. Parse tool_call patterns from LLM output
   153→ * 3. If tool calls found → execute → feed results back → repeat
   154→ * 4. If no tool calls → return final answer
   155→ * 5. Max 5 iterations to prevent infinite loops
   156→ */
   157→async function runAgentReActLoop(params: {
   158→  systemPrompt: string
   159→  userMessage: string
   160→  chatHistory: Array<{ role: string; content: string }>
   161→  agentProvider: string
   162→  agentModel: string
   163→  agentId: string
   164→  agentName: string
   165→  agentTemperature?: number
   166→  agentMaxTokens?: number
   167→  agentTools?: string[]
   168→  maxIterations?: number
   169→}): Promise<{ content: string; iterations: number; toolCallsExecuted: number }> {
   170→  const {
   171→    systemPrompt,
   172→    userMessage,
   173→    chatHistory,
   174→    agentProvider,
   175→    agentModel,
   176→    agentId,
   177→    agentName,
   178→    agentTemperature,
   179→    agentMaxTokens,
   180→    agentTools = [],
   181→    maxIterations = 5,
   182→  } = params
   183→
   184→  // Overall timeout for the entire ReAct loop — prevents runaway loops
   185→  const REACT_LOOP_TIMEOUT_MS = 45000  // 45s max for entire loop
   186→  const loopStartTime = Date.now()
   187→
   188→  // Build messages array for multi-turn conversation using ChatMessageForLLM type
   189→  // This preserves structured tool_calls with tool_call_id for proper multi-turn tool calling
   190→  const messages: ChatMessageForLLM[] = [
   191→    { role: 'system', content: systemPrompt },
   192→    // Include last 6 messages from chat history for context
   193→    ...chatHistory.slice(-6).map(m => ({
   194→      role: m.role as 'user' | 'assistant',
   195→      content: m.content,
   196→    })),
   197→    { role: 'user', content: userMessage },
   198→  ]
   199→
   200→  let finalContent = ''
   201→  let iterations = 0
   202→  let totalToolCalls = 0
   203→
   204→  // Cache tool definitions — avoid recomputing every iteration (saves ~5ms per iteration)
   205→  const cachedToolDefs = agentTools.length > 0 ? getToolDefinitionsForLLM(agentTools) : []
   206→
   207→  while (iterations < maxIterations) {
   208→    iterations++
   209→
   210→    // Check overall loop timeout
   211→    if (Date.now() - loopStartTime > REACT_LOOP_TIMEOUT_MS) {
   212→      finalContent += `\n\n⏱️ ReAct loop timed out after ${Math.round((Date.now() - loopStartTime) / 1000)}s.`
   213→      break
   214→    }
   215→
   216→    // Call LLM with full messages array — preserves tool_calls structure (CRITICAL FIX)
   217→    // Previously used formatMessagesForLLM() which flattened tool_calls to text, losing tool_call_id
   218→    const result = await callLLMForAgentWithMessages(
   219→      messages,
   220→      { provider: agentProvider, model: agentModel },
   221→      {
   222→        agentId,
   223→        agentName,
   224→        temperature: agentTemperature,
   225→        maxTokens: agentMaxTokens,
   226→        tools: cachedToolDefs.length > 0 ? cachedToolDefs : undefined,
   227→      }
   228→    )
   229→
   230→    if (!result || (result.error && !result.content && !result.toolCalls?.length)) {
   231→      finalContent = result?.error || 'LLM call failed'
   232→      break
   233→    }
   234→
   235→    finalContent = result.content || ''
   236→
   237→    // Parse tool calls — prefer structured tool_calls from LLM, fallback to regex
   238→    let toolCalls: Array<{ id: string; name: string; args: Record<string, unknown> }> = []
   239→
   240→    if (result.toolCalls && result.toolCalls.length > 0) {
   241→      // ✅ Structured tool_calls from LLM (preferred — P0.4)
   242→      toolCalls = result.toolCalls.map(tc => ({
   243→        id: tc.id,
   244→        name: tc.function.name,
   245→        args: (() => { try { return JSON.parse(tc.function.arguments) } catch { return {} } })(),
   246→      }))
   247→      console.log(`[ReAct] ${agentName} structured tool_calls: ${toolCalls.map(tc => tc.name).join(', ')}`)
   248→    } else {
   249→      // ❌ Fallback regex parsing (for models that don't support tool calling)
   250→      toolCalls = parseToolCallsFromOutput(result.content)
   251→    }
   252→
   253→    if (toolCalls.length === 0) {
   254→      // No tool calls → agent is done
   255→      break
   256→    }
   257→
   258→    // Execute tool calls
   259→    console.log(`[ReAct] ${agentName} calling ${toolCalls.length} tools (iteration ${iterations})`)
   260→
   261→    // Add assistant message with tool calls to conversation
   262→    messages.push({
   263→      role: 'assistant',
   264→      content: result.content,
   265→      ...(result.toolCalls && result.toolCalls.length > 0 ? {
   266→        tool_calls: result.toolCalls,
   267→      } : {}),
   268→    })
   269→
   270→    // PARALLEL TOOL EXECUTION — execute multiple tools concurrently instead of sequentially
   271→    // This reduces latency when LLM returns multiple tool calls in one response
   272→    const toolPromises = toolCalls.map(async (tc) => {
   273→      // Only execute tools that the agent has permission for
   274→      if (agentTools.length > 0 && !agentTools.includes(tc.name)) {
   275→        return {
   276→          id: tc.id,
   277→          name: tc.name,
   278→          content: JSON.stringify({ tool: tc.name, success: false, result: `Tool "${tc.name}" not available for this agent. Available tools: ${agentTools.join(', ')}` }),
   279→        }
   280→      }
   281→
   282→      // P2.2: Check tool permission before executing
   283→      const permCheck = await checkToolPermission(agentId || 'default', tc.name)
   284→      if (!permCheck.allowed) {
   285→        return {
   286→          id: tc.id,
   287→          name: tc.name,
   288→          content: JSON.stringify({
   289→            tool: tc.name,
   290→            success: false,
   291→            result: permCheck.reason || `Tool "${tc.name}" is not permitted for this agent`,
   292→          }),
   293→        }
   294→      }
   295→
   296→      console.log(`[ReAct] Executing: ${tc.name}(${JSON.stringify(tc.args).slice(0, 200)})`)
   297→
   298→      const toolResult = await executeTool(tc.name, tc.args)
   299→      return {
   300→        id: tc.id,
   301→        name: tc.name,
   302→        content: JSON.stringify({
   303→          tool: tc.name,
   304→          success: toolResult.success,
   305→          result: toolResult.result,
   306→        }),
   307→      }
   308→    })
   309→
   310→    const toolResults = await Promise.all(toolPromises)
   311→    totalToolCalls += toolResults.length
   312→
   313→    // Add all tool results to messages
   314→    for (const tr of toolResults) {
   315→      messages.push({
   316→        role: 'tool',
   317→        content: tr.content,
   318→        tool_call_id: tr.id,
   319→      })
   320→    }
   321→
   322→    // If max iterations reached, break
   323→    if (iterations >= maxIterations) {
   324→      finalContent += `\n\n⚠️ Đạt max ${maxIterations} vòng iteration.`
   325→      break
   326→    }
   327→  }
   328→
   329→  console.log(`[ReAct] ${agentName} done: ${iterations} iterations, ${totalToolCalls} tool calls`)
   330→  return { content: finalContent, iterations, toolCallsExecuted: totalToolCalls }
   331→}

   659→/**
   660→ * Route chat through hybrid Gateway + local architecture.
   661→ *
   662→ * Architecture (Hybrid ReAct):
   663→ *   - LLM calls → NVIDIA NIM directly (callLLMForAgent)
   664→ *   - Tool execution → Gateway /tools/invoke when available, local fallback
   665→ *
   666→ * This replaces the old approach of routing ALL through Gateway /v1/chat/completions
   667→ * which returned 404 (Gateway uses WebSocket internally, not REST for chat).
   668→ */
   669→async function routeThroughGateway(params: {
   670→  messages: Array<{ role: string; content: string }>
   671→  agentProvider: string
   672→  agentModel: string
   673→  sessionId?: string
   674→  temperature?: number
   675→  maxTokens?: number
   676→  agentProfileId?: string
   677→  agentProfileName?: string
   678→  teamMode?: string
   679→  teamName?: string
   680→  agentTools?: string[]
   681→}): Promise<NextResponse | null> {
   682→  const { messages, agentProvider, agentModel, sessionId, temperature, maxTokens,
   683→          agentProfileId, agentProfileName, teamMode, teamName, agentTools = [] } = params
   684→
   685→  try {
   686→    // Hybrid approach: Use local ReAct loop with Gateway tool proxy
   687→    // This avoids the 404 from /v1/chat/completions while still using Gateway tools
   688→    const lastUserMsg = [...messages].reverse().find(m => m.role === 'user')
   689→    const userContent = typeof lastUserMsg?.content === 'string' ? lastUserMsg.content : ''
   690→    const systemMsg = messages.find(m => m.role === 'system')
   691→    const systemPrompt = systemMsg?.content || ''
   692→    const chatHistory = messages.filter(m => m.role === 'user' || m.role === 'assistant')
   693→
   694→    // Use the existing ReAct loop which already supports Gateway tool proxy
   695→    const reactResult = await runAgentReActLoop({
   696→      systemPrompt,
   697→      userMessage: userContent,
   698→      chatHistory,
   699→      agentProvider,
   700→      agentModel,
   701→      agentId: agentProfileId || 'unknown',
   702→      agentName: agentProfileName || 'Agent',
   703→      agentTemperature: temperature,
   704→      agentMaxTokens: maxTokens,
   705→      agentTools,
   706→      maxIterations: 5,
   707→    })
   708→
   709→    // Track session — validate agentProfileId exists before upsert to avoid FK constraint violation
   710→    if (sessionId) {
   711→      try {
   712→        const validProfileId = await validateAgentProfileId(agentProfileId)
   713→        await db.agentSession.upsert({
   714→          where: { sessionId },
   715→          create: {
   716→            sessionId,
   717→            model: agentModel,
   718→            provider: agentProvider,
   719→            title: userContent.slice(0, 50) || 'New Chat',
   720→            messageCount: messages.length + 1,
   721→            agentProfileId: validProfileId,
   722→            teamMode: teamMode || null,
   723→            teamName: teamName || null,
   724→          },
   725→          update: { messageCount: { increment: 1 } },
   726→        })
   727→      } catch {}
   728→    }
   729→
   730→    return NextResponse.json({
   731→      content: reactResult.content,
   732→      model: agentModel,
   733→      provider: agentProvider,
   734→      sessionId: sessionId || `session-${Date.now()}`,
   735→      agentProfileId: agentProfileId || undefined,
   736→      sources: [],
   737→      confidence: 0.7,
   738→      kbResults: 'hybrid-react',
   739→      gatewayToolsAvailable: true,
   740→      reactIterations: reactResult.iterations,
   741→      toolCallsExecuted: reactResult.toolCallsExecuted,
   742→    })
   743→  } catch (err) {
   744→    console.warn('[Gateway Hybrid] Error:', err instanceof Error ? err.message : String(err))
   745→    return null
   746→  }
   747→}

  1042→              console.warn('[Memory] Periodic decay error:', err instanceof Error ? err.message : String(err))
  1043→            )
  1044→          }
  1045→        }
  1046→
  1047→        return NextResponse.json({
  1048→          content,
  1049→          model: agentModel,
  1050→          provider: agentProvider,
  1051→          sessionId: sessionId || `session-${Date.now()}`,

  1053→          sources: [],
  1054→          confidence: 0.6,
  1055→          kbResults: 'agent-react',
  1056→          reactIterations: reactResult.iterations,
  1057→          toolCallsExecuted: reactResult.toolCallsExecuted,
  1058→          memoriesRecalled: recalledMemories.length || undefined,

  1060→      } catch (err) {
  1061→        console.warn('[OpenClaw Chat] ReAct loop error, falling back to /api/query:', err instanceof Error ? err.message : String(err))
  1062→        // Fall through to /api/query fallback below

  1065→
  1066→    if (!hasAgentProviderModel) {
  1067→      // No agent config → try OpenClaw Gateway first (original flow)

  1069→        const onlineResult = await isGatewayOnline()
  1070→        if (onlineResult.online) {
  1071→          const res = await chatCompletion({
  1072→            messages: fullMessages,
  1073→            model: model || 'openclaw/default',
  1074→            sessionId,
  1075→            stream: false,
  1076→            temperature: agentTemperature,
  1077→            max_tokens: agentMaxTokens,
  1078→            tools: [
  1079→              {
  1080→                type: 'function',
  1081→                function: {
  1082→                  name: 'knowledge_search',
  1083→                  description: 'Tìm kiếm semantic trong Knowledge Base',
  1084→                  parameters: {
  1085→                    type: 'object',
  1086→                    properties: {
  1087→                      query: { type: 'string', description: 'Câu hỏi cần tìm kiếm' },
  1088→                      topK: { type: 'number', description: 'Số kết quả trả về', default: 5 },
  1089→                    },
  1090→                    required: ['query'],
  1091→                  },
  1092→                },
  1093→              },
  1094→            ],
  1095→          })
  1096→
  1097→          const data = await res.json()
  1098→          const content = data.choices?.[0]?.message?.content || data.content || data.message || 'No response'
  1099→          const toolCalls = data.choices?.[0]?.message?.tool_calls
  1100→
  1101→          // Track token usage from gateway response (if available)
  1102→          const gwUsage = data.usage as { prompt_tokens?: number; completion_tokens?: number; total_tokens?: number } | undefined
  1103→          const gwTotalTokens = gwUsage?.total_tokens ?? ((gwUsage?.prompt_tokens ?? 0) + (gwUsage?.completion_tokens ?? 0))
  1104→          if (gwTotalTokens > 0) {
  1105→            addTokensUsed(gwTotalTokens, 'openclaw', undefined, data.model || model)
  1106→          }
  1107→
  1108→          // Update session in SQLite
  1109→          if (sessionId) {
  1110→            try {
  1111→              await db.agentSession.upsert({
  1112→                where: { sessionId },
  1113→                create: {
  1114→                  sessionId,
  1115→                  model: model || 'openclaw/default',
  1116→                  provider: 'openclaw',
  1117→                  title: messages[0]?.content?.slice(0, 50) || 'New Chat',
  1118→                  messageCount: messages.length + 1,
  1119→                  agentProfileId: await validateAgentProfileId(agentProfileId),
  1120→                  teamMode: teamMode || null,
  1121→                  teamName: teamName || null,
  1122→                },
  1123→                update: {
  1124→                  messageCount: { increment: 1 },
  1125→                  model: model || undefined,
  1126→                },
  1127→              })
  1128→            } catch {}
  1129→          }
  1130→
  1131→          return NextResponse.json({
  1132→            content,
  1133→            toolCalls,
  1134→            model: data.model || model,
  1135→            provider: 'openclaw',
  1136→            agentProfileId: agentProfileId || undefined,
  1137→            usage: data.usage,
  1138→            sessionId: sessionId || data.session_id,
  1139→            kbResults: 'gateway',
  1140→          })
  1141→        }
  1142→      } catch (err) {
  1143→        console.warn('[OpenClaw Chat] Gateway error:', err instanceof Error ? err.message : String(err))
  1144→      }
  1145→    }
  1146→
  1147→    // /api/query fallback path — used for:
  1148→    //   - Agent with configured provider/model when ReAct loop fails (fallback)
  1149→    //   - Gateway offline fallback (no agent config)
  1150→    if (hasAgentProviderModel) {
  1151→      console.log(`[OpenClaw Chat] ReAct loop failed, falling back to /api/query for agent (${agentProvider}/${agentModel})`)
  1152→    }
  1153→    try {
  1154→      const baseUrl = process.env.NEXT_PUBLIC_BASE_URL || 'http://localhost:3000'
  1155→      const queryRes = await fetch(`${baseUrl}/api/query`, {
  1156→        method: 'POST',
  1157→        headers: { 'Content-Type': 'application/json' },
  1158→        body: JSON.stringify({
  1159→          query: lastUserContent,
  1160→          options: {
  1161→            chatHistory: messages.filter((m: { role: string }) => m.role === 'user' || m.role === 'assistant').slice(-6).map((m: { role: string; content: string }) => ({ role: m.role, content: m.content })),
  1162→            // Skip meta-cognitive reasoning for chat flow — reduces latency by ~15s
  1163→            // KB results are already enriched with memories in the system prompt
  1164→            skipMetaCog: true,
  1165→            ...(agentTemperature !== undefined ? { temperature: agentTemperature } : {}),
  1166→            ...(agentMaxTokens !== undefined ? { maxTokens: agentMaxTokens } : {}),

  1168→            ...(agentProfileName ? { agentName: agentProfileName } : {}),
  1169→            ...(agentProvider ? { agentProvider } : {}),
  1170→            ...(agentModel ? { agentModel } : {}),
  1171→          },
  1172→        }),
  1173→        signal: AbortSignal.timeout(60000),
  1174→      })
  1175→
  1176→      if (queryRes.ok) {
  1177→        const data = await queryRes.json()
  1178→        const content = data.answer || 'Không thể tạo câu trả lời.'
  1179→
  1180→        // If this was a code query but OpenCode was offline, add a note
  1181→        const codeNote = isCodeQuery(lastUserContent)
  1182→          ? '\n\n⚠️ **OpenCode Server đang offline** — Không thể thực hiện code editing. Kết quả trên chỉ dựa trên Knowledge Base. Hãy bật OpenCode Server để sử dụng đầy đủ tính năng code.'

  1187→        if (sessionId) {

  1189→            const validProfileId = await validateAgentProfileId(agentProfileId)
  1190→            await db.agentSession.upsert({

  1193→                sessionId,

  1198→                agentProfileId: validProfileId,
  1199→                teamMode: teamMode || null,
  1200→                teamName: teamName || null,

  1222→            model: agentModel!,

  1226→
  1227→        // Phase 5: Memory & Learning — save messages + extract memories in background
  1228→        // Non-blocking: fire-and-forget, don't delay the response.

  1230→          // Save chat messages to DB for persistence

  1232→            { role: 'user', content: lastUserContent },
  1233→            { role: 'assistant', content, model: agentModel || model, provider: agentProvider || 'local-fallback', metadata: { confidence: queryConfidence, sources: data.sources?.length || 0 } },
  1234→          ]).catch(err => console.warn('[Memory] Failed to save chat messages:', err instanceof Error ? err.message : String(err)))
  1235→
  1236→          // Extract memories from this conversation (skip trivial exchanges to save LLM tokens)
  1237→          if (!isTrivialExchange(lastUserContent, content)) {

  1239→              agentId: agentProfileId,
  1240→              agentName: agentProfileName || 'unknown',
  1241→              sessionId,
  1242→              userMessage: lastUserContent,

  1246→

  1248→          // This prevents stale memories from accumulating without requiring a cron scheduler
  1249→          memoryDecayCounter++
  1250→          if (memoryDecayCounter >= 20 && agentProfileId) {
  1251→            memoryDecayCounter = 0

  1278→      ? `Xin chào! Tôi là AI Agent của The Magnum Opus. Gateway hiện đang offline nên tôi hoạt động ở chế độ mock. Hãy kết nối Gateway để nhận phản hồi AI thực sự.`

  1281→    return NextResponse.json({

  1288→  } catch (err) {
