   124→// -------------------------------------------------------
   125→// Helper: Extract tool references from SKILL.md content
   126→// -------------------------------------------------------
   127→function extractToolReferences(content: string): string[] {
   128→  const tools = new Set<string>()
   129→
   130→  // Pattern 1: `tool_name` backtick references
   131→  const backtickPattern = /`([a-z_]+)`/g
   132→  let match: RegExpExecArray | null
   133→  while ((match = backtickPattern.exec(content)) !== null) {
   134→    const name = match[1]
   135→    // Check if it looks like a tool name (lowercase with underscores, 2+ chars)
   136→    if (
   137→      name.length >= 2 &&
   138→      (name.includes('_') ||
   139→        ['exec', 'read', 'write', 'edit', 'tts', 'cron', 'image', 'nodes', 'canvas', 'message'].includes(name))
   140→    ) {
   141→      tools.add(name)
   142→    }
   143→  }
   144→
   145→  // Pattern 2: "Call tool_name" or "use tool_name"
   146→  const callPattern = /(?:call|use|invoke)\s+([a-z_]+)/gi
   147→  while ((match = callPattern.exec(content)) !== null) {
   148→    tools.add(match[1])
   149→  }
   150→
   151→  // Filter to only known Gateway tools
   152→  return Array.from(tools).filter(t => KNOWN_TOOLS.includes(t))
   153→}

   155→// -------------------------------------------------------
   156→// Helper: Download and extract a ClawHub skill archive
   157→// -------------------------------------------------------
   158→async function downloadAndExtractArchive(slug: string): Promise<{
   159→  files: string[]
   160→  skillMd: string
   161→  metaJson: Record<string, unknown> | null
   162→  toolReferences: string[]
   163→}> {
   164→  try {
   165→    // Download archive (follow redirects — endpoint returns 307 → ZIP)
   166→    const archiveRes = await fetch(`${CLAWHUB_API}/packages/${slug}/download`, {
   167→      redirect: 'follow',
   168→      signal: AbortSignal.timeout(30000),
   169→    })
   170→    if (!archiveRes.ok) {
   171→      console.warn(`[Skills] Archive download failed for ${slug}: ${archiveRes.status}`)
   172→      return { files: [], skillMd: '', metaJson: null, toolReferences: [] }
   173→    }
   174→
   175→    const archiveBuffer = Buffer.from(await archiveRes.arrayBuffer())
   176→
   177→    // Extract ZIP
   178→    const zip = new AdmZip(archiveBuffer)
   179→    const zipEntries = zip.getEntries()
   180→
   181→    const files: string[] = []
   182→    let skillMd = ''
   183→    let metaJson: Record<string, unknown> | null = null
   184→
   185→    const skillDir = join(process.cwd(), 'skills', slug)
   186→    mkdirSync(skillDir, { recursive: true })
   187→
   188→    for (const entry of zipEntries) {
   189→      if (entry.isDirectory) continue
   190→      const fileName = entry.entryName
   191→      files.push(fileName)
   192→
   193→      // Extract to filesystem
   194→      const filePath = join(skillDir, fileName)
   195→      const fileDir = join(filePath, '..')
   196→      mkdirSync(fileDir, { recursive: true })
   197→
   198→      // Detect binary vs text files
   199→      const isBinary = fileName.endsWith('.png') || fileName.endsWith('.jpg') || fileName.endsWith('.jpeg') ||
   200→        fileName.endsWith('.gif') || fileName.endsWith('.ico') || fileName.endsWith('.woff') ||
   201→        fileName.endsWith('.woff2') || fileName.endsWith('.ttf') || fileName.endsWith('.eot') ||
   202→        fileName.endsWith('.zip') || fileName.endsWith('.tar') || fileName.endsWith('.gz')
   203→
   204→      if (isBinary) {
   205→        writeFileSync(filePath, entry.getData())
   206→      } else {
   207→        writeFileSync(filePath, entry.getData().toString('utf-8'), 'utf-8')
   208→      }
   209→
   210→      // Parse special files
   211→      if (fileName === 'SKILL.md' || fileName.endsWith('/SKILL.md')) {
   212→        skillMd = entry.getData().toString('utf-8')
   213→      }
   214→      if (fileName === '_meta.json' || fileName.endsWith('/_meta.json')) {
   215→        try {
   216→          metaJson = JSON.parse(entry.getData().toString('utf-8'))
   217→        } catch {
   218→          // Invalid JSON, ignore
   219→        }
   220→      }
   221→    }
   222→
   223→    // Find tool references in SKILL.md
   224→    const toolReferences = extractToolReferences(skillMd)
   225→
   226→    console.log(`[Skills] Extracted archive for ${slug}: ${files.length} files, ${toolReferences.length} tool references`)
   227→    return { files, skillMd, metaJson, toolReferences }
   228→  } catch (err) {
   229→    console.error(`[Skills] Archive extraction failed for ${slug}:`, err)
   230→    return { files: [], skillMd: '', metaJson: null, toolReferences: [] }
   231→  }
   232→}

   235→// Helper: Recursively delete a directory

   237→function removeDirectoryRecursive(dirPath: string) {
   238→  if (!existsSync(dirPath)) return
   239→  try {
   240→    rmSync(dirPath, { recursive: true, force: true })
   241→    console.log(`[Skills] Removed directory: ${dirPath}`)
   242→  } catch (err) {
   243→    console.warn(`[Skills] Failed to remove directory ${dirPath}:`, err instanceof Error ? err.message : String(err))
   244→  }

   259→          content: skill.content,
   260→          source: skill.source,
   261→          enabled: skill.enabled,
   262→          version: skill.version,
   263→        },
   264→      })
   265→
   266→      // Sync bundled skill to filesystem for z.ai platform layer injection
   267→      try {
   268→        const skillsDir = join(process.cwd(), 'skills', skill.slug)
   269→        mkdirSync(skillsDir, { recursive: true })

   273→        console.warn(`[Skills] Failed to write bundled SKILL.md to filesystem:`, fsErr instanceof Error ? fsErr.message : String(fsErr))
   274→      }
   275→    }
   276→  }

   282→export async function GET(request: NextRequest) {
   283→  const { searchParams } = new URL(request.url)
   284→  const action = searchParams.get('action') || 'installed'
   285→  const query = searchParams.get('q') || ''
   286→  const category = searchParams.get('category') || 'All'
   287→
   288→  // Ensure bundled skills exist (only for installed action)
   289→  if (action === 'installed') {
   290→    await ensureBundledSkills()
   291→  }
   292→
   293→  if (action === 'installed') {
   294→    // Get installed skills from SQLite
   295→    const skills = await db.agentSkill.findMany({
   296→      where: { agentId: 'default' },
   297→      orderBy: { installedAt: 'desc' },
   298→    })
   299→    return NextResponse.json({ skills, total: skills.length })
   300→  }
   301→
   302→  if (action === 'search') {
   303→    // Get installed slugs for marking
   304→    const installed = await db.agentSkill.findMany({
   305→      where: { agentId: 'default' },
   306→      select: { slug: true },
   307→    })
   308→    const installedSlugs = new Set(installed.map(s => s.slug))
   309→
   310→    // Try real ClawHub.ai API first
   311→    try {
   312→      const searchQuery = query || '*'
   313→      const limit = searchParams.get('limit') || '20'
   314→      const clawhubUrl = `${CLAWHUB_API}/search?q=${encodeURIComponent(searchQuery)}&limit=${limit}`
   315→      const res = await fetch(clawhubUrl, {
   316→        signal: AbortSignal.timeout(10000),
   317→        headers: { 'Accept': 'application/json' },
   318→      })
   319→      if (res.ok) {
   320→        const data = await res.json()
   321→        if (data.results && Array.isArray(data.results)) {
   322→          const skills = data.results.map((r: {
   323→            slug: string; displayName: string; summary: string;
   324→            ownerHandle: string; owner?: { handle: string; displayName: string; image?: string };
   325→            score: number; updatedAt: number;
   326→            tags?: Record<string, string>; stats?: { downloads: number; installsAllTime: number; stars: number };
   327→            latestVersion?: { version: string };
   328→          }) => ({
   329→            slug: r.slug,
   330→            name: r.displayName,
   331→            version: r.latestVersion?.version || r.tags?.latest || '0.0.0',
   332→            description: r.summary || '',
   333→            author: r.owner?.displayName || r.ownerHandle || '',
   334→            authorHandle: r.ownerHandle,
   335→            authorImage: r.owner?.image || '',
   336→            downloads: r.stats?.downloads || 0,
   337→            installs: r.stats?.installsAllTime || 0,
   338→            stars: r.stats?.stars || 0,
   339→            score: r.score || 0,
   340→            category: 'All',
   341→            installed: installedSlugs.has(r.slug),
   342→            source: 'clawhub',
   343→            url: `https://clawhub.ai/${r.ownerHandle}/${r.slug}`,
   344→          }))
   345→          return NextResponse.json({ skills, total: skills.length, source: 'clawhub-api' })
   346→        }
   347→      }
   348→    } catch (error) {
   349→      console.error('ClawHub API failed:', error)
   350→      // Fall through to gateway fallback
   351→    }
   352→
   353→    // Try OpenClaw gateway as fallback
   354→    try {
   355→      const res = await fetch(`${GATEWAY_URL}/v1/skills/search?q=${encodeURIComponent(query)}&category=${category}`, {
   356→        signal: AbortSignal.timeout(5000),
   357→      })
   358→      if (res.ok) {
   359→        const data = await res.json()
   360→        if (data.skills && data.skills.length > 0) {
   361→          return NextResponse.json(data)
   362→        }
   363→      }
   364→    } catch {
   365→      // Gateway unavailable
   366→    }
   367→
   368→    // No results from any source
   369→    return NextResponse.json({ skills: [], total: 0, source: 'none', message: 'Không thể kết nối ClawHub.ai. Vui lòng thử lại sau.' })
   370→  }
   371→
   372→  if (action === 'detail') {
   373→    // Get single skill detail (from local DB)
   374→    const slug = searchParams.get('slug')
   375→    if (!slug) {
   376→      return NextResponse.json({ error: 'Missing slug parameter' }, { status: 400 })
   377→    }
   378→    const skill = await db.agentSkill.findUnique({
   379→      where: { agentId_slug: { agentId: 'default', slug } },
   380→    })
   381→    if (!skill) {
   382→      return NextResponse.json({ error: 'Skill not found' }, { status: 404 })
   383→    }
   384→    return NextResponse.json({ skill })
   385→  }
   386→
   387→  if (action === 'content') {
   388→    // Fetch skill content (SKILL.md) from ClawHub.ai for a search result
   389→    const slug = searchParams.get('slug')
   390→    if (!slug) {
   391→      return NextResponse.json({ error: 'Missing slug parameter' }, { status: 400 })
   392→    }
   393→
   394→    // First check if it's already installed locally
   395→    const local = await db.agentSkill.findUnique({
   396→      where: { agentId_slug: { agentId: 'default', slug } },
   397→    })
   398→    if (local) {
   399→      return NextResponse.json({
   400→        name: local.name,
   401→        slug: local.slug,
   402→        content: local.content,
   403→        version: local.version,
   404→        source: local.source,
   405→        installed: true,
   406→        description: local.description,
   407→        authorHandle: local.authorHandle,
   408→        fileCount: local.fileCount,
   409→        toolCount: local.toolCount,
   410→        tools: local.tools ? JSON.parse(local.tools) : [],
   411→      })
   412→    }
   413→
   414→    // P1.2 FIX: Use correct ClawHub API endpoints (no ownerHandle in path)
   415→    // Try fetching SKILL.md content directly
   416→    let skillContent = ''
   417→    let skillName = slug
   418→    let skillDescription = ''
   419→    let skillVersion = '0.0.0'
   420→    let skillAuthorHandle = ''
   421→    let skillAuthor = ''
   422→
   423→    // Step 1: Fetch SKILL.md content from packages endpoint
   424→    try {
   425→      const fileRes = await fetch(`${CLAWHUB_API}/packages/${slug}/file?path=SKILL.md`, {
   426→        signal: AbortSignal.timeout(8000),
   427→      })
   428→      if (fileRes.ok) {
   429→        skillContent = await fileRes.text()
   430→      }
   431→    } catch (error) {
   432→      console.error('ClawHub SKILL.md fetch failed:', error)
   433→    }
   434→
   435→    // Step 2: Fetch skill detail/metadata from skills endpoint
   436→    try {
   437→      const detailRes = await fetch(`${CLAWHUB_API}/skills/${slug}`, {
   438→        signal: AbortSignal.timeout(8000),
   439→      })
   440→      if (detailRes.ok) {
   441→        const detailData = await detailRes.json()
   442→        const skill = detailData.skill || detailData
   443→        skillName = skill.displayName || skill.name || slug
   444→        skillDescription = skill.summary || skill.description || ''
   445→        skillVersion = detailData.latestVersion?.version || skill.tags?.latest || '0.0.0'
   446→        skillAuthor = detailData.owner?.displayName || ''
   447→        skillAuthorHandle = detailData.owner?.handle || ''
   448→      }
   449→    } catch {
   450→      // Detail fetch failed, use defaults
   451→    }
   452→
   453→    // If we got SKILL.md content, return it
   454→    if (skillContent) {
   455→      // Also parse tool references from the content
   456→      const toolRefs = extractToolReferences(skillContent)
   457→      return NextResponse.json({
   458→        name: skillName,
   459→        slug,
   460→        content: skillContent,
   461→        version: skillVersion,
   462→        source: 'clawhub',
   463→        installed: false,
   464→        author: skillAuthor,
   465→        authorHandle: skillAuthorHandle,
   466→        description: skillDescription,
   467→        toolCount: toolRefs.length,
   468→        tools: toolRefs,
   469→      })
   470→    }
   471→
   472→    // Fallback: try searching for the skill to get basic info
   473→    try {
   474→      const searchRes = await fetch(`${CLAWHUB_API}/search?q=${encodeURIComponent(slug)}&limit=5`, {
   475→        signal: AbortSignal.timeout(8000),
   476→      })
   477→      if (searchRes.ok) {
   478→        const searchData = await searchRes.json()
   479→        const match = searchData.results?.find((r: { slug: string }) => r.slug === slug || r.slug === slug.split('/').pop())
   480→        if (match) {
   481→          return NextResponse.json({
   482→            name: match.displayName || match.slug,
   483→            slug,
   484→            content: `# ${match.displayName || match.slug}\n\n${match.summary || ''}\n\n## Chi tiết\nNội dung đầy đủ cần tải về từ ClawHub.ai.\n\nVisit https://clawhub.ai/${match.ownerHandle}/${slug} for more details.`,
   485→            version: match.latestVersion?.version || match.tags?.latest || '0.0.0',
   486→            source: 'clawhub',
   487→            installed: false,
   488→            author: match.owner?.displayName || match.ownerHandle || '',
   489→            authorHandle: match.ownerHandle,
   490→            description: match.summary || '',
   491→          })
   492→        }
   493→      }
   494→    } catch {
   495→      // Search fallback failed
   496→    }
   497→
   498→    return NextResponse.json({ error: 'Không thể tải nội dung skill từ ClawHub', name: slug, slug, content: '', source: 'none', installed: false }, { status: 404 })
   499→  }
   500→
   501→  if (action === 'categories') {
   502→    const categories = ['All', 'Browser', 'File System', 'Developer', 'Data', 'Communication', 'Productivity', 'Media', 'Smart Home']
   503→    return NextResponse.json({ categories })
   504→  }
   505→
   506→  return NextResponse.json({ error: 'Invalid action' }, { status: 400 })
   507→}

   519→      let skillDescription = ''

   522→      let skillAuthorHandle = body.ownerHandle || ''
   523→      let skillContent = ''
   524→      let archiveFiles: string[] = []
   525→      let archiveToolRefs: string[] = []
   526→      let archiveFileCount = 0
   527→
   528→      // Step 1: Fetch skill metadata from ClawHub (corrected endpoint)
   529→      try {
   530→        const detailRes = await fetch(`${CLAWHUB_API}/skills/${slug}`, {
   531→          signal: AbortSignal.timeout(8000),
   532→        })

   534→          const detailData = await detailRes.json()
   535→          const skill = detailData.skill || detailData
   536→          skillName = skill.displayName || skill.name || slug
   537→          skillDescription = skill.summary || skill.description || ''
   538→          skillVersion = detailData.latestVersion?.version || skill.tags?.latest || '0.0.0'
   539→          skillAuthor = detailData.owner?.displayName || ''
   540→          skillAuthorHandle = detailData.owner?.handle || skillAuthorHandle
   541→        }
   542→      } catch {
   543→        // API unavailable, use slug as name
   544→      }
   545→
   546→      // Step 2: Try to download and extract full archive
   547→      const archiveResult = await downloadAndExtractArchive(slug)
   548→      if (archiveResult.files.length > 0) {
   549→        archiveFiles = archiveResult.files
   550→        archiveFileCount = archiveResult.files.length
   551→        archiveToolRefs = archiveResult.toolReferences
   552→        if (archiveResult.skillMd) {
   553→          skillContent = archiveResult.skillMd
   554→        }
   555→      }
   556→
   557→      // Step 3: If no SKILL.md from archive, fetch it directly
   558→      if (!skillContent) {
   559→        try {
   560→          const fileRes = await fetch(`${CLAWHUB_API}/packages/${slug}/file?path=SKILL.md`, {
   561→            signal: AbortSignal.timeout(8000),
   562→          })
   563→          if (fileRes.ok) {
   564→            skillContent = await fileRes.text()
   565→          }
   566→        } catch {
   567→          // Content fetch failed
   568→        }
   569→      }
   570→
   571→      if (!skillContent) {
   572→        skillContent = `# ${skillName}\n\n${skillDescription}\n\n## Usage\nThis skill was installed from ClawHub.\n\nVisit https://clawhub.ai for more details.`
   573→      }
   574→
   575→      // Step 4: Also write SKILL.md to filesystem if not already done by archive extraction
   576→      if (archiveFileCount === 0) {
   577→        try {
   578→          const skillsDir = join(process.cwd(), 'skills', slug)
   579→          mkdirSync(skillsDir, { recursive: true })
   580→          writeFileSync(join(skillsDir, 'SKILL.md'), skillContent, 'utf-8')
   581→        } catch {}
   582→      }
   583→
   584→      // Step 5: If we still don't have metadata from detail endpoint, try search
   585→      if (!skillDescription) {
   586→        try {
   587→          const searchRes = await fetch(`${CLAWHUB_API}/search?q=${encodeURIComponent(slug)}&limit=5`, {
   588→            signal: AbortSignal.timeout(8000),
   589→          })
   590→          if (searchRes.ok) {
   591→            const searchData = await searchRes.json()
   592→            const match = searchData.results?.find((r: { slug: string }) =>
   593→              r.slug === slug || r.slug === slug.split('/').pop()
   594→            )
   595→            if (match) {
   596→              if (!skillDescription) skillDescription = match.summary || ''
   597→              if (skillName === slug) skillName = match.displayName || match.slug
   598→              if (skillVersion === '0.0.0') skillVersion = match.latestVersion?.version || match.tags?.latest || '0.0.0'
   599→              if (!skillAuthor) skillAuthor = match.owner?.displayName || match.ownerHandle || ''
   600→              if (!skillAuthorHandle) skillAuthorHandle = match.ownerHandle
   601→            }
   602→          }
   603→        } catch {
   604→          // Search fallback failed
   605→        }
   606→      }
   607→
   608→      // Parse tool references from SKILL.md if not already found via archive
   609→      if (archiveToolRefs.length === 0 && skillContent) {
   610→        archiveToolRefs = extractToolReferences(skillContent)
   611→      }
   612→
   613→      // Step 6: Upsert into DB with full metadata
   614→      const existing = await db.agentSkill.findUnique({
   615→        where: { agentId_slug: { agentId: 'default', slug } },
   616→      })
   617→
   618→      const skill = await db.agentSkill.upsert({
   619→        where: { agentId_slug: { agentId: 'default', slug } },
   620→        create: {
   621→          agentId: 'default',
   622→          slug,
   623→          name: skillName,
   624→          content: skillContent,
   625→          source: 'clawhub',
   626→          enabled: true,
   627→          version: skillVersion,
   628→          description: skillDescription,
   629→          authorHandle: skillAuthorHandle,
   630→          fileCount: archiveFileCount,
   631→          files: archiveFiles.length > 0 ? JSON.stringify(archiveFiles) : null,
   632→          toolCount: archiveToolRefs.length,
   633→          tools: archiveToolRefs.length > 0 ? JSON.stringify(archiveToolRefs) : null,
   634→          hasPlugin: archiveFiles.some(f => f.endsWith('.plugin.js') || f.endsWith('.plugin.ts') || f.endsWith('.plugin.json')),
   635→        },
   636→        update: {
   637→          name: skillName,
   638→          version: skillVersion,
   639→          source: 'clawhub',
   640→          content: skillContent,
   641→          description: skillDescription,
   642→          authorHandle: skillAuthorHandle,
   643→          fileCount: archiveFileCount,
   644→          files: archiveFiles.length > 0 ? JSON.stringify(archiveFiles) : null,
   645→          toolCount: archiveToolRefs.length,
   646→          tools: archiveToolRefs.length > 0 ? JSON.stringify(archiveToolRefs) : null,
   647→          hasPlugin: archiveFiles.some(f => f.endsWith('.plugin.js') || f.endsWith('.plugin.ts') || f.endsWith('.plugin.json')),
   648→        },
   649→      })
   650→
   651→      const message = existing
   652→        ? `Skill "${skillName}" đã cập nhật lên v${skillVersion} (${archiveFileCount} files, ${archiveToolRefs.length} tools)`
   653→        : `Skill "${skillName}" v${skillVersion} đã cài đặt thành công (${archiveFileCount} files, ${archiveToolRefs.length} tools)`
   654→
   655→      return NextResponse.json({
   656→        success: true,
   657→        skill,
   658→        message,
   659→        archiveFileCount,
   660→        toolReferences: archiveToolRefs,
   661→      })
   662→    }
   663→
   664→    if (action === 'uninstall' && slug) {
   665→      const skill = await db.agentSkill.findUnique({
   666→        where: { agentId_slug: { agentId: 'default', slug } },
   667→      })
   668→      if (!skill) {
   669→        return NextResponse.json({ error: 'Skill not found' }, { status: 404 })
   670→      }
   671→      if (skill.source === 'bundled') {
   672→        return NextResponse.json({ error: 'Không thể gỡ skill mặc định' }, { status: 400 })
   673→      }
   674→      // Allow uninstall for clawhub, custom, and template sources
   675→      await db.agentSkill.delete({
   676→        where: { agentId_slug: { agentId: 'default', slug } },
   677→      })
   678→
   679→      // P1.2: Clean up ALL extracted archive files from skills/{slug}/ directory
   680→      try {
   681→        const skillDir = join(process.cwd(), 'skills', slug)
   682→        if (existsSync(skillDir)) {
   683→          const isDir = statSync(skillDir).isDirectory()
   684→          if (isDir) {
   685→            removeDirectoryRecursive(skillDir)
   686→          } else {
   687→            // Shouldn't happen, but handle gracefully
   688→            unlinkSync(skillDir)
   689→          }
   690→        }
   691→      } catch (fsErr) {
   692→        console.warn(`[Skills] Failed to remove skill directory:`, fsErr instanceof Error ? fsErr.message : String(fsErr))
   693→      }
   694→
   695→      return NextResponse.json({ success: true, message: `Skill "${skill.name}" đã gỡ cài đặt` })
   696→    }
   697→
   698→    if (action === 'toggle' && slug) {
   699→      const skill = await db.agentSkill.findUnique({
   700→        where: { agentId_slug: { agentId: 'default', slug } },
   701→      })
   702→      if (!skill) {
   703→        return NextResponse.json({ error: 'Skill not found' }, { status: 404 })
   704→      }
   705→      const newEnabled = enabled !== undefined ? enabled : !skill.enabled
   706→      await db.agentSkill.update({
   707→        where: { agentId_slug: { agentId: 'default', slug } },
   708→        data: { enabled: newEnabled },
   709→      })
   710→
   711→      // Sync filesystem with enabled state
   712→      try {
   713→        const skillFilePath = join(process.cwd(), 'skills', slug, 'SKILL.md')
   714→        if (newEnabled) {
   715→          const skillsDir = join(process.cwd(), 'skills', slug)
   716→          mkdirSync(skillsDir, { recursive: true })
   717→          writeFileSync(skillFilePath, skill.content, 'utf-8')
   718→          console.log(`[Skills] Synced SKILL.md to filesystem (enabled): skills/${slug}/SKILL.md`)
   719→        } else {
   720→          if (existsSync(skillFilePath)) {
   721→            unlinkSync(skillFilePath)
   722→            console.log(`[Skills] Removed SKILL.md from filesystem (disabled): skills/${slug}/SKILL.md`)
   723→          }
   724→        }
   725→      } catch (fsErr) {
   726→        console.warn(`[Skills] Failed to sync filesystem for toggle:`, fsErr instanceof Error ? fsErr.message : String(fsErr))
   727→      }
   728→
   729→      return NextResponse.json({ success: true, enabled: newEnabled, message: `Skill "${skill.name}" ${newEnabled ? 'đã bật' : 'đã tắt'}` })
   730→    }
   731→
   732→    return NextResponse.json({ error: 'Invalid action' }, { status: 400 })
   733→  } catch {
   734→    return NextResponse.json({ error: 'Invalid request body' }, { status: 400 })
   735→  }
   736→}
