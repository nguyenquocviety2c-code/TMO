   227→// ==================== UPLOAD SECTION ====================
   228→
   229→function UploadSection({ onUploadComplete, existingDocNames }: { onUploadComplete: (uploadedDocIds: string[]) => void; existingDocNames: Set<string> }) {
   230→  const [uploading, setUploading] = useState(false)
   231→  const [selectedDomain, setSelectedDomain] = useState<string>('auto')
   232→  const [uploadResult, setUploadResult] = useState<{
   233→    documents: Array<{ id: string; title: string; status: string; domain: string }>
   234→    totalUploaded: number; totalErrors: number; errors?: Array<{ file: string; error: string }>
   235→  } | null>(null)
   236→  /** Tracks per-file upload progress: "Uploading 3/7..." */
   237→  const [uploadProgress, setUploadProgress] = useState<string | null>(null)
   238→  /** Duplicate file names detected before upload — shown as a blocking warning */
   239→  const [duplicateWarning, setDuplicateWarning] = useState<string[] | null>(null)
   240→  const fileInputRef = useRef<HTMLInputElement>(null)
   241→
   242→  /** Helper: read the actual error message from a non-OK response */
   243→  const readServerError = async (res: Response): Promise<string> => {
   244→    let serverError = `Lỗi server (${res.status})`
   245→    try {
   246→      const text = await res.text()
   247→      try {
   248→        const errData = JSON.parse(text)
   249→        if (errData.error) serverError = errData.error
   250→        else if (errData.message) serverError = errData.message
   251→      } catch {
   252→        // Response is not JSON — might be HTML from Next.js framework error
   253→        if (text) serverError = text.slice(0, 200)
   254→      }
   255→    } catch {}
   256→    if (res.status === 413) serverError = 'File quá lớn — vượt giới hạn tải lên'
   257→    return serverError
   258→  }
   259→
   260→  const handleUpload = async () => {
   261→    const files = fileInputRef.current?.files
   262→    if (!files || files.length === 0) return
   263→
   264→    // Clear previous duplicate warning
   265→    setDuplicateWarning(null)
   266→
   267→    // Client-side file validation
   268→    const MAX_CLIENT_SIZE = 100 * 1024 * 1024 // 100MB per file
   269→    const oversizedFiles: string[] = []
   270→    const nonPdfFiles: string[] = []
   271→    const duplicates: string[] = []
   272→    for (let i = 0; i < files.length; i++) {
   273→      if (!files[i].name.toLowerCase().endsWith('.pdf') && files[i].type !== 'application/pdf') {
   274→        nonPdfFiles.push(files[i].name)
   275→      }
   276→      if (files[i].size > MAX_CLIENT_SIZE) {
   277→        oversizedFiles.push(`${files[i].name} (${(files[i].size / 1024 / 1024).toFixed(1)}MB)`)
   278→      }
   279→      // Check for duplicate document names
   280→      if (existingDocNames.has(files[i].name)) {
   281→        duplicates.push(files[i].name)
   282→      }
   283→    }
   284→
   285→    // Show validation errors (non-PDF, oversized)
   286→    if (nonPdfFiles.length > 0 || oversizedFiles.length > 0) {
   287→      const errors: Array<{ file: string; error: string }> = []
   288→      nonPdfFiles.forEach(f => errors.push({ file: f, error: 'Chỉ hỗ trợ file PDF' }))
   289→      oversizedFiles.forEach(f => errors.push({ file: f, error: 'File quá lớn (tối đa 100MB/file)' }))
   290→      setUploadResult({ documents: [], totalUploaded: 0, totalErrors: errors.length, errors })
   291→    }
   292→
   293→    // Show duplicate warning and block those files from uploading
   294→    if (duplicates.length > 0) {
   295→      setDuplicateWarning(duplicates)
   296→      // If ALL files are duplicates, stop here
   297→      if (duplicates.length === files.length) return
   298→    }
   299→
   300→    setUploading(true)
   301→    setUploadResult(null)
   302→
   303→    // Upload files ONE AT A TIME to avoid body size limit issues.
   304→    // Sending all files in a single request creates a large multipart body
   305→    // that can exceed Next.js's internal limits and cause 400 errors.
   306→    // SKIP files that are duplicates — they are already shown in the warning.
   307→    const allUploaded: Array<{ id: string; title: string; status: string; domain: string }> = []
   308→    const allErrors: Array<{ file: string; error: string }> = []
   309→    const duplicateSet = new Set(duplicates)
   310→    const filesToUpload = Array.from(files).filter(f => !duplicateSet.has(f.name))
   311→
   312→    for (let i = 0; i < filesToUpload.length; i++) {
   313→      const file = filesToUpload[i]
   314→      setUploadProgress(`Đang tải ${i + 1}/${filesToUpload.length}: ${file.name}`)
   315→
   316→      try {
   317→        const formData = new FormData()
   318→        formData.append('files', file)
   319→        if (selectedDomain !== 'auto') formData.append('domain', selectedDomain)
   320→
   321→        const res = await fetch('/api/ingestion/upload', { method: 'POST', body: formData })
   322→        if (!res.ok) {
   323→          const serverError = await readServerError(res)
   324→          allErrors.push({ file: file.name, error: serverError })
   325→          continue
   326→        }
   327→        const data = await res.json()
   328→        if (data.documents && data.documents.length > 0) {
   329→          allUploaded.push(...data.documents)
   330→        }
   331→        if (data.errors && data.errors.length > 0) {
   332→          allErrors.push(...data.errors)
   333→        }
   334→      } catch (err) {
   335→        const message = err instanceof Error ? err.message : String(err)
   336→        allErrors.push({ file: file.name, error: message })
   337→      }
   338→    }
   339→
   340→    setUploadProgress(null)
   341→    setUploadResult({
   342→      documents: allUploaded,
   343→      totalUploaded: allUploaded.length,
   344→      totalErrors: allErrors.length,
   345→      errors: allErrors.length > 0 ? allErrors : undefined,
   346→    })
   347→    // Sonner toast notifications for upload results
   348→    if (allUploaded.length > 0) {
   349→      sonnerToast.success('Tải lên thành công', {
   350→        description: `Đã tải ${allUploaded.length} tài liệu lên hệ thống`,
   351→        duration: 4000,
   352→      })
   353→      onUploadComplete(allUploaded.map(d => d.id))
   354→    }
   355→    if (allErrors.length > 0) {
   356→      sonnerToast.error('Tải lên thất bại', {
   357→        description: `${allErrors.length} file không thể tải lên`,
   358→        duration: 6000,
   359→      })
   360→    }
   361→    setUploading(false)
   362→  }
   363→
   364→  return (
   365→    <div className="nc-wrap nc-cyan nc-corner-glow">
   366→    <Card className="nc-panel nc-md nc-border-cyan">
   367→      <CardHeader className="pb-4">
   368→        <div className="flex items-center gap-2.5">
   369→          <div className="p-1.5 rounded-lg bg-gradient-to-br from-emerald-500 to-teal-600">
   370→            <Upload className="h-4 w-4 text-white" />
   371→          </div>
   372→          <div>
   373→            <CardTitle className="text-sm font-semibold tracking-tight text-stone-100">Tải lên PDF</CardTitle>
   374→            <CardDescription className="text-xs mt-0.5 text-stone-400">Tải tài liệu PDF — tối đa 100MB/file, mỗi file tải riêng biệt</CardDescription>
   375→          </div>
   376→        </div>
   377→      </CardHeader>
   378→      <CardContent>
   379→        <div className="space-y-4">
   380→          <div className="flex gap-3">
   381→            <div className="flex-1">
   382→              <Input ref={fileInputRef} type="file" accept=".pdf" multiple disabled={uploading}
   383→                className="text-xs h-9 text-stone-200 file:mr-2 file:py-1 file:px-3 file:rounded-lg file:border-0 file:text-xs file:font-medium file:bg-gradient-to-r file:from-emerald-900/40 file:to-teal-900/40 file:text-teal-400 hover:file:from-emerald-800/40 hover:file:to-teal-800/40 rounded-lg border-cyan-400/35 focus:border-emerald-600 focus:ring-emerald-600/20" />
   384→            </div>
   385→            <Select value={selectedDomain} onValueChange={setSelectedDomain}>
   386→              <SelectTrigger className="w-[160px] h-9 text-xs rounded-lg border-cyan-400/35 text-stone-200 focus:border-emerald-600"><SelectValue placeholder="Phân loại" /></SelectTrigger>
   387→              <SelectContent className="bg-slate-950/80 border-cyan-400/35">
   388→                <SelectItem value="auto" className="text-stone-200 focus:text-white focus:bg-stone-800/80">Tự động phân loại</SelectItem>
   389→                <SelectItem value="programming" className="text-stone-200 focus:text-white focus:bg-stone-800/80">Lập trình</SelectItem>
   390→                <SelectItem value="algorithm" className="text-stone-200 focus:text-white focus:bg-stone-800/80">Thuật toán</SelectItem>
   391→                <SelectItem value="ml" className="text-stone-200 focus:text-white focus:bg-stone-800/80">Machine Learning</SelectItem>
   392→                <SelectItem value="meta_cognitive" className="text-stone-200 focus:text-white focus:bg-stone-800/80">Siêu nhận thức</SelectItem>
   393→                <SelectItem value="linux" className="text-stone-200 focus:text-white focus:bg-stone-800/80">Linux</SelectItem>
   394→                <SelectItem value="security" className="text-stone-200 focus:text-white focus:bg-stone-800/80">Bảo mật</SelectItem>
   395→                <SelectItem value="ux_ui" className="text-stone-200 focus:text-white focus:bg-stone-800/80">Thiết kế UX/UI</SelectItem>
   396→              </SelectContent>
   397→            </Select>
   398→            <Button onClick={handleUpload} disabled={uploading} size="sm" className="lens-flare chamfer-sm h-9 px-4 bg-gradient-to-r from-emerald-500 to-teal-600 hover:from-emerald-600 hover:to-teal-700 text-white shadow-sm border-0">
   399→              {uploading ? <><Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> {uploadProgress || 'Đang tải...'}</> : <><FolderUp className="mr-1.5 h-3.5 w-3.5" /> Tải lên</>}
   400→            </Button>
   401→          </div>
   402→          {/* Per-file upload progress indicator */}
   403→          {uploadProgress && (
   404→            <div className="flex items-center gap-2 p-2.5 rounded-xl bg-teal-950/40 border border-teal-500/55 text-xs text-teal-400">
   405→              <Loader2 className="h-3.5 w-3.5 animate-spin flex-shrink-0" />
   406→              <span className="truncate">{uploadProgress}</span>
   407→            </div>
   408→          )}
   409→          {/* Duplicate name warning — shown when selected files match existing documents */}
   410→          {duplicateWarning && duplicateWarning.length > 0 && (
   411→            <div className="p-3 rounded-xl bg-amber-950/40 border border-amber-500/55 text-xs text-amber-300">
   412→              <div className="flex items-start gap-2">

  1009→const MAX_CHAT_HISTORY = 10
  1010→const CHAT_HISTORY_CONTEXT_SIZE = 5
  1011→
  1012→function ChatSection() {
  1013→  const [messages, setMessages] = useState<ChatMessage[]>([])
  1014→  const [input, setInput] = useState('')
  1015→  const [loading, setLoading] = useState(false)
  1016→  const scrollRef = useRef<HTMLDivElement>(null)
  1017→  const bottomRef = useRef<HTMLDivElement>(null)
  1018→
  1019→  const buildChatHistory = useCallback((msgs: ChatMessage[]): Array<{ role: 'user' | 'assistant'; content: string }> => {
  1020→    // Include last N messages as context for the query service
  1021→    const recent = msgs.slice(-CHAT_HISTORY_CONTEXT_SIZE * 2) // *2 because user+assistant pairs
  1022→    return recent.map(m => ({ role: m.role, content: m.content }))
  1023→  }, [])
  1024→
  1025→  const sendMessage = useCallback(async (queryText?: string) => {
  1026→    const text = queryText || input.trim()
  1027→    if (!text || loading) return
  1028→    const userMsg: ChatMessage = { id: crypto.randomUUID(), role: 'user', content: text, timestamp: new Date() }
  1029→    setMessages(prev => {
  1030→      const updated = [...prev, userMsg]
  1031→      // Keep only last MAX_CHAT_HISTORY messages
  1032→      if (updated.length > MAX_CHAT_HISTORY) return updated.slice(-MAX_CHAT_HISTORY)
  1033→      return updated
  1034→    })
  1035→    if (!queryText) setInput('')
  1036→    setLoading(true)
  1037→
  1038→    try {
  1039→      // Build chat history from messages before this one (last 5 messages as context)
  1040→      const chatHistory = buildChatHistory([...messages, userMsg].slice(0, -1))
  1041→
  1042→      const res = await fetch('/api/query', {
  1043→        method: 'POST',
  1044→        headers: { 'Content-Type': 'application/json' },
  1045→        body: JSON.stringify({
  1046→          query: text,
  1047→          options: { chatHistory },
  1048→        }),
  1049→      })
  1050→      if (!res.ok) {
  1051→        let errorMsg = 'Query API error'
  1052→        try { const errData = await res.json(); errorMsg = errData.error || errData.message || errorMsg } catch {}
  1053→        throw new Error(errorMsg)
  1054→      }
  1055→      const data = await res.json() as QueryResult
  1056→
  1057→      const assistantMsg: ChatMessage = {
  1058→        id: crypto.randomUUID(),
  1059→        role: 'assistant',
  1060→        content: data.answer || 'Không thể tạo câu trả lời.',
  1061→        timestamp: new Date(),
  1062→        queryResult: data,
  1063→      }
  1064→      setMessages(prev => {
  1065→        const updated = [...prev, assistantMsg]
  1066→        if (updated.length > MAX_CHAT_HISTORY) return updated.slice(-MAX_CHAT_HISTORY)
  1067→        return updated
  1068→      })
  1069→    } catch (err) {
  1070→      const errorMsg: ChatMessage = {
  1071→        id: crypto.randomUUID(), role: 'assistant',
  1072→        content: `Lỗi khi truy vấn: ${err instanceof Error ? err.message : 'Unknown error'}`,
  1073→        timestamp: new Date(),
  1074→      }
  1075→      setMessages(prev => [...prev, errorMsg])
  1076→    }
  1077→    setLoading(false)
  1078→  }, [input, loading, messages, buildChatHistory])
  1079→
  1080→  const clearChat = useCallback(() => {
  1081→    setMessages([])
  1082→    setInput('')
  1083→  }, [])
  1084→
  1085→  // Smooth scroll to latest message
  1086→  useEffect(() => {
  1087→    bottomRef.current?.scrollIntoView({ behavior: 'smooth' })
