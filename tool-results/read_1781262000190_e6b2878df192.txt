   510→// EMBEDDING CACHE (Phase 6)
   511→// SQLite-backed L2 cache — survives server restarts
   512→// L1 = in-memory Map (1h TTL, fast), L2 = SQLite (7d TTL, persistent)

   515→/// Embedding cache — stores previously computed embeddings to avoid redundant API calls
   516→/// Survives server restarts (unlike in-memory cache)
   517→/// IMPORTANT: Same text gets DIFFERENT embeddings for "query" vs "passage" (NVIDIA input_type)
   518→/// So the unique key is (hash, inputType) — NOT just hash
   519→model EmbeddingCache {
   520→  id        String   @id @default(cuid())
   521→  hash     String   /// SHA-256 hash of the input text
   522→  inputType String  @default("passage") /// "query" | "passage" — different embeddings for same text
   523→  model    String   /// Model used to generate the embedding (e.g. "nvidia/llama-nemotron-embed-1b-v2")
   524→  vector   String   /// JSON array of the embedding vector (1536 floats)
   525→  hitCount Int      @default(1) /// Number of cache hits (for LRU eviction)
   526→  createdAt DateTime @default(now())
   527→  lastHitAt DateTime @default(now()) /// Last time this cache entry was used
   528→  expiresAt DateTime /// When this cache entry expires
   529→
   530→  @@unique([hash, inputType]) /// Composite key — same text has separate query & passage embeddings
   531→  @@index([expiresAt])
   532→  @@index([lastHitAt])
   533→}

   643→/// Agent episodic memories — extracted from conversations, stored as vectors in Qdrant
   644→/// Each memory is a discrete piece of knowledge the agent learned from interaction
   645→model AgentMemory {
   646→  id             String   @id @default(cuid())

   650→  category       String   /// insight | fact | preference | correction | procedure | user_info
   651→  content        String   /// The actual memory content (text)
   652→  context        String?  /// Context: what triggered this memory (query, situation, etc.)
   653→  importance     Float    @default(0.5) /// 0.0-1.0 — how important this memory is
   654→  accessCount    Int      @default(0) /// How many times this memory was recalled
   655→  lastAccessedAt DateTime? /// Last time this memory was used in a response
   656→  qdrantPointId  String?  /// Qdrant point ID for vector search
   657→  embeddingModel String?  /// Embedding model used (e.g. nvidia/llama-nemotron-embed-1b-v2)
   658→  source         String   @default("auto") /// auto | manual | feedback | correction
   659→  tags           String?  /// JSON array of tags for categorization
   660→  isActive       Boolean  @default(true) /// Soft delete — inactive memories are not recalled
   661→  expiresAt      DateTime? /// Optional expiration for time-sensitive memories
   662→  createdAt      DateTime @default(now())
   663→  updatedAt      DateTime @updatedAt
   664→
   665→  @@index([agentId])
   666→  @@index([category])
   667→  @@index([importance])
   668→  @@index([accessCount])
   669→  @@index([isActive])
   670→  @@index([createdAt])
   671→  @@index([lastAccessedAt])
