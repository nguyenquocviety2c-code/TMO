   184→/**
   185→ * Execute a tool via Gateway HTTP /tools/invoke endpoint.
   186→ * Only works for HTTP-allowed tools (web_search, web_fetch, etc.).
   187→ * File operations and exec require session context and are NOT available.
   188→ */
   189→async function executeGatewayTool(name: string, args: Record<string, unknown>): Promise<unknown> {
   190→  if (!GATEWAY_HTTP_ALLOWED_TOOLS.has(name)) {
   191→    throw new Error(`Tool "${name}" requires Gateway session context — not available via HTTP proxy. Use /v1/chat/completions instead.`)
   192→  }
   193→
   194→  const res = await fetch(`${GATEWAY_URL}/tools/invoke`, {
   195→    method: 'POST',
   196→    headers: { 'Content-Type': 'application/json' },
   197→    body: JSON.stringify({ name, args }),
   198→    signal: AbortSignal.timeout(30000),
   199→  })
   200→
   201→  if (!res.ok) {
   202→    const errorText = await res.text().catch(() => 'Unknown error')
   203→    throw new Error(`Gateway tool error ${res.status}: ${errorText.slice(0, 200)}`)
   204→  }
   205→
   206→  const data = await res.json()
   207→  if (!data.ok && data.error) {
   208→    throw new Error(`Tool error: ${data.error?.message || data.error || 'Unknown'}`)
   209→  }
   210→
   211→  return data.result ?? data
   212→}

   214→// ==================== Z-AI-SDK LAZY INITIALIZATION ====================

   218→// This SDK ONLY works inside the z.ai sandbox environment where backend
   219→// services (web_search, page_reader) are available. If you clone this

   221→// and NOT an error. Direct API keys (TAVILY_API_KEY, SERPER_API_KEY,

   227→let zaiInstance: any = null
   228→let zaiInitPromise: Promise<any> | null = null
   229→let zaiAvailable: boolean | null = null // null = not yet checked
   230→
   231→async function getZAI(): Promise<any> {
   232→  if (zaiInstance) return zaiInstance
   233→  if (zaiInitPromise) return zaiInitPromise
   234→
   235→  zaiInitPromise = (async () => {
   236→    try {
   237→      const ZAI = (await import('z-ai-web-dev-sdk')).default
   238→      const instance = await ZAI.create()
   239→      console.log('[ToolExecutor] ✅ z-ai-web-dev-sdk initialized — web_search + page_reader available as fallback')
   240→      zaiInstance = instance
   241→      zaiAvailable = true
   242→      return instance
   243→    } catch (err) {
   244→      zaiAvailable = false
   245→      console.warn(
   246→        '[ToolExecutor] z-ai-web-dev-sdk not available (this is normal for local deployment — direct API keys will be used).',
   247→        `Detail: ${err instanceof Error ? err.message : String(err)}`
   248→      )
   249→      return null
   250→    }
   251→  })()
   252→
   253→  return zaiInitPromise
   254→}

   453→// ==================== TOOL EXECUTION ====================

   455→export interface ToolExecutionResult {
   456→  success: boolean
   457→  result: unknown
   458→  duration: number
   459→}

   461→/**
   462→ * Execute a tool call from LLM.
   463→ * Called in the ReAct loop when LLM returns a tool_call.
   464→ *
   465→ * @param toolName - Name of the tool to execute
   466→ * @param args - Arguments from LLM's tool_call
   467→ * @returns Execution result with success status
   468→ */
   469→export async function executeTool(
   470→  toolName: string,
   471→  args: Record<string, unknown>
   472→): Promise<ToolExecutionResult> {
   473→  const startTime = Date.now()
   474→
   475→  try {
   476→    let result: unknown
   477→
   478→    switch (toolName) {
   479→      case 'opencode':
   480→        result = await executeOpenCode(args)
   481→        break
   482→      case 'knowledge_search':
   483→        result = await executeKnowledgeSearch(args)
   484→        break
   485→      case 'knowledge_graph':
   486→        result = await executeKnowledgeGraph(args)
   487→        break
   488→      case 'knowledge_write':
   489→        result = await executeKnowledgeWrite(args)
   490→        break
   491→      case 'tavily':
   492→        result = await executeTavily(args)
   493→        break
   494→      case 'serper':
   495→        result = await executeSerper(args)
   496→        break
   497→      case 'jina':
   498→        result = await executeJina(args)
   499→        break
   500→      default: {
   501→        // P0.5: Try Gateway proxy for tools not in local set
   502→        if (GATEWAY_HTTP_ALLOWED_TOOLS.has(toolName)) {
   503→          try {
   504→            const gwResult = await executeGatewayTool(toolName, args)
   505→            return { success: true, result: gwResult, duration: Date.now() - startTime }
   506→          } catch (gwErr) {
   507→            return {
   508→              success: false,
   509→              result: `Gateway tool "${toolName}" failed: ${gwErr instanceof Error ? gwErr.message : String(gwErr)}`,
   510→              duration: Date.now() - startTime,
   511→            }
   512→          }
   513→        }
   514→        return { success: false, result: `Unknown tool: ${toolName}. Available local tools: opencode, knowledge_search, knowledge_graph, knowledge_write, tavily, serper, jina. Gateway tools: ${Array.from(GATEWAY_HTTP_ALLOWED_TOOLS).join(', ')}`, duration: Date.now() - startTime }
   515→      }
   516→    }
   517→
   518→    return {
   519→      success: true,
   520→      result,
   521→      duration: Date.now() - startTime,
   522→    }
   523→  } catch (err) {
   524→    const errorMsg = err instanceof Error ? err.message : String(err)
   525→    console.error(`[ToolExecutor] Error executing ${toolName}:`, errorMsg)
   526→    return {
   527→      success: false,
   528→      result: `Tool error: ${errorMsg}`,
   529→      duration: Date.now() - startTime,
   530→    }
   531→  }
   532→}

   534→// ==================== INDIVIDUAL TOOL IMPLEMENTATIONS ====================
   535→
   536→/**
   537→ * OpenCode — Code operations (read, write, bash, mkdir)
   538→ *
   539→ * Uses the OpenCode API server (configured via OPENCODE_SERVER_URL env var) if available,
   540→ * otherwise falls back to direct file system operations via the project API.
   541→ */
   542→async function executeOpenCode(args: Record<string, unknown>): Promise<unknown> {
   543→  const action = args.action as string
   544→  const path = args.path as string | undefined
   545→  const content = args.content as string | undefined
   546→  const command = args.command as string | undefined
   547→
   548→  // Try OpenCode server first
   549→  try {
   550→    const opencodeUrl = process.env.OPENCODE_SERVER_URL || 'http://localhost:18790'
   551→    const res = await fetch(`${opencodeUrl}/api/execute`, {
   552→      method: 'POST',
   553→      headers: { 'Content-Type': 'application/json' },
   554→      body: JSON.stringify({ action, path, content, command }),
   555→      signal: AbortSignal.timeout(30000),
   556→    })
   557→
   558→    if (res.ok) {
   559→      const data = await res.json()
   560→      return data
   561→    }
   562→  } catch {
   563→    // OpenCode server not available, fall through to direct implementation
   564→  }
   565→
   566→  // Fallback: Use the project's own API for file operations
   567→  switch (action) {
   568→    case 'read':
   569→      return await executeFileRead(path || '')
   570→    case 'write':
   571→      return await executeFileWrite(path || '', content || '')
   572→    case 'bash':
   573→      return await executeBash(command || '')
   574→    case 'mkdir':
   575→      return { success: true, message: `mkdir not needed — write will create parent dirs` }
   576→    default:
   577→      return { success: false, error: `Unknown opencode action: ${action}` }
   578→  }
   579→}

   581→/**
   582→ * Read a file from the project directory.
   583→ * Security: Only allows reading files within the project root.
   584→ * Uses path.resolve + startsWith check to prevent path traversal.
   585→ */
   586→async function executeFileRead(filePath: string): Promise<{ success: boolean; content?: string; error?: string }> {
   587→  if (!filePath) return { success: false, error: 'Path is required' }
   588→
   589→  try {
   590→    const pathModule = await import('path')
   591→    const projectRoot = process.cwd()
   592→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   593→
   594→    // Security: Prevent path traversal — resolved path must be within project root
   595→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   596→      return { success: false, error: `Access denied: path escapes project root` }
   597→    }
   598→
   599→    const fs = await import('fs/promises')
   600→    const content = await fs.readFile(resolvedPath, 'utf-8')
   601→
   602→    // Limit output to prevent overwhelming LLM context
   603→    const maxChars = 50000
   604→    const truncated = content.length > maxChars
   605→    const output = truncated ? content.slice(0, maxChars) + '\n\n... [TRUNCATED — file too large]' : content
   606→
   607→    return { success: true, content: output }
   608→  } catch (err) {
   609→    return {
   610→      success: false,
   611→      error: `Failed to read ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   612→    }
   613→  }
   614→}

   621→async function executeFileWrite(filePath: string, content: string): Promise<{ success: boolean; message: string; error?: string }> {

   654→/**
   655→ * Execute a bash command safely.
   656→ * Security: Only allows pre-approved commands.
   657→ */
   658→async function executeBash(command: string): Promise<{ success: boolean; output?: string; error?: string }> {
   659→  if (!command) return { success: false, error: 'Command is required' }
   660→
   661→  // Security: Whitelist of allowed commands
   662→  // NOTE: 'node' is excluded because `node -e "..."` can execute arbitrary code.
   663→  //       'cat'/'head'/'tail'/'grep' are excluded — use opencode read instead.
   664→  //       'npx'/'bunx' are excluded — can download+execute arbitrary npm packages.
   665→  const allowedPrefixes = [
   666→    'npm run',
   667→    'npm test',
   668→    'npm lint',
   669→    'bun run',
   670→    'bun test',
   671→    'bun lint',
   672→    'tsc',
   673→    'eslint',
   674→    'prettier',
   675→    'echo',
   676→    'ls',
   677→    'wc',
   678→    'rg',       // ripgrep — search only
   679→    'which',
   680→    'pwd',
   681→    'prisma',
   682→    'git status',
   683→    'git diff',
   684→    'git log',
   685→  ]
   686→
   687→  // Block dangerous flags that could allow code execution or file access
   688→  const dangerousFlags = ['-e ', '--eval ', '-r ', '--require ', '-p ', '--print ']
   689→  const normalizedCmd = command.trim()
   690→  const isAllowed = allowedPrefixes.some(prefix => normalizedCmd.startsWith(prefix))
   691→  const usesDangerousFlag = dangerousFlags.some(flag => normalizedCmd.includes(flag))
   692→
   693→  // Security: Block shell metacharacters that enable command chaining/injection
   694→  // These allow bypassing the whitelist by chaining additional commands:
   695→  //   "npm run build; rm -rf /"   →  ;  chains commands
   696→  //   "git status && cat /etc/p"  →  && chains on success
   697→  //   "echo hi | bash"            →  |  pipes output
   698→  //   "tsc || rm -rf /"           →  || chains on failure
   699→  //   "echo $(cat /etc/passwd)"   →  $() command substitution
   700→  //   "echo `cat /etc/passwd`"    →  backtick command substitution
   701→  //   "echo hi > /etc/passwd"     →  >  redirect write
   702→  //   "echo hi >> /etc/passwd"    →  >> redirect append
   703→  const shellMetacharacters = /(;|&&|\|\||\||`|\$\(|\$\{|>|>>)/
   704→  const usesShellMetacharacters = shellMetacharacters.test(normalizedCmd)
   705→
   706→  if (!isAllowed || usesDangerousFlag || usesShellMetacharacters) {
   707→    return {
   708→      success: false,
   709→      error: `Command not allowed: "${normalizedCmd.slice(0, 100)}". ${!isAllowed ? 'Prefix not in whitelist.' : ''} ${usesDangerousFlag ? 'Dangerous flag detected.' : ''} ${usesShellMetacharacters ? 'Shell metacharacter (;, &&, ||, |, `, $(), >, >>) detected — command chaining is forbidden.' : ''}`,
   710→    }
   711→  }
   712→
   713→  try {
   714→    const { exec } = await import('child_process')
   715→    const { promisify } = await import('util')
   716→    const execAsync = promisify(exec)
   717→
   718→    const { stdout, stderr } = await execAsync(normalizedCmd, {
   719→      cwd: process.cwd(),
   720→      timeout: 30000,
   721→      maxBuffer: 1024 * 1024, // 1MB
   722→    })
   723→
   724→    const output = [
   725→      stdout ? `STDOUT:\n${stdout}` : '',
   726→      stderr ? `STDERR:\n${stderr}` : '',
   727→    ].filter(Boolean).join('\n')
   728→
   729→    return { success: true, output: output || '(no output)' }
   730→  } catch (err) {
   731→    const error = err as { stdout?: string; stderr?: string; message?: string }
   732→    const output = [
   733→      error.stdout ? `STDOUT:\n${error.stdout}` : '',
   734→      error.stderr ? `STDERR:\n${error.stderr}` : '',
   735→    ].filter(Boolean).join('\n')
   736→
   737→    return {
   738→      success: false,
   739→      output: output || undefined,
   740→      error: `Command failed: ${error.message}`,
   741→    }
   742→  }
   743→}

   852→/**
   853→ * Retry a service call with the next available key from the pool.
   854→ * Called when the first key fails — tries remaining keys before falling back to SDK.
   855→ *
   856→ * @param service - 'tavily' | 'serper' | 'jina'
   857→ * @param param - The query (for search) or URL (for reader)
   858→ * @returns Result if a key succeeds, null if all keys fail
   859→ */
   860→async function retryWithNextKey(
   861→  service: 'tavily' | 'serper' | 'jina',
   862→  param: string
   863→): Promise<unknown | null> {
   864→  const pool = service === 'tavily' ? tavilyKeyPool : service === 'serper' ? serperKeyPool : jinaKeyPool
   865→  const maxRetries = pool.getTotalCount() - 1 // Already tried 1 key
   866→
   867→  for (let i = 0; i < maxRetries; i++) {
   868→    const nextKey = pool.getNextKey()
   869→    if (!nextKey || nextKey.isLastResort) break // No more healthy keys
   870→
   871→    console.log(`[ToolExecutor] ${service} retrying with key ${nextKey.index + 1}...`)
   872→
   873→    try {
   874→      let res: Response
   875→
   876→      if (service === 'tavily') {
   877→        res = await fetch('https://api.tavily.com/search', {
   878→          method: 'POST',
   879→          headers: { 'Content-Type': 'application/json' },
   880→          body: JSON.stringify({
   881→            api_key: nextKey.key,
   882→            query: param,
   883→            search_depth: 'advanced',
   884→            include_answer: true,
   885→            max_results: 5,
   886→          }),
   887→          signal: AbortSignal.timeout(30000),
   888→        })
   889→      } else if (service === 'serper') {
   890→        res = await fetch('https://google.serper.dev/search', {
   891→          method: 'POST',
   892→          headers: {
   893→            'X-API-KEY': nextKey.key,
   894→            'Content-Type': 'application/json',
   895→          },
   896→          body: JSON.stringify({ q: param, num: 5 }),
   897→          signal: AbortSignal.timeout(15000),
   898→        })
   899→      } else {
   900→        // Jina
   901→        const headers: Record<string, string> = { 'Accept': 'text/plain' }
   902→        if (nextKey.key) headers['Authorization'] = `Bearer ${nextKey.key}`
   903→        res = await fetch(`https://r.jina.ai/${param}`, {
   904→          method: 'GET',
   905→          headers,
   906→          signal: AbortSignal.timeout(30000),
   907→        })
   908→      }
   909→
   910→      if (res.ok) {
   911→        pool.reportResult(nextKey.index, true, res.status)
   912→        console.log(`[ToolExecutor] ${service} key ${nextKey.index + 1} succeeded on retry`)
   913→
   914→        if (service === 'tavily') {
   915→          const data = await res.json()
   916→          return {
   917→            answer: data.answer,
   918→            results: (data.results || []).map((r: Record<string, unknown>) => ({
   919→              title: r.title, url: r.url, content: r.content, score: r.score,
   920→            })),
   921→            source: `tavily-key${nextKey.index + 1}`,
   922→          }
   923→        } else if (service === 'serper') {
   924→          const data = await res.json()
   925→          return {
   926→            organic: (data.organic || []).map((r: Record<string, unknown>) => ({
   927→              title: r.title, link: r.link, snippet: r.snippet,
   928→            })),
   929→            knowledgeGraph: data.knowledgeGraph || null,
   930→            source: `serper-key${nextKey.index + 1}`,
   931→          }
   932→        } else {
   933→          // Jina
   934→          const text = await res.text()
   935→          const maxChars = 30000
   936→          const truncated = text.length > maxChars
   937→          const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
   938→          return {
   939→            url: param, content, length: text.length, truncated,
   940→            source: `jina-key${nextKey.index + 1}`,
   941→          }
   942→        }
   943→      }
   944→
   945→      // This key also failed — report and try next
   946→      const errorBody = await res.text().catch(() => '')
   947→      pool.reportResult(nextKey.index, false, res.status, `HTTP ${res.status}: ${errorBody.slice(0, 200)}`)
   948→      console.warn(`[ToolExecutor] ${service} key ${nextKey.index + 1} also failed: ${res.status}`)
   949→    } catch (err) {
   950→      const errMsg = err instanceof Error ? err.message : String(err)
   951→      pool.reportResult(nextKey.index, false, undefined, errMsg)
   952→      console.warn(`[ToolExecutor] ${service} key ${nextKey.index + 1} exception: ${errMsg}`)
   953→    }
   954→  }
   955→
   956→  return null // All keys failed
   957→}

  1118→      if (Array.isArray(results) && results.length > 0) {
  1119→        return {
  1120→          organic: results.map((r: any) => ({
  1121→            title: r.name || r.title,
  1122→            link: r.url,
  1123→            snippet: r.snippet || '',
  1124→          })),
  1125→          knowledgeGraph: null,

  1134→  return {

  1136→    hint: !serperKeyPool.hasKeys() ? 'Add SERPER_API_KEY_1..4 to .env' : `All ${serperKeyPool.getTotalCount()} keys failed (${serperKeyPool.getAvailableCount()} available)`,
  1137→    poolStatus: serperKeyPool.getSummary(),

  1141→/**
  1142→ * Jina — Web page reader
  1143→ * Uses ServiceKeyPool for multi-key rotation with automatic fallback.
  1144→ * Primary: Direct Jina API with rotating keys from pool
  1145→ * Fallback: z-ai-web-dev-sdk page_reader function (Clawhub Market skill)
  1146→ */
  1147→async function executeJina(args: Record<string, unknown>): Promise<unknown> {
  1148→  const url = args.url as string
  1149→  if (!url) return { error: 'URL is required' }
  1150→
  1151→  // Validate URL
  1152→  try {
  1153→    new URL(url)
  1154→  } catch {
  1155→    return { error: `Invalid URL: ${url}` }
  1156→  }
  1157→
  1158→  // Primary: Direct Jina API with key pool rotation
  1159→  if (jinaKeyPool.hasKeys()) {
  1160→    const keySelection = jinaKeyPool.getNextKey()
  1161→    if (keySelection) {
  1162→      try {
  1163→        const headers: Record<string, string> = {
  1164→          'Accept': 'text/plain',
  1165→        }
  1166→        if (keySelection.key) {
  1167→          headers['Authorization'] = `Bearer ${keySelection.key}`
  1168→        }
  1169→
  1170→        const res = await fetch(`https://r.jina.ai/${url}`, {
  1171→          method: 'GET',
  1172→          headers,
  1173→          signal: AbortSignal.timeout(30000),
  1174→        })
  1175→
  1176→        if (res.ok) {
  1177→          jinaKeyPool.reportResult(keySelection.index, true, res.status)
  1178→          const text = await res.text()
  1179→
  1180→          // Limit output to prevent overwhelming LLM context
  1181→          const maxChars = 30000
  1182→          const truncated = text.length > maxChars
  1183→          const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
  1184→
  1185→          return {
  1186→            url,
  1187→            content,
  1188→            length: text.length,
  1189→            truncated,
  1190→            source: `jina-key${keySelection.index + 1}`,
  1191→          }
  1192→        }
  1193→
  1194→        // API returned error status — report to pool and try next key
  1195→        const errorBody = await res.text().catch(() => '')
  1196→        jinaKeyPool.reportResult(keySelection.index, false, res.status, `HTTP ${res.status}: ${errorBody.slice(0, 200)}`)
  1197→        console.warn(`[ToolExecutor] Jina key ${keySelection.index + 1} error: ${res.status}`)
  1198→
  1199→        // Try remaining keys before falling back to SDK
  1200→        const retryResult = await retryWithNextKey('jina', url)
  1201→        if (retryResult) return retryResult
  1202→      } catch (err) {
  1203→        const errMsg = err instanceof Error ? err.message : String(err)
  1204→        jinaKeyPool.reportResult(keySelection.index, false, undefined, errMsg)
  1205→        console.warn(`[ToolExecutor] Jina key ${keySelection.index + 1} failed: ${errMsg}`)
  1206→
  1207→        // Try remaining keys before falling back to SDK
  1208→        const retryResult = await retryWithNextKey('jina', url)
  1209→        if (retryResult) return retryResult
  1210→      }
  1211→    }
  1212→  }
  1213→
  1214→  // Fallback: z-ai-web-dev-sdk page_reader
  1215→  try {
  1216→    const zai = await getZAI()
  1217→    if (zai) {
  1218→      const result = await zai.functions.invoke('page_reader', { url })
  1219→      if (result?.data?.html) {
  1220→        // Convert HTML to clean text
  1221→        const text = result.data.html
  1222→          .replace(/<script[^>]*>[\s\S]*?<\/script>/gi, '')
  1223→          .replace(/<style[^>]*>[\s\S]*?<\/style>/gi, '')
  1224→          .replace(/<[^>]*>/g, ' ')
  1225→          .replace(/&nbsp;/g, ' ')
  1226→          .replace(/&amp;/g, '&')
  1227→          .replace(/&lt;/g, '<')
  1228→          .replace(/&gt;/g, '>')
  1229→          .replace(/\s+/g, ' ')
  1230→          .trim()
  1231→
  1232→        const maxChars = 30000
  1233→        const truncated = text.length > maxChars
  1234→        const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
  1235→
  1236→        return {
  1237→          url,
  1238→          title: result.data.title || null,
  1239→          content,
  1240→          length: text.length,
  1241→          truncated,
  1242→          source: 'z-ai-sdk-page_reader',
  1243→        }
  1244→      }
  1245→    }
  1246→  } catch (err) {
  1247→    console.warn(`[ToolExecutor] z-ai-sdk page_reader fallback failed: ${err instanceof Error ? err.message : String(err)}`)
  1248→  }
  1249→
  1250→  return {
  1251→    error: 'Jina reader failed and no fallback available.',
  1252→    hint: !jinaKeyPool.hasKeys() ? 'Add JINA_API_KEY_1..2 to .env' : `All ${jinaKeyPool.getTotalCount()} keys failed (${jinaKeyPool.getAvailableCount()} available)`,
  1253→    poolStatus: jinaKeyPool.getSummary(),
  1254→  }
  1255→}

  1321→// ==================== REACT LOOP HELPER ====================

  1323→/**
  1324→ * Process a single LLM response — check for tool_calls and execute them.
  1325→ * Returns array of tool execution results for feeding back to LLM.
  1326→ *
  1327→ * This is the core of the ReAct loop:
  1328→ * 1. LLM returns response with tool_calls
  1329→ * 2. We execute each tool_call
  1330→ * 3. Results are fed back as tool messages
  1331→ * 4. LLM generates next response (may have more tool_calls or final text)
  1332→ */
  1333→export async function processToolCalls(
  1334→  toolCalls: Array<{ id: string; function: { name: string; arguments: string } }>,
  1335→): Promise<Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }>> {
  1336→  const results: Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }> = []
  1337→
  1338→  for (const tc of toolCalls) {
  1339→    let args: Record<string, unknown> = {}
  1340→    try {
  1341→      args = JSON.parse(tc.function.arguments)
  1342→    } catch {
  1343→      args = { _parseError: tc.function.arguments }
  1344→    }
  1345→
  1346→    console.log(`[ToolExecutor] Executing: ${tc.function.name}(${JSON.stringify(args).slice(0, 200)})`)
  1347→
  1348→    const result = await executeTool(tc.function.name, args)
  1349→
  1350→    results.push({
  1351→      toolCallId: tc.id,
  1352→      toolName: tc.function.name,
  1353→      result,
  1354→    })
  1355→
  1356→    console.log(`[ToolExecutor] Result: ${tc.function.name} → ${result.success ? 'OK' : 'FAIL'} (${result.duration}ms)`)
  1357→  }
  1358→
  1359→  return results
  1360→}

  1362→/**
  1363→ * Format tool results into messages for the LLM.
  1364→ * Each tool result becomes a "tool" role message.
  1365→ * Note: Duration is excluded from LLM messages to save tokens — LLM doesn't need timing info.
  1366→ */
  1367→export function formatToolResultsAsMessages(
  1368→  results: Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }>,
  1369→): Array<{ role: 'tool'; tool_call_id: string; content: string }> {

  1372→    tool_call_id: r.toolCallId,
  1373→    content: JSON.stringify({
  1374→      tool: r.toolName,
  1375→      success: r.result.success,

  1377→    }),
  1378→  }))
  1379→}
