   312→   312→/**
   313→   313→ * Execute a tool via Gateway HTTP /tools/invoke endpoint.
   314→   314→ * Only works for HTTP-allowed tools (web_search, web_fetch, etc.).
   315→   315→ */
   316→   316→async function executeGatewayTool(name: string, args: Record<string, unknown>): Promise<unknown> {
   317→   317→  if (!GATEWAY_HTTP_ALLOWED_TOOLS.has(name)) {
   318→   318→    throw new Error(`Tool "${name}" requires Gateway session context. Try the Bridge instead.`)
   319→   319→  }
   320→   320→
   321→   321→  const res = await fetch(`${GATEWAY_URL}/tools/invoke`, {
   322→   322→    method: 'POST',
   323→   323→    headers: { 'Content-Type': 'application/json' },
   324→   324→    body: JSON.stringify({ name, args }),
   325→   325→    signal: AbortSignal.timeout(30000),
   326→   326→  })
   327→   327→
   328→   328→  if (!res.ok) {
   329→   329→    const errorText = await res.text().catch(() => 'Unknown error')
   330→   330→    throw new Error(`Gateway tool error ${res.status}: ${errorText.slice(0, 200)}`)
   331→   331→  }
   332→   332→
   333→   333→  const data = await res.json()
   334→   334→  if (!data.ok && data.error) {
   335→   335→    throw new Error(`Tool error: ${data.error?.message || data.error || 'Unknown'}`)
   336→   336→  }
   337→   337→
   338→   338→  return data.result ?? data
   339→   339→}
   340→   340→
   341→   341→/**
   342→   342→ * Execute a tool via the Gateway Bridge (port 18791).
   343→   343→ * The Bridge connects to the Gateway via WebSocket with full session context,
   344→   344→ * enabling access to ALL 36 Gateway tools including session-context ones
   345→   345→ * (exec, read, write, edit, image_generate, etc.).
   346→   346→ */
   347→   347→async function executeBridgeTool(name: string, args: Record<string, unknown>): Promise<unknown> {
   348→   348→  const res = await fetch(`${BRIDGE_URL}/tools/invoke`, {
   349→   349→    method: 'POST',
   350→   350→    headers: { 'Content-Type': 'application/json' },
   351→   351→    body: JSON.stringify({ name, args }),
   352→   352→    signal: AbortSignal.timeout(60000),
   353→   353→  })
   354→   354→
   355→   355→  if (!res.ok) {
   356→   356→    const errorText = await res.text().catch(() => 'Unknown error')
   357→   357→    throw new Error(`Bridge tool error ${res.status}: ${errorText.slice(0, 200)}`)
   358→   358→  }
   359→   359→
   360→   360→  const data = await res.json()
   361→   361→  if (!data.ok && data.error) {
   362→   362→    throw new Error(`Bridge tool error: ${typeof data.error === 'string' ? data.error : data.error?.message || 'Unknown'}`)
   363→   363→  }
   364→   364→
   365→   365→  return data.result ?? data
   366→   366→}

   369→   369→ * Check if the Gateway Bridge is online and authenticated.

   371→   371→export async function isBridgeOnline(): Promise<boolean> {
   372→   372→  try {
   373→   373→    const res = await fetch(`${BRIDGE_URL}/health`, { signal: AbortSignal.timeout(3000) })
   374→   374→    if (res.ok) {
   375→   375→      const data = await res.json()
   376→   376→      return data?.bridge?.authenticated === true
   377→   377→    }
   378→   378→  } catch {}
   379→   379→  return false
   380→   380→}

   382→   382→// ==================== Z-AI-SDK LAZY INITIALIZATION ====================
   383→   383→//
   384→   384→// ⚠️  IMPORTANT: z-ai-web-dev-sdk is an OPTIONAL dependency!
   385→   385→//
   386→   386→// This SDK ONLY works inside the z.ai sandbox environment where backend
   387→   387→// services (web_search, page_reader) are available. If you clone this
   388→   388→// project and run locally, the SDK will not initialize — this is expected
   389→   389→// and NOT an error. Direct API keys (TAVILY_API_KEY, SERPER_API_KEY,
   390→   390→// JINA_API_KEY) in .env will be used instead.
   391→   391→//
   392→   392→// Fallback chain: Direct API key → z-ai-sdk → error with helpful hint
   393→   393→//
   394→   394→
   395→   395→let zaiInstance: any = null
   396→   396→let zaiInitPromise: Promise<any> | null = null
   397→   397→let zaiAvailable: boolean | null = null // null = not yet checked
   398→   398→
   399→   399→async function getZAI(): Promise<any> {
   400→   400→  if (zaiInstance) return zaiInstance
   401→   401→  if (zaiInitPromise) return zaiInitPromise
   402→   402→
   403→   403→  zaiInitPromise = (async () => {
   404→   404→    try {
   405→   405→      const ZAI = (await import('z-ai-web-dev-sdk')).default
   406→   406→      const instance = await ZAI.create()
   407→   407→      console.log('[ToolExecutor] ✅ z-ai-web-dev-sdk initialized — web_search + page_reader available as fallback')
   408→   408→      zaiInstance = instance
   409→   409→      zaiAvailable = true
   410→   410→      return instance
   411→   411→    } catch (err) {
   412→   412→      zaiAvailable = false
   413→   413→      console.warn(
   414→   414→        '[ToolExecutor] z-ai-web-dev-sdk not available (this is normal for local deployment — direct API keys will be used).',
   415→   415→        `Detail: ${err instanceof Error ? err.message : String(err)}`
   416→   416→      )
   417→   417→      return null
   418→   418→    }
   419→   419→  })()
   420→   420→
   421→   421→  return zaiInitPromise
   422→   422→}

   623→   623→export interface ToolExecutionResult {

   625→   625→  result: unknown
   626→   626→  duration: number

   629→   629→/**
   630→   630→ * Execute a tool call from LLM.
   631→   631→ * Called in the ReAct loop when LLM returns a tool_call.
   632→   632→ *
   633→   633→ * @param toolName - Name of the tool to execute
   634→   634→ * @param args - Arguments from LLM's tool_call
   635→   635→ * @returns Execution result with success status
   636→   636→ */
   637→   637→export async function executeTool(
   638→   638→  toolName: string,
   639→   639→  args: Record<string, unknown>
   640→   640→): Promise<ToolExecutionResult> {
   641→   641→  const startTime = Date.now()
   642→   642→
   643→   643→  try {
   644→   644→    let result: unknown
   645→   645→
   646→   646→    switch (toolName) {
   647→   647→      case 'opencode':
   648→   648→        result = await executeOpenCode(args)
   649→   649→        break
   650→   650→      case 'knowledge_search':
   651→   651→        result = await executeKnowledgeSearch(args)
   652→   652→        break
   653→   653→      case 'knowledge_graph':
   654→   654→        result = await executeKnowledgeGraph(args)
   655→   655→        break
   656→   656→      case 'knowledge_write':
   657→   657→        result = await executeKnowledgeWrite(args)
   658→   658→        break
   659→   659→      case 'tavily':
   660→   660→        result = await executeTavily(args)
   661→   661→        break
   662→   662→      case 'serper':
   663→   663→        result = await executeSerper(args)
   664→   664→        break
   665→   665→      case 'jina':
   666→   666→        result = await executeJina(args)
   667→   667→        break
   668→   668→      default: {
   669→   669→        // V2: Route to Gateway HTTP, Bridge, or error
   670→   670→        // 1. Try Gateway HTTP for HTTP-accessible tools
   671→   671→        if (GATEWAY_HTTP_ALLOWED_TOOLS.has(toolName)) {
   672→   672→          try {
   673→   673→            const gwResult = await executeGatewayTool(toolName, args)
   674→   674→            return { success: true, result: gwResult, duration: Date.now() - startTime }
   675→   675→          } catch (gwErr) {
   676→   676→            return {
   677→   677→              success: false,
   678→   678→              result: `Gateway tool "${toolName}" failed: ${gwErr instanceof Error ? gwErr.message : String(gwErr)}`,
   679→   679→              duration: Date.now() - startTime,
   680→   680→            }
   681→   681→          }
   682→   682→        }
   683→   683→        // 2. Try Gateway Bridge for session-context tools (exec, read, write, image_generate, etc.)
   684→   684→        if (GATEWAY_BRIDGE_TOOLS.has(toolName)) {
   685→   685→          try {
   686→   686→            const bridgeResult = await executeBridgeTool(toolName, args)
   687→   687→            return { success: true, result: bridgeResult, duration: Date.now() - startTime }
   688→   688→          } catch (bridgeErr) {
   689→   689→            return {
   690→   690→              success: false,
   691→   691→              result: `Bridge tool "${toolName}" failed: ${bridgeErr instanceof Error ? bridgeErr.message : String(bridgeErr)}`,
   692→   692→              duration: Date.now() - startTime,
   693→   693→            }
   694→   694→          }
   695→   695→        }
   696→   696→        return { success: false, result: `Unknown tool: ${toolName}. Available: local(7) + Gateway HTTP(${GATEWAY_HTTP_ALLOWED_TOOLS.size}) + Bridge(${GATEWAY_BRIDGE_TOOLS.size}) = ${7 + GATEWAY_HTTP_ALLOWED_TOOLS.size + GATEWAY_BRIDGE_TOOLS.size} tools total`, duration: Date.now() - startTime }
   697→   697→      }
   698→   698→    }
   699→   699→
   700→   700→    return {
   701→   701→      success: true,
   702→   702→      result,
   703→   703→      duration: Date.now() - startTime,
   704→   704→    }
   705→   705→  } catch (err) {
   706→   706→    const errorMsg = err instanceof Error ? err.message : String(err)
   707→   707→    console.error(`[ToolExecutor] Error executing ${toolName}:`, errorMsg)
   708→   708→    return {
   709→   709→      success: false,
   710→   710→      result: `Tool error: ${errorMsg}`,
   711→   711→      duration: Date.now() - startTime,
   712→   712→    }
   713→   713→  }
   714→   714→}

   718→   718→/**
   719→   719→ * OpenCode — Code operations (read, write, bash, mkdir)
   720→   720→ *
   721→   721→ * Uses the OpenCode API server (configured via OPENCODE_SERVER_URL env var) if available,
   722→   722→ * otherwise falls back to direct file system operations via the project API.
   723→   723→ */
   724→   724→async function executeOpenCode(args: Record<string, unknown>): Promise<unknown> {
   725→   725→  const action = args.action as string
   726→   726→  const path = args.path as string | undefined
   727→   727→  const content = args.content as string | undefined
   728→   728→  const command = args.command as string | undefined
   729→   729→
   730→   730→  // Try OpenCode server first
   731→   731→  try {
   732→   732→    const opencodeUrl = process.env.OPENCODE_SERVER_URL || 'http://localhost:18790'
   733→   733→    const res = await fetch(`${opencodeUrl}/api/execute`, {
   734→   734→      method: 'POST',
   735→   735→      headers: { 'Content-Type': 'application/json' },
   736→   736→      body: JSON.stringify({ action, path, content, command }),
   737→   737→      signal: AbortSignal.timeout(30000),
   738→   738→    })
   739→   739→
   740→   740→    if (res.ok) {
   741→   741→      const data = await res.json()
   742→   742→      return data
   743→   743→    }
   744→   744→  } catch {
   745→   745→    // OpenCode server not available, fall through to direct implementation
   746→   746→  }
   747→   747→
   748→   748→  // Fallback: Use the project's own API for file operations
   749→   749→  switch (action) {
   750→   750→    case 'read':
   751→   751→      return await executeFileRead(path || '')
   752→   752→    case 'write':
   753→   753→      return await executeFileWrite(path || '', content || '')
   754→   754→    case 'bash':
   755→   755→      return await executeBash(command || '')
   756→   756→    case 'mkdir':
   757→   757→      return { success: true, message: `mkdir not needed — write will create parent dirs` }
   758→   758→    default:
   759→   759→      return { success: false, error: `Unknown opencode action: ${action}` }
   760→   760→  }
   761→   761→}
   762→   762→
   763→   763→/**
   764→   764→ * Read a file from the project directory.
   765→   765→ * Security: Only allows reading files within the project root.
   766→   766→ * Uses path.resolve + startsWith check to prevent path traversal.
   767→   767→ */
   768→   768→async function executeFileRead(filePath: string): Promise<{ success: boolean; content?: string; error?: string }> {
   769→   769→  if (!filePath) return { success: false, error: 'Path is required' }
   770→   770→
   771→   771→  try {
   772→   772→    const pathModule = await import('path')
   773→   773→    const projectRoot = process.cwd()
   774→   774→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   775→   775→
   776→   776→    // Security: Prevent path traversal — resolved path must be within project root
   777→   777→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   778→   778→      return { success: false, error: `Access denied: path escapes project root` }
   779→   779→    }
   780→   780→
   781→   781→    const fs = await import('fs/promises')
   782→   782→    const content = await fs.readFile(resolvedPath, 'utf-8')
   783→   783→
   784→   784→    // Limit output to prevent overwhelming LLM context
   785→   785→    const maxChars = 50000
   786→   786→    const truncated = content.length > maxChars
   787→   787→    const output = truncated ? content.slice(0, maxChars) + '\n\n... [TRUNCATED — file too large]' : content
   788→   788→
   789→   789→    return { success: true, content: output }
   790→   790→  } catch (err) {
   791→   791→    return {
   792→   792→      success: false,
   793→   793→      error: `Failed to read ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   794→   794→    }
   795→   795→  }
   796→   796→}
   797→   797→
   798→   798→/**
   799→   799→ * Write a file to the project directory.
   800→   800→ * Security: Only allows writing within the project root. Uses path.resolve to prevent traversal.
   801→   801→ * Auto-creates parent directories.
   802→   802→ */
   803→   803→async function executeFileWrite(filePath: string, content: string): Promise<{ success: boolean; message: string; error?: string }> {
   804→   804→  if (!filePath) return { success: false, message: '', error: 'Path is required' }
   805→   805→  if (!content) return { success: false, message: '', error: 'Content is required' }
   806→   806→
   807→   807→  try {
   808→   808→    const pathModule = await import('path')
   809→   809→    const projectRoot = process.cwd()
   810→   810→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   811→   811→
   812→   812→    // Security: Prevent path traversal — resolved path must be within project root
   813→   813→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   814→   814→      return { success: false, message: '', error: 'Access denied: path escapes project root' }
   815→   815→    }
   816→   816→
   817→   817→    const fs = await import('fs/promises')
   818→   818→
   819→   819→    // Auto-create parent directories
   820→   820→    const dir = pathModule.dirname(resolvedPath)
   821→   821→    await fs.mkdir(dir, { recursive: true })
   822→   822→
   823→   823→    // Write file
   824→   824→    await fs.writeFile(resolvedPath, content, 'utf-8')
   825→   825→
   826→   826→    return { success: true, message: `File written: ${filePath} (${content.length} chars)` }
   827→   827→  } catch (err) {
   828→   828→    return {
   829→   829→      success: false,
   830→   830→      message: '',
   831→   831→      error: `Failed to write ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   832→   832→    }
   833→   833→  }
   834→   834→}

   836→   836→/**
   837→   837→ * Execute a bash command safely.
   838→   838→ * Security: Only allows pre-approved commands.
   839→   839→ */
   840→   840→async function executeBash(command: string): Promise<{ success: boolean; output?: string; error?: string }> {
   841→   841→  if (!command) return { success: false, error: 'Command is required' }
   842→   842→
   843→   843→  // Security: Whitelist of allowed commands
   844→   844→  // NOTE: 'node' is excluded because `node -e "..."` can execute arbitrary code.
   845→   845→  //       'cat'/'head'/'tail'/'grep' are excluded — use opencode read instead.
   846→   846→  //       'npx'/'bunx' are excluded — can download+execute arbitrary npm packages.
   847→   847→  const allowedPrefixes = [
   848→   848→    'npm run',
   849→   849→    'npm test',
   850→   850→    'npm lint',
   851→   851→    'bun run',
   852→   852→    'bun test',
   853→   853→    'bun lint',
   854→   854→    'tsc',
   855→   855→    'eslint',
   856→   856→    'prettier',
   857→   857→    'echo',
   858→   858→    'ls',
   859→   859→    'wc',
   860→   860→    'rg',       // ripgrep — search only
   861→   861→    'which',
   862→   862→    'pwd',
   863→   863→    'prisma',
   864→   864→    'git status',
   865→   865→    'git diff',
   866→   866→    'git log',
   867→   867→  ]
   868→   868→
   869→   869→  // Block dangerous flags that could allow code execution or file access
   870→   870→  const dangerousFlags = ['-e ', '--eval ', '-r ', '--require ', '-p ', '--print ']
   871→   871→  const normalizedCmd = command.trim()
   872→   872→  const isAllowed = allowedPrefixes.some(prefix => normalizedCmd.startsWith(prefix))
   873→   873→  const usesDangerousFlag = dangerousFlags.some(flag => normalizedCmd.includes(flag))
   874→   874→
   875→   875→  // Security: Block shell metacharacters that enable command chaining/injection
   876→   876→  // These allow bypassing the whitelist by chaining additional commands:
   877→   877→  //   "npm run build; rm -rf /"   →  ;  chains commands
   878→   878→  //   "git status && cat /etc/p"  →  && chains on success
   879→   879→  //   "echo hi | bash"            →  |  pipes output
   880→   880→  //   "tsc || rm -rf /"           →  || chains on failure
   881→   881→  //   "echo $(cat /etc/passwd)"   →  $() command substitution
   882→   882→  //   "echo `cat /etc/passwd`"    →  backtick command substitution
   883→   883→  //   "echo hi > /etc/passwd"     →  >  redirect write
   884→   884→  //   "echo hi >> /etc/passwd"    →  >> redirect append
   885→   885→  const shellMetacharacters = /(;|&&|\|\||\||`|\$\(|\$\{|>|>>)/
   886→   886→  const usesShellMetacharacters = shellMetacharacters.test(normalizedCmd)
   887→   887→
   888→   888→  if (!isAllowed || usesDangerousFlag || usesShellMetacharacters) {
   889→   889→    return {
   890→   890→      success: false,
   891→   891→      error: `Command not allowed: "${normalizedCmd.slice(0, 100)}". ${!isAllowed ? 'Prefix not in whitelist.' : ''} ${usesDangerousFlag ? 'Dangerous flag detected.' : ''} ${usesShellMetacharacters ? 'Shell metacharacter (;, &&, ||, |, `, $(), >, >>) detected — command chaining is forbidden.' : ''}`,
   892→   892→    }
   893→   893→  }
   894→   894→
   895→   895→  try {
   896→   896→    const { exec } = await import('child_process')
   897→   897→    const { promisify } = await import('util')
   898→   898→    const execAsync = promisify(exec)
   899→   899→
   900→   900→    const { stdout, stderr } = await execAsync(normalizedCmd, {
   901→   901→      cwd: process.cwd(),
   902→   902→      timeout: 30000,
   903→   903→      maxBuffer: 1024 * 1024, // 1MB
   904→   904→    })
   905→   905→
   906→   906→    const output = [
   907→   907→      stdout ? `STDOUT:\n${stdout}` : '',
   908→   908→      stderr ? `STDERR:\n${stderr}` : '',
   909→   909→    ].filter(Boolean).join('\n')
   910→   910→
   911→   911→    return { success: true, output: output || '(no output)' }
   912→   912→  } catch (err) {
   913→   913→    const error = err as { stdout?: string; stderr?: string; message?: string }
   914→   914→    const output = [
   915→   915→      error.stdout ? `STDOUT:\n${error.stdout}` : '',
   916→   916→      error.stderr ? `STDERR:\n${error.stderr}` : '',
   917→   917→    ].filter(Boolean).join('\n')
   918→   918→
   919→   919→    return {
   920→   920→      success: false,
   921→   921→      output: output || undefined,
   922→   922→      error: `Command failed: ${error.message}`,
   923→   923→    }
   924→   924→  }
   925→   925→}
   926→   926→
   927→   927→/**
   928→   928→ * Knowledge Search — Semantic search in KB (Qdrant + Neo4j)
   929→   929→ * Uses knowledge-bridge which already handles embedding + search.
   930→   930→ */
   931→   931→async function executeKnowledgeSearch(args: Record<string, unknown>): Promise<unknown> {
   932→   932→  const query = args.query as string
   933→   933→  const topK = Math.min((args.topK as number) || 5, 20)
   934→   934→
   935→   935→  if (!query) return { error: 'Query is required' }
   936→   936→
   937→   937→  try {
   938→   938→    const { agentKnowledgeSearch } = await import('@/lib/knowledge-bridge')
   939→   939→    const result = await agentKnowledgeSearch(query, { topK })
   940→   940→

  1361→  1361→
  1362→  1362→          // Limit output to prevent overwhelming LLM context
  1363→  1363→          const maxChars = 30000
  1364→  1364→          const truncated = text.length > maxChars
  1365→  1365→          const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
  1366→  1366→
  1367→  1367→          return {
  1368→  1368→            url,
  1369→  1369→            content,
  1370→  1370→            length: text.length,
  1371→  1371→            truncated,
  1372→  1372→            source: `jina-key${keySelection.index + 1}`,
  1373→  1373→          }
  1374→  1374→        }
  1375→  1375→
  1376→  1376→        // API returned error status — report to pool and try next key
  1377→  1377→        const errorBody = await res.text().catch(() => '')
  1378→  1378→        jinaKeyPool.reportResult(keySelection.index, false, res.status, `HTTP ${res.status}: ${errorBody.slice(0, 200)}`)
  1379→  1379→        console.warn(`[ToolExecutor] Jina key ${keySelection.index + 1} error: ${res.status}`)
  1380→  1380→
  1381→  1381→        // Try remaining keys before falling back to SDK
  1382→  1382→        const retryResult = await retryWithNextKey('jina', url)
  1383→  1383→        if (retryResult) return retryResult
  1384→  1384→      } catch (err) {
  1385→  1385→        const errMsg = err instanceof Error ? err.message : String(err)
  1386→  1386→        jinaKeyPool.reportResult(keySelection.index, false, undefined, errMsg)
  1387→  1387→        console.warn(`[ToolExecutor] Jina key ${keySelection.index + 1} failed: ${errMsg}`)
  1388→  1388→
  1389→  1389→        // Try remaining keys before falling back to SDK
  1390→  1390→        const retryResult = await retryWithNextKey('jina', url)
  1391→  1391→        if (retryResult) return retryResult
  1392→  1392→      }
  1393→  1393→    }
  1394→  1394→  }
  1395→  1395→
  1396→  1396→  // Fallback: z-ai-web-dev-sdk page_reader
  1397→  1397→  try {
  1398→  1398→    const zai = await getZAI()
  1399→  1399→    if (zai) {
  1400→  1400→      const result = await zai.functions.invoke('page_reader', { url })
  1401→  1401→      if (result?.data?.html) {
  1402→  1402→        // Convert HTML to clean text
  1403→  1403→        const text = result.data.html
  1404→  1404→          .replace(/<script[^>]*>[\s\S]*?<\/script>/gi, '')
  1405→  1405→          .replace(/<style[^>]*>[\s\S]*?<\/style>/gi, '')
  1406→  1406→          .replace(/<[^>]*>/g, ' ')
  1407→  1407→          .replace(/&nbsp;/g, ' ')
  1408→  1408→          .replace(/&amp;/g, '&')
  1409→  1409→          .replace(/&lt;/g, '<')
  1410→  1410→          .replace(/&gt;/g, '>')
  1411→  1411→          .replace(/\s+/g, ' ')
  1412→  1412→          .trim()
  1413→  1413→
  1414→  1414→        const maxChars = 30000
  1415→  1415→        const truncated = text.length > maxChars
  1416→  1416→        const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
  1417→  1417→
  1418→  1418→        return {
  1419→  1419→          url,
  1420→  1420→          title: result.data.title || null,
  1421→  1421→          content,
  1422→  1422→          length: text.length,
  1423→  1423→          truncated,
  1424→  1424→          source: 'z-ai-sdk-page_reader',
  1425→  1425→        }
  1426→  1426→      }
  1427→  1427→    }
  1428→  1428→  } catch (err) {
  1429→  1429→    console.warn(`[ToolExecutor] z-ai-sdk page_reader fallback failed: ${err instanceof Error ? err.message : String(err)}`)
  1430→  1430→  }
  1431→  1431→
  1432→  1432→  return {
  1433→  1433→    error: 'Jina reader failed and no fallback available.',
  1434→  1434→    hint: !jinaKeyPool.hasKeys() ? 'Add JINA_API_KEY_1..2 to .env' : `All ${jinaKeyPool.getTotalCount()} keys failed (${jinaKeyPool.getAvailableCount()} available)`,
  1435→  1435→    poolStatus: jinaKeyPool.getSummary(),
  1436→  1436→  }
  1437→  1437→}

  1505→  1505→/**
  1506→  1506→ * Process a single LLM response — check for tool_calls and execute them.
  1507→  1507→ * Returns array of tool execution results for feeding back to LLM.
  1508→  1508→ *
  1509→  1509→ * This is the core of the ReAct loop:
  1510→  1510→ * 1. LLM returns response with tool_calls
  1511→  1511→ * 2. We execute each tool_call
  1512→  1512→ * 3. Results are fed back as tool messages
  1513→  1513→ * 4. LLM generates next response (may have more tool_calls or final text)
  1514→  1514→ */
  1515→  1515→export async function processToolCalls(
  1516→  1516→  toolCalls: Array<{ id: string; function: { name: string; arguments: string } }>,
  1517→  1517→): Promise<Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }>> {
  1518→  1518→  const results: Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }> = []
  1519→  1519→
  1520→  1520→  for (const tc of toolCalls) {
  1521→  1521→    let args: Record<string, unknown> = {}
  1522→  1522→    try {
  1523→  1523→      args = JSON.parse(tc.function.arguments)
  1524→  1524→    } catch {
  1525→  1525→      args = { _parseError: tc.function.arguments }
  1526→  1526→    }
  1527→  1527→
  1528→  1528→    console.log(`[ToolExecutor] Executing: ${tc.function.name}(${JSON.stringify(args).slice(0, 200)})`)
  1529→  1529→
  1530→  1530→    const result = await executeTool(tc.function.name, args)
  1531→  1531→
  1532→  1532→    results.push({
  1533→  1533→      toolCallId: tc.id,
  1534→  1534→      toolName: tc.function.name,
  1535→  1535→      result,
  1536→  1536→    })
  1537→  1537→
  1538→  1538→    console.log(`[ToolExecutor] Result: ${tc.function.name} → ${result.success ? 'OK' : 'FAIL'} (${result.duration}ms)`)
  1539→  1539→  }
  1540→  1540→
  1541→  1541→  return results
  1542→  1542→}
