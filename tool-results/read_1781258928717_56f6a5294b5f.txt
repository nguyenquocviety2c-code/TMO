    76→/**
    77→ * Execute a tool via Gateway HTTP /tools/invoke endpoint.
    78→ * Only works for HTTP-allowed tools (web_search, web_fetch, etc.).
    79→ */
    80→async function executeGatewayTool(name: string, args: Record<string, unknown>): Promise<unknown> {
    81→  if (!GATEWAY_HTTP_ALLOWED_TOOLS.has(name)) {
    82→    throw new Error(`Tool "${name}" requires Gateway session context. Try the Bridge instead.`)
    83→  }
    84→
    85→  const res = await fetch(`${GATEWAY_URL}/tools/invoke`, {
    86→    method: 'POST',
    87→    headers: { 'Content-Type': 'application/json' },
    88→    body: JSON.stringify({ name, args }),
    89→    signal: AbortSignal.timeout(30000),
    90→  })
    91→
    92→  if (!res.ok) {
    93→    const errorText = await res.text().catch(() => 'Unknown error')
    94→    throw new Error(`Gateway tool error ${res.status}: ${errorText.slice(0, 200)}`)
    95→  }
    96→
    97→  const data = await res.json()
    98→  if (!data.ok && data.error) {
    99→    throw new Error(`Tool error: ${data.error?.message || data.error || 'Unknown'}`)
   100→  }
   101→
   102→  return data.result ?? data
   103→}

   105→/**
   106→ * Execute a tool via the Gateway Bridge (port 18791).
   107→ * The Bridge connects to the Gateway via WebSocket with full session context,
   108→ * enabling access to ALL 36 Gateway tools including session-context ones
   109→ * (exec, read, write, edit, image_generate, etc.).
   110→ */
   111→async function executeBridgeTool(name: string, args: Record<string, unknown>): Promise<unknown> {
   112→  const res = await fetch(`${BRIDGE_URL}/tools/invoke`, {
   113→    method: 'POST',
   114→    headers: { 'Content-Type': 'application/json' },
   115→    body: JSON.stringify({ name, args }),
   116→    signal: AbortSignal.timeout(60000),
   117→  })
   118→
   119→  if (!res.ok) {
   120→    const errorText = await res.text().catch(() => 'Unknown error')
   121→    throw new Error(`Bridge tool error ${res.status}: ${errorText.slice(0, 200)}`)
   122→  }
   123→
   124→  const data = await res.json()
   125→  if (!data.ok && data.error) {
   126→    throw new Error(`Bridge tool error: ${typeof data.error === 'string' ? data.error : data.error?.message || 'Unknown'}`)
   127→  }
   128→
   129→  return data.result ?? data
   130→}

   159→let zaiInstance: any = null
   160→let zaiInitPromise: Promise<any> | null = null
   161→let zaiAvailable: boolean | null = null // null = not yet checked
   162→
   163→async function getZAI(): Promise<any> {
   164→  if (zaiInstance) return zaiInstance
   165→  if (zaiInitPromise) return zaiInitPromise
   166→
   167→  zaiInitPromise = (async () => {
   168→    try {
   169→      const ZAI = (await import('z-ai-web-dev-sdk')).default
   170→      const instance = await ZAI.create()
   171→      console.log('[ToolExecutor] ✅ z-ai-web-dev-sdk initialized — web_search + page_reader available as fallback')
   172→      zaiInstance = instance
   173→      zaiAvailable = true
   174→      return instance
   175→    } catch (err) {
   176→      zaiAvailable = false
   177→      console.warn(
   178→        '[ToolExecutor] z-ai-web-dev-sdk not available (this is normal for local deployment — direct API keys will be used).',
   179→        `Detail: ${err instanceof Error ? err.message : String(err)}`
   180→      )
   181→      return null
   182→    }
   183→  })()
   184→
   185→  return zaiInitPromise
   186→}

   385→// ==================== TOOL EXECUTION ====================
   386→
   387→export interface ToolExecutionResult {
   388→  success: boolean
   389→  result: unknown
   390→  duration: number
   391→}
   392→
   393→/**
   394→ * Execute a tool call from LLM.
   395→ * Called in the ReAct loop when LLM returns a tool_call.
   396→ *
   397→ * @param toolName - Name of the tool to execute
   398→ * @param args - Arguments from LLM's tool_call
   399→ * @returns Execution result with success status
   400→ */
   401→export async function executeTool(
   402→  toolName: string,
   403→  args: Record<string, unknown>
   404→): Promise<ToolExecutionResult> {
   405→  const startTime = Date.now()
   406→
   407→  // P2.3: Security check — validate tool arguments before execution
   408→  const securityCheck = sanitizeToolArgs(toolName, args)
   409→  if (!securityCheck.safe) {
   410→    return {
   411→      success: false,
   412→      result: `Security blocked: ${securityCheck.warnings.join('; ')}`,
   413→      duration: Date.now() - startTime,
   414→    }
   415→  }
   416→
   417→  // Use sanitized args for all subsequent operations
   418→  const safeArgs = securityCheck.sanitizedArgs
   419→
   420→  try {
   421→    let result: unknown
   422→
   423→    // Check if this tool is provided by a loaded plugin
   424→    const loadedPlugins = getLoadedPlugins()
   425→    for (const plugin of loadedPlugins) {
   426→      const toolDef = plugin.manifest.tools?.find(t => t.name === toolName)
   427→      if (toolDef) {
   428→        try {
   429→          const pluginResult = await executePlugin(plugin.slug, {
   430→            toolName,
   431→            args: safeArgs,
   432→            allowedApis: ['fetch'],
   433→            timeout: 30000,
   434→            maxOutputLength: 10000,
   435→          })
   436→          if (pluginResult.success) {
   437→            return { success: true, result: pluginResult.result, duration: pluginResult.duration }
   438→          } else {
   439→            return { success: false, result: pluginResult.error || 'Plugin execution failed', duration: pluginResult.duration }
   440→          }
   441→        } catch (err) {
   442→          return { success: false, result: `Plugin error: ${err instanceof Error ? err.message : String(err)}`, duration: 0 }
   443→        }
   444→      }
   445→    }
   446→
   447→    switch (toolName) {
   448→      case 'opencode':
   449→        result = await executeOpenCode(safeArgs)
   450→        break
   451→      case 'knowledge_search':
   452→        result = await executeKnowledgeSearch(safeArgs)
   453→        break
   454→      case 'knowledge_graph':
   455→        result = await executeKnowledgeGraph(safeArgs)
   456→        break
   457→      case 'knowledge_write':
   458→        result = await executeKnowledgeWrite(safeArgs)
   459→        break
   460→      case 'tavily':
   461→        result = await executeTavily(safeArgs)
   462→        break
   463→      case 'serper':
   464→        result = await executeSerper(safeArgs)
   465→        break
   466→      case 'jina':
   467→        result = await executeJina(safeArgs)
   468→        break
   469→      default: {
   470→        // V2: Route to Gateway HTTP, Bridge, or error
   471→        // 1. Try Gateway HTTP for HTTP-accessible tools
   472→        if (GATEWAY_HTTP_ALLOWED_TOOLS.has(toolName)) {
   473→          try {
   474→            const gwResult = await executeGatewayTool(toolName, safeArgs)
   475→            return { success: true, result: gwResult, duration: Date.now() - startTime }
   476→          } catch (gwErr) {
   477→            return {
   478→              success: false,
   479→              result: `Gateway tool "${toolName}" failed: ${gwErr instanceof Error ? gwErr.message : String(gwErr)}`,
   480→              duration: Date.now() - startTime,
   481→            }
   482→          }
   483→        }
   484→        // 2. Try Gateway Bridge for session-context tools (exec, read, write, image_generate, etc.)
   485→        if (GATEWAY_BRIDGE_TOOLS.has(toolName)) {
   486→          try {
   487→            const bridgeResult = await executeBridgeTool(toolName, safeArgs)
   488→            return { success: true, result: bridgeResult, duration: Date.now() - startTime }
   489→          } catch (bridgeErr) {
   490→            return {
   491→              success: false,
   492→              result: `Bridge tool "${toolName}" failed: ${bridgeErr instanceof Error ? bridgeErr.message : String(bridgeErr)}`,
   493→              duration: Date.now() - startTime,
   494→            }
   495→          }
   496→        }
   497→        return { success: false, result: `Unknown tool: ${toolName}. Available: local(7) + Gateway HTTP(${GATEWAY_HTTP_ALLOWED_TOOLS.size}) + Bridge(${GATEWAY_BRIDGE_TOOLS.size}) = ${7 + GATEWAY_HTTP_ALLOWED_TOOLS.size + GATEWAY_BRIDGE_TOOLS.size} tools total`, duration: Date.now() - startTime }
   498→      }
   499→    }
   500→
   501→    return {
   502→      success: true,
   503→      result,
   504→      duration: Date.now() - startTime,
   505→    }
   506→  } catch (err) {
   507→    const errorMsg = err instanceof Error ? err.message : String(err)
   508→    console.error(`[ToolExecutor] Error executing ${toolName}:`, errorMsg)
   509→    return {
   510→      success: false,
   511→      result: `Tool error: ${errorMsg}`,
   512→      duration: Date.now() - startTime,
   513→    }
   514→  }
   515→}

   517→// ==================== INDIVIDUAL TOOL IMPLEMENTATIONS ====================
   518→
   519→/**
   520→ * OpenCode — Code operations (read, write, bash, mkdir)
   521→ *
   522→ * Uses the OpenCode API server (configured via OPENCODE_SERVER_URL env var) if available,
   523→ * otherwise falls back to direct file system operations via the project API.
   524→ */
   525→async function executeOpenCode(args: Record<string, unknown>): Promise<unknown> {
   526→  const action = args.action as string
   527→  const path = args.path as string | undefined
   528→  const content = args.content as string | undefined
   529→  const command = args.command as string | undefined
   530→
   531→  // Try OpenCode server first
   532→  try {
   533→    const opencodeUrl = process.env.OPENCODE_SERVER_URL || 'http://localhost:18790'
   534→    const res = await fetch(`${opencodeUrl}/api/execute`, {
   535→      method: 'POST',
   536→      headers: { 'Content-Type': 'application/json' },
   537→      body: JSON.stringify({ action, path, content, command }),
   538→      signal: AbortSignal.timeout(30000),
   539→    })
   540→
   541→    if (res.ok) {
   542→      const data = await res.json()
   543→      return data
   544→    }
   545→  } catch {
   546→    // OpenCode server not available, fall through to direct implementation
   547→  }
   548→
   549→  // Fallback: Use the project's own API for file operations
   550→  switch (action) {
   551→    case 'read':
   552→      return await executeFileRead(path || '')
   553→    case 'write':
   554→      return await executeFileWrite(path || '', content || '')
   555→    case 'bash':
   556→      return await executeBash(command || '')
   557→    case 'mkdir':
   558→      return { success: true, message: `mkdir not needed — write will create parent dirs` }
   559→    default:
   560→      return { success: false, error: `Unknown opencode action: ${action}` }
   561→  }
   562→}

   564→/**
   565→ * Read a file from the project directory.
   566→ * Security: Only allows reading files within the project root.
   567→ * Uses path.resolve + startsWith check to prevent path traversal.
   568→ */
   569→async function executeFileRead(filePath: string): Promise<{ success: boolean; content?: string; error?: string }> {
   570→  if (!filePath) return { success: false, error: 'Path is required' }
   571→
   572→  try {
   573→    const pathModule = await import('path')
   574→    const projectRoot = process.cwd()
   575→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   576→
   577→    // Security: Prevent path traversal — resolved path must be within project root
   578→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   579→      return { success: false, error: `Access denied: path escapes project root` }
   580→    }
   581→
   582→    const fs = await import('fs/promises')
   583→    const content = await fs.readFile(resolvedPath, 'utf-8')
   584→
   585→    // Limit output to prevent overwhelming LLM context
   586→    const maxChars = 50000
   587→    const truncated = content.length > maxChars
   588→    const output = truncated ? content.slice(0, maxChars) + '\n\n... [TRUNCATED — file too large]' : content
   589→
   590→    return { success: true, content: output }
   591→  } catch (err) {
   592→    return {
   593→      success: false,
   594→      error: `Failed to read ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   595→    }
   596→  }
   597→}

   637→/**
   638→ * Execute a bash command safely.
   639→ * Security: Only allows pre-approved commands.
   640→ */
   641→async function executeBash(command: string): Promise<{ success: boolean; output?: string; error?: string }> {
   642→  if (!command) return { success: false, error: 'Command is required' }
   643→
   644→  // Security: Whitelist of allowed commands
   645→  // NOTE: 'node' is excluded because `node -e "..."` can execute arbitrary code.
   646→  //       'cat'/'head'/'tail'/'grep' are excluded — use opencode read instead.
   647→  //       'npx'/'bunx' are excluded — can download+execute arbitrary npm packages.
   648→  const allowedPrefixes = [
   649→    'npm run',
   650→    'npm test',
   651→    'npm lint',
   652→    'bun run',
   653→    'bun test',
   654→    'bun lint',
   655→    'tsc',
   656→    'eslint',
   657→    'prettier',
   658→    'echo',
   659→    'ls',
   660→    'wc',
   661→    'rg',       // ripgrep — search only
   662→    'which',
   663→    'pwd',
   664→    'prisma',
   665→    'git status',
   666→    'git diff',
   667→    'git log',
   668→  ]
   669→
   670→  // Block dangerous flags that could allow code execution or file access
   671→  const dangerousFlags = ['-e ', '--eval ', '-r ', '--require ', '-p ', '--print ']
   672→  const normalizedCmd = command.trim()
   673→  const isAllowed = allowedPrefixes.some(prefix => normalizedCmd.startsWith(prefix))
   674→  const usesDangerousFlag = dangerousFlags.some(flag => normalizedCmd.includes(flag))
   675→
   676→  // Security: Block shell metacharacters that enable command chaining/injection
   677→  // These allow bypassing the whitelist by chaining additional commands:
   678→  //   "npm run build; rm -rf /"   →  ;  chains commands
   679→  //   "git status && cat /etc/p"  →  && chains on success
   680→  //   "echo hi | bash"            →  |  pipes output
   681→  //   "tsc || rm -rf /"           →  || chains on failure
   682→  //   "echo $(cat /etc/passwd)"   →  $() command substitution
   683→  //   "echo `cat /etc/passwd`"    →  backtick command substitution
   684→  //   "echo hi > /etc/passwd"     →  >  redirect write
   685→  //   "echo hi >> /etc/passwd"    →  >> redirect append
   686→  const shellMetacharacters = /(;|&&|\|\||\||`|\$\(|\$\{|>|>>)/
   687→  const usesShellMetacharacters = shellMetacharacters.test(normalizedCmd)
   688→
   689→  if (!isAllowed || usesDangerousFlag || usesShellMetacharacters) {
   690→    return {
   691→      success: false,
   692→      error: `Command not allowed: "${normalizedCmd.slice(0, 100)}". ${!isAllowed ? 'Prefix not in whitelist.' : ''} ${usesDangerousFlag ? 'Dangerous flag detected.' : ''} ${usesShellMetacharacters ? 'Shell metacharacter (;, &&, ||, |, `, $(), >, >>) detected — command chaining is forbidden.' : ''}`,
   693→    }
   694→  }
   695→
   696→  try {
   697→    const { exec } = await import('child_process')
   698→    const { promisify } = await import('util')
   699→    const execAsync = promisify(exec)
   700→
   701→    const { stdout, stderr } = await execAsync(normalizedCmd, {
   702→      cwd: process.cwd(),
   703→      timeout: 30000,
   704→      maxBuffer: 1024 * 1024, // 1MB
   705→    })
   706→
   707→    const output = [
   708→      stdout ? `STDOUT:\n${stdout}` : '',
   709→      stderr ? `STDERR:\n${stderr}` : '',
   710→    ].filter(Boolean).join('\n')
   711→
   712→    return { success: true, output: output || '(no output)' }
   713→  } catch (err) {
   714→    const error = err as { stdout?: string; stderr?: string; message?: string }
   715→    const output = [
   716→      error.stdout ? `STDOUT:\n${error.stdout}` : '',
   717→      error.stderr ? `STDERR:\n${error.stderr}` : '',
   718→    ].filter(Boolean).join('\n')
   719→
   720→    return {
   721→      success: false,
   722→      output: output || undefined,
   723→      error: `Command failed: ${error.message}`,
   724→    }
   725→  }
   726→}

   835→/**
   836→ * Retry a service call with the next available key from the pool.
   837→ * Called when the first key fails — tries remaining keys before falling back to SDK.
   838→ *
   839→ * @param service - 'tavily' | 'serper' | 'jina'
   840→ * @param param - The query (for search) or URL (for reader)
   841→ * @returns Result if a key succeeds, null if all keys fail
   842→ */
   843→async function retryWithNextKey(
   844→  service: 'tavily' | 'serper' | 'jina',
   845→  param: string
   846→): Promise<unknown | null> {
   847→  const pool = service === 'tavily' ? tavilyKeyPool : service === 'serper' ? serperKeyPool : jinaKeyPool
   848→  const maxRetries = pool.getTotalCount() - 1 // Already tried 1 key
   849→
   850→  for (let i = 0; i < maxRetries; i++) {
   851→    const nextKey = pool.getNextKey()
   852→    if (!nextKey || nextKey.isLastResort) break // No more healthy keys
   853→
   854→    console.log(`[ToolExecutor] ${service} retrying with key ${nextKey.index + 1}...`)
   855→
   856→    try {
   857→      let res: Response
   858→
   859→      if (service === 'tavily') {
   860→        res = await fetch('https://api.tavily.com/search', {
   861→          method: 'POST',
   862→          headers: { 'Content-Type': 'application/json' },
   863→          body: JSON.stringify({
   864→            api_key: nextKey.key,
   865→            query: param,
   866→            search_depth: 'advanced',
   867→            include_answer: true,
   868→            max_results: 5,
   869→          }),
   870→          signal: AbortSignal.timeout(30000),
   871→        })
   872→      } else if (service === 'serper') {
   873→        res = await fetch('https://google.serper.dev/search', {
   874→          method: 'POST',
   875→          headers: {
   876→            'X-API-KEY': nextKey.key,
   877→            'Content-Type': 'application/json',
   878→          },
   879→          body: JSON.stringify({ q: param, num: 5 }),
   880→          signal: AbortSignal.timeout(15000),
   881→        })
   882→      } else {
   883→        // Jina
   884→        const headers: Record<string, string> = { 'Accept': 'text/plain' }
   885→        if (nextKey.key) headers['Authorization'] = `Bearer ${nextKey.key}`
   886→        res = await fetch(`https://r.jina.ai/${param}`, {
   887→          method: 'GET',
   888→          headers,
   889→          signal: AbortSignal.timeout(30000),
   890→        })
   891→      }
   892→
   893→      if (res.ok) {
   894→        pool.reportResult(nextKey.index, true, res.status)
   895→        console.log(`[ToolExecutor] ${service} key ${nextKey.index + 1} succeeded on retry`)
   896→
   897→        if (service === 'tavily') {
   898→          const data = await res.json()
   899→          return {
   900→            answer: data.answer,
   901→            results: (data.results || []).map((r: Record<string, unknown>) => ({
   902→              title: r.title, url: r.url, content: r.content, score: r.score,
   903→            })),
   904→            source: `tavily-key${nextKey.index + 1}`,
   905→          }
   906→        } else if (service === 'serper') {
   907→          const data = await res.json()
   908→          return {
   909→            organic: (data.organic || []).map((r: Record<string, unknown>) => ({
   910→              title: r.title, link: r.link, snippet: r.snippet,
   911→            })),
   912→            knowledgeGraph: data.knowledgeGraph || null,
   913→            source: `serper-key${nextKey.index + 1}`,
   914→          }
   915→        } else {
   916→          // Jina
   917→          const text = await res.text()
   918→          const maxChars = 30000
   919→          const truncated = text.length > maxChars
   920→          const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
   921→          return {
   922→            url: param, content, length: text.length, truncated,
   923→            source: `jina-key${nextKey.index + 1}`,
   924→          }
   925→        }
   926→      }
   927→
   928→      // This key also failed — report and try next
   929→      const errorBody = await res.text().catch(() => '')
   930→      pool.reportResult(nextKey.index, false, res.status, `HTTP ${res.status}: ${errorBody.slice(0, 200)}`)
   931→      console.warn(`[ToolExecutor] ${service} key ${nextKey.index + 1} also failed: ${res.status}`)
   932→    } catch (err) {
   933→      const errMsg = err instanceof Error ? err.message : String(err)
   934→      pool.reportResult(nextKey.index, false, undefined, errMsg)
   935→      console.warn(`[ToolExecutor] ${service} key ${nextKey.index + 1} exception: ${errMsg}`)
   936→    }
   937→  }
   938→
   939→  return null // All keys failed
   940→}

  1089→        // Try remaining keys before falling back to SDK
  1090→        const retryResult = await retryWithNextKey('serper', query)
  1091→        if (retryResult) return retryResult
  1092→      }
  1093→    }
  1094→  }
  1095→
  1096→  // Fallback: z-ai-web-dev-sdk web_search
  1097→  try {
  1098→    const zai = await getZAI()
  1099→    if (zai) {
  1100→      const results = await zai.functions.invoke('web_search', { query, num: 5 })
  1101→      if (Array.isArray(results) && results.length > 0) {
  1102→        return {
  1103→          organic: results.map((r: any) => ({
  1104→            title: r.name || r.title,
  1105→            link: r.url,
  1106→            snippet: r.snippet || '',
  1107→          })),
  1108→          knowledgeGraph: null,
  1109→          source: 'z-ai-sdk-web_search',
  1110→        }
  1111→      }
  1112→    }
  1113→  } catch (err) {
  1114→    console.warn(`[ToolExecutor] z-ai-sdk web_search fallback failed: ${err instanceof Error ? err.message : String(err)}`)
  1115→  }
  1116→
  1117→  return {
  1118→    error: 'Serper search failed and no fallback available.',
  1119→    hint: !serperKeyPool.hasKeys() ? 'Add SERPER_API_KEY_1..4 to .env' : `All ${serperKeyPool.getTotalCount()} keys failed (${serperKeyPool.getAvailableCount()} available)`,
  1120→    poolStatus: serperKeyPool.getSummary(),
  1121→  }
  1122→}
  1123→
  1124→/**
  1125→ * Jina — Web page reader
  1126→ * Uses ServiceKeyPool for multi-key rotation with automatic fallback.
  1127→ * Primary: Direct Jina API with rotating keys from pool
  1128→ * Fallback: z-ai-web-dev-sdk page_reader function (Clawhub Market skill)
  1129→ */
  1130→async function executeJina(args: Record<string, unknown>): Promise<unknown> {
  1131→  const url = args.url as string
  1132→  if (!url) return { error: 'URL is required' }
  1133→
  1134→  // Validate URL
  1135→  try {
  1136→    new URL(url)
  1137→  } catch {
  1138→    return { error: `Invalid URL: ${url}` }
  1139→  }
  1140→
  1141→  // Primary: Direct Jina API with key pool rotation
  1142→  if (jinaKeyPool.hasKeys()) {
  1143→    const keySelection = jinaKeyPool.getNextKey()
  1144→    if (keySelection) {
  1145→      try {
  1146→        const headers: Record<string, string> = {
  1147→          'Accept': 'text/plain',
  1148→        }
  1149→        if (keySelection.key) {
  1150→          headers['Authorization'] = `Bearer ${keySelection.key}`
  1151→        }
  1152→
  1153→        const res = await fetch(`https://r.jina.ai/${url}`, {
  1154→          method: 'GET',
  1155→          headers,
  1156→          signal: AbortSignal.timeout(30000),
  1157→        })
  1158→
  1159→        if (res.ok) {
  1160→          jinaKeyPool.reportResult(keySelection.index, true, res.status)
  1161→          const text = await res.text()
  1162→
  1163→          // Limit output to prevent overwhelming LLM context
  1164→          const maxChars = 30000
  1165→          const truncated = text.length > maxChars
  1166→          const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
  1167→
  1168→          return {
  1169→            url,
  1170→            content,
  1171→            length: text.length,
  1172→            truncated,
  1173→            source: `jina-key${keySelection.index + 1}`,
  1174→          }
  1175→        }
  1176→
  1177→        // API returned error status — report to pool and try next key
  1178→        const errorBody = await res.text().catch(() => '')
  1179→        jinaKeyPool.reportResult(keySelection.index, false, res.status, `HTTP ${res.status}: ${errorBody.slice(0, 200)}`)
  1180→        console.warn(`[ToolExecutor] Jina key ${keySelection.index + 1} error: ${res.status}`)
  1181→
  1182→        // Try remaining keys before falling back to SDK
  1183→        const retryResult = await retryWithNextKey('jina', url)
  1184→        if (retryResult) return retryResult
  1185→      } catch (err) {
  1186→        const errMsg = err instanceof Error ? err.message : String(err)
  1187→        jinaKeyPool.reportResult(keySelection.index, false, undefined, errMsg)
  1188→        console.warn(`[ToolExecutor] Jina key ${keySelection.index + 1} failed: ${errMsg}`)
  1189→
  1190→        // Try remaining keys before falling back to SDK
  1191→        const retryResult = await retryWithNextKey('jina', url)
  1192→        if (retryResult) return retryResult
  1193→      }
  1194→    }
  1195→  }
  1196→
  1197→  // Fallback: z-ai-web-dev-sdk page_reader
  1198→  try {
  1199→    const zai = await getZAI()
  1200→    if (zai) {
  1201→      const result = await zai.functions.invoke('page_reader', { url })
  1202→      if (result?.data?.html) {
  1203→        // Convert HTML to clean text
  1204→        const text = result.data.html
  1205→          .replace(/<script[^>]*>[\s\S]*?<\/script>/gi, '')
  1206→          .replace(/<style[^>]*>[\s\S]*?<\/style>/gi, '')
  1207→          .replace(/<[^>]*>/g, ' ')
  1208→          .replace(/&nbsp;/g, ' ')
  1209→          .replace(/&amp;/g, '&')
  1210→          .replace(/&lt;/g, '<')
  1211→          .replace(/&gt;/g, '>')
  1212→          .replace(/\s+/g, ' ')
  1213→          .trim()
  1214→
  1215→        const maxChars = 30000
  1216→        const truncated = text.length > maxChars
  1217→        const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
  1218→
  1219→        return {
  1220→          url,
  1221→          title: result.data.title || null,
  1222→          content,
  1223→          length: text.length,
  1224→          truncated,
  1225→          source: 'z-ai-sdk-page_reader',
  1226→        }
  1227→      }
  1228→    }
  1229→  } catch (err) {
  1230→    console.warn(`[ToolExecutor] z-ai-sdk page_reader fallback failed: ${err instanceof Error ? err.message : String(err)}`)
  1231→  }
  1232→
  1233→  return {
  1234→    error: 'Jina reader failed and no fallback available.',
  1235→    hint: !jinaKeyPool.hasKeys() ? 'Add JINA_API_KEY_1..2 to .env' : `All ${jinaKeyPool.getTotalCount()} keys failed (${jinaKeyPool.getAvailableCount()} available)`,
  1236→    poolStatus: jinaKeyPool.getSummary(),
  1237→  }
  1238→}

  1304→// ==================== REACT LOOP HELPER ====================
  1305→
  1306→/**
  1307→ * Process a single LLM response — check for tool_calls and execute them.
  1308→ * Returns array of tool execution results for feeding back to LLM.
  1309→ *
  1310→ * This is the core of the ReAct loop:
  1311→ * 1. LLM returns response with tool_calls
  1312→ * 2. We execute each tool_call
  1313→ * 3. Results are fed back as tool messages
  1314→ * 4. LLM generates next response (may have more tool_calls or final text)
  1315→ */
  1316→export async function processToolCalls(
  1317→  toolCalls: Array<{ id: string; function: { name: string; arguments: string } }>,
  1318→): Promise<Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }>> {
  1319→  const results: Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }> = []
  1320→
  1321→  for (const tc of toolCalls) {
  1322→    let args: Record<string, unknown> = {}
  1323→    try {
  1324→      args = JSON.parse(tc.function.arguments)
  1325→    } catch {
  1326→      args = { _parseError: tc.function.arguments }
  1327→    }
  1328→
  1329→    console.log(`[ToolExecutor] Executing: ${tc.function.name}(${JSON.stringify(args).slice(0, 200)})`)
  1330→
  1331→    const result = await executeTool(tc.function.name, args)
  1332→
  1333→    results.push({
  1334→      toolCallId: tc.id,
  1335→      toolName: tc.function.name,
  1336→      result,
  1337→    })
  1338→
  1339→    console.log(`[ToolExecutor] Result: ${tc.function.name} → ${result.success ? 'OK' : 'FAIL'} (${result.duration}ms)`)
  1340→  }
  1341→
  1342→  return results
  1343→}
  1344→
  1345→/**
  1346→ * Format tool results into messages for the LLM.
  1347→ * Each tool result becomes a "tool" role message.
  1348→ * Note: Duration is excluded from LLM messages to save tokens — LLM doesn't need timing info.
  1349→ */
  1350→export function formatToolResultsAsMessages(
  1351→  results: Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }>,
  1352→): Array<{ role: 'tool'; tool_call_id: string; content: string }> {
  1353→  return results.map(r => ({
  1354→    role: 'tool' as const,
  1355→    tool_call_id: r.toolCallId,
  1356→    content: JSON.stringify({
  1357→      tool: r.toolName,
  1358→      success: r.result.success,
  1359→      result: r.result.result,
  1360→    }),
  1361→  }))
  1362→}
