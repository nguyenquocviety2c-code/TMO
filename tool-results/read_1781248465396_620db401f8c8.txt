     4→ * Central place to execute all tools that agents can call via LLM function calling.
     5→ * Supports ReAct loop: LLM returns tool_call → execute → feed result back → repeat.
     6→ *
     7→ * Tools available (per-agent permissions from agents.ts):

    14→ *   - jina:             Web page reader — via Jina API + z-ai-sdk fallback
    15→ *
    16→ * Architecture:
    17→ *   Workflow Engine → callLLMForAgent() with tool definitions
    18→ *   → LLM returns tool_call → executeTool() → feed result back
    19→ *   → Repeat until LLM outputs text (no more tool_calls)
    20→ *
    21→ * Phase 5 Enhancement:
    22→ *   - Tavily/Serper/Jina use direct API keys (from .env)
    23→ *   - z-ai-web-dev-sdk provides automatic fallback when API keys fail

   312→/**
   313→ * Execute a tool via Gateway HTTP /tools/invoke endpoint.
   314→ * Only works for HTTP-allowed tools (web_search, web_fetch, etc.).
   315→ */
   316→async function executeGatewayTool(name: string, args: Record<string, unknown>): Promise<unknown> {
   317→  if (!GATEWAY_HTTP_ALLOWED_TOOLS.has(name)) {
   318→    throw new Error(`Tool "${name}" requires Gateway session context. Try the Bridge instead.`)
   319→  }
   320→
   321→  const res = await fetch(`${GATEWAY_URL}/tools/invoke`, {
   322→    method: 'POST',
   323→    headers: { 'Content-Type': 'application/json' },
   324→    body: JSON.stringify({ name, args }),
   325→    signal: AbortSignal.timeout(30000),
   326→  })
   327→
   328→  if (!res.ok) {
   329→    const errorText = await res.text().catch(() => 'Unknown error')
   330→    throw new Error(`Gateway tool error ${res.status}: ${errorText.slice(0, 200)}`)
   331→  }
   332→
   333→  const data = await res.json()
   334→  if (!data.ok && data.error) {
   335→    throw new Error(`Tool error: ${data.error?.message || data.error || 'Unknown'}`)
   336→  }
   337→
   338→  return data.result ?? data
   339→}
   340→
   341→/**
   342→ * Execute a tool via the Gateway Bridge (port 18791).
   343→ * The Bridge connects to the Gateway via WebSocket with full session context,
   344→ * enabling access to ALL 36 Gateway tools including session-context ones
   345→ * (exec, read, write, edit, image_generate, etc.).
   346→ */
   347→async function executeBridgeTool(name: string, args: Record<string, unknown>): Promise<unknown> {
   348→  const res = await fetch(`${BRIDGE_URL}/tools/invoke`, {
   349→    method: 'POST',
   350→    headers: { 'Content-Type': 'application/json' },
   351→    body: JSON.stringify({ name, args }),
   352→    signal: AbortSignal.timeout(60000),
   353→  })
   354→
   355→  if (!res.ok) {
   356→    const errorText = await res.text().catch(() => 'Unknown error')
   357→    throw new Error(`Bridge tool error ${res.status}: ${errorText.slice(0, 200)}`)
   358→  }
   359→
   360→  const data = await res.json()
   361→  if (!data.ok && data.error) {
   362→    throw new Error(`Bridge tool error: ${typeof data.error === 'string' ? data.error : data.error?.message || 'Unknown'}`)
   363→  }
   364→
   365→  return data.result ?? data
   366→}

   368→/**
   369→ * Check if the Gateway Bridge is online and authenticated.
   370→ */
   371→export async function isBridgeOnline(): Promise<boolean> {
   372→  try {
   373→    const res = await fetch(`${BRIDGE_URL}/health`, { signal: AbortSignal.timeout(3000) })
   374→    if (res.ok) {
   375→      const data = await res.json()
   376→      return data?.bridge?.authenticated === true
   377→    }
   378→  } catch {}
   379→  return false
   380→}

   382→// ==================== Z-AI-SDK LAZY INITIALIZATION ====================
   383→//
   384→// ⚠️  IMPORTANT: z-ai-web-dev-sdk is an OPTIONAL dependency!
   385→//
   386→// This SDK ONLY works inside the z.ai sandbox environment where backend
   387→// services (web_search, page_reader) are available. If you clone this
   388→// project and run locally, the SDK will not initialize — this is expected
   389→// and NOT an error. Direct API keys (TAVILY_API_KEY, SERPER_API_KEY,
   390→// JINA_API_KEY) in .env will be used instead.
   391→//
   392→// Fallback chain: Direct API key → z-ai-sdk → error with helpful hint
   393→//
   394→
   395→let zaiInstance: any = null
   396→let zaiInitPromise: Promise<any> | null = null
   397→let zaiAvailable: boolean | null = null // null = not yet checked
   398→
   399→async function getZAI(): Promise<any> {
   400→  if (zaiInstance) return zaiInstance
   401→  if (zaiInitPromise) return zaiInitPromise
   402→
   403→  zaiInitPromise = (async () => {
   404→    try {
   405→      const ZAI = (await import('z-ai-web-dev-sdk')).default
   406→      const instance = await ZAI.create()
   407→      console.log('[ToolExecutor] ✅ z-ai-web-dev-sdk initialized — web_search + page_reader available as fallback')
   408→      zaiInstance = instance
   409→      zaiAvailable = true
   410→      return instance
   411→    } catch (err) {
   412→      zaiAvailable = false
   413→      console.warn(
   414→        '[ToolExecutor] z-ai-web-dev-sdk not available (this is normal for local deployment — direct API keys will be used).',
   415→        `Detail: ${err instanceof Error ? err.message : String(err)}`
   416→      )
   417→      return null
   418→    }
   419→  })()
   420→
   421→  return zaiInitPromise
   422→}

   629→/**
   630→ * Execute a tool call from LLM.
   631→ * Called in the ReAct loop when LLM returns a tool_call.
   632→ *
   633→ * @param toolName - Name of the tool to execute
   634→ * @param args - Arguments from LLM's tool_call
   635→ * @returns Execution result with success status
   636→ */
   637→export async function executeTool(
   638→  toolName: string,
   639→  args: Record<string, unknown>
   640→): Promise<ToolExecutionResult> {
   641→  const startTime = Date.now()
   642→
   643→  try {
   644→    let result: unknown
   645→
   646→    switch (toolName) {
   647→      case 'opencode':
   648→        result = await executeOpenCode(args)
   649→        break
   650→      case 'knowledge_search':
   651→        result = await executeKnowledgeSearch(args)
   652→        break
   653→      case 'knowledge_graph':
   654→        result = await executeKnowledgeGraph(args)
   655→        break
   656→      case 'knowledge_write':
   657→        result = await executeKnowledgeWrite(args)
   658→        break
   659→      case 'tavily':
   660→        result = await executeTavily(args)
   661→        break
   662→      case 'serper':
   663→        result = await executeSerper(args)
   664→        break
   665→      case 'jina':
   666→        result = await executeJina(args)
   667→        break
   668→      default: {
   669→        // V2: Route to Gateway HTTP, Bridge, or error
   670→        // 1. Try Gateway HTTP for HTTP-accessible tools
   671→        if (GATEWAY_HTTP_ALLOWED_TOOLS.has(toolName)) {
   672→          try {
   673→            const gwResult = await executeGatewayTool(toolName, args)
   674→            return { success: true, result: gwResult, duration: Date.now() - startTime }
   675→          } catch (gwErr) {
   676→            return {
   677→              success: false,
   678→              result: `Gateway tool "${toolName}" failed: ${gwErr instanceof Error ? gwErr.message : String(gwErr)}`,
   679→              duration: Date.now() - startTime,
   680→            }
   681→          }
   682→        }
   683→        // 2. Try Gateway Bridge for session-context tools (exec, read, write, image_generate, etc.)
   684→        if (GATEWAY_BRIDGE_TOOLS.has(toolName)) {
   685→          try {
   686→            const bridgeResult = await executeBridgeTool(toolName, args)
   687→            return { success: true, result: bridgeResult, duration: Date.now() - startTime }
   688→          } catch (bridgeErr) {
   689→            return {
   690→              success: false,
   691→              result: `Bridge tool "${toolName}" failed: ${bridgeErr instanceof Error ? bridgeErr.message : String(bridgeErr)}`,
   692→              duration: Date.now() - startTime,
   693→            }
   694→          }
   695→        }
   696→        return { success: false, result: `Unknown tool: ${toolName}. Available: local(7) + Gateway HTTP(${GATEWAY_HTTP_ALLOWED_TOOLS.size}) + Bridge(${GATEWAY_BRIDGE_TOOLS.size}) = ${7 + GATEWAY_HTTP_ALLOWED_TOOLS.size + GATEWAY_BRIDGE_TOOLS.size} tools total`, duration: Date.now() - startTime }
   697→      }
   698→    }
   699→
   700→    return {
   701→      success: true,
   702→      result,
   703→      duration: Date.now() - startTime,
   704→    }
   705→  } catch (err) {
   706→    const errorMsg = err instanceof Error ? err.message : String(err)
   707→    console.error(`[ToolExecutor] Error executing ${toolName}:`, errorMsg)
   708→    return {
   709→      success: false,
   710→      result: `Tool error: ${errorMsg}`,
   711→      duration: Date.now() - startTime,
   712→    }
   713→  }
   714→}

   718→/**
   719→ * OpenCode — Code operations (read, write, bash, mkdir)
   720→ *
   721→ * Uses the OpenCode API server (configured via OPENCODE_SERVER_URL env var) if available,
   722→ * otherwise falls back to direct file system operations via the project API.
   723→ */
   724→async function executeOpenCode(args: Record<string, unknown>): Promise<unknown> {
   725→  const action = args.action as string
   726→  const path = args.path as string | undefined
   727→  const content = args.content as string | undefined
   728→  const command = args.command as string | undefined
   729→
   730→  // Try OpenCode server first
   731→  try {
   732→    const opencodeUrl = process.env.OPENCODE_SERVER_URL || 'http://localhost:18790'
   733→    const res = await fetch(`${opencodeUrl}/api/execute`, {
   734→      method: 'POST',
   735→      headers: { 'Content-Type': 'application/json' },
   736→      body: JSON.stringify({ action, path, content, command }),
   737→      signal: AbortSignal.timeout(30000),
   738→    })
   739→
   740→    if (res.ok) {
   741→      const data = await res.json()
   742→      return data
   743→    }
   744→  } catch {
   745→    // OpenCode server not available, fall through to direct implementation
   746→  }
   747→
   748→  // Fallback: Use the project's own API for file operations
   749→  switch (action) {
   750→    case 'read':
   751→      return await executeFileRead(path || '')
   752→    case 'write':
   753→      return await executeFileWrite(path || '', content || '')
   754→    case 'bash':
   755→      return await executeBash(command || '')
   756→    case 'mkdir':
   757→      return { success: true, message: `mkdir not needed — write will create parent dirs` }
   758→    default:
   759→      return { success: false, error: `Unknown opencode action: ${action}` }
   760→  }
   761→}

   763→/**
   764→ * Read a file from the project directory.
   765→ * Security: Only allows reading files within the project root.
   766→ * Uses path.resolve + startsWith check to prevent path traversal.
   767→ */
   768→async function executeFileRead(filePath: string): Promise<{ success: boolean; content?: string; error?: string }> {
   769→  if (!filePath) return { success: false, error: 'Path is required' }
   770→
   771→  try {
   772→    const pathModule = await import('path')
   773→    const projectRoot = process.cwd()
   774→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   775→
   776→    // Security: Prevent path traversal — resolved path must be within project root
   777→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   778→      return { success: false, error: `Access denied: path escapes project root` }
   779→    }
   780→
   781→    const fs = await import('fs/promises')
   782→    const content = await fs.readFile(resolvedPath, 'utf-8')
   783→
   784→    // Limit output to prevent overwhelming LLM context
   785→    const maxChars = 50000
   786→    const truncated = content.length > maxChars
   787→    const output = truncated ? content.slice(0, maxChars) + '\n\n... [TRUNCATED — file too large]' : content
   788→
   789→    return { success: true, content: output }
   790→  } catch (err) {
   791→    return {
   792→      success: false,
   793→      error: `Failed to read ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   794→    }
   795→  }
   796→}

   798→/**
   799→ * Write a file to the project directory.
   800→ * Security: Only allows writing within the project root. Uses path.resolve to prevent traversal.
   801→ * Auto-creates parent directories.
   802→ */
   803→async function executeFileWrite(filePath: string, content: string): Promise<{ success: boolean; message: string; error?: string }> {
   804→  if (!filePath) return { success: false, message: '', error: 'Path is required' }
   805→  if (!content) return { success: false, message: '', error: 'Content is required' }
   806→
   807→  try {
   808→    const pathModule = await import('path')
   809→    const projectRoot = process.cwd()
   810→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   811→
   812→    // Security: Prevent path traversal — resolved path must be within project root
   813→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   814→      return { success: false, message: '', error: 'Access denied: path escapes project root' }
   815→    }
   816→
   817→    const fs = await import('fs/promises')
   818→
   819→    // Auto-create parent directories
   820→    const dir = pathModule.dirname(resolvedPath)
   821→    await fs.mkdir(dir, { recursive: true })
   822→
   823→    // Write file
   824→    await fs.writeFile(resolvedPath, content, 'utf-8')
   825→
   826→    return { success: true, message: `File written: ${filePath} (${content.length} chars)` }
   827→  } catch (err) {
   828→    return {
   829→      success: false,
   830→      message: '',
   831→      error: `Failed to write ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   832→    }
   833→  }
   834→}

   836→/**
   837→ * Execute a bash command safely.
   838→ * Security: Only allows pre-approved commands.
   839→ */
   840→async function executeBash(command: string): Promise<{ success: boolean; output?: string; error?: string }> {
   841→  if (!command) return { success: false, error: 'Command is required' }
   842→
   843→  // Security: Whitelist of allowed commands
   844→  // NOTE: 'node' is excluded because `node -e "..."` can execute arbitrary code.
   845→  //       'cat'/'head'/'tail'/'grep' are excluded — use opencode read instead.
   846→  //       'npx'/'bunx' are excluded — can download+execute arbitrary npm packages.
   847→  const allowedPrefixes = [
   848→    'npm run',
   849→    'npm test',
   850→    'npm lint',
   851→    'bun run',
   852→    'bun test',
   853→    'bun lint',
   854→    'tsc',
   855→    'eslint',
   856→    'prettier',
   857→    'echo',
   858→    'ls',
   859→    'wc',
   860→    'rg',       // ripgrep — search only
   861→    'which',
   862→    'pwd',
   863→    'prisma',
   864→    'git status',
   865→    'git diff',
   866→    'git log',
   867→  ]
   868→
   869→  // Block dangerous flags that could allow code execution or file access
   870→  const dangerousFlags = ['-e ', '--eval ', '-r ', '--require ', '-p ', '--print ']
   871→  const normalizedCmd = command.trim()
   872→  const isAllowed = allowedPrefixes.some(prefix => normalizedCmd.startsWith(prefix))
   873→  const usesDangerousFlag = dangerousFlags.some(flag => normalizedCmd.includes(flag))
   874→
   875→  // Security: Block shell metacharacters that enable command chaining/injection
   876→  // These allow bypassing the whitelist by chaining additional commands:
   877→  //   "npm run build; rm -rf /"   →  ;  chains commands
   878→  //   "git status && cat /etc/p"  →  && chains on success
   879→  //   "echo hi | bash"            →  |  pipes output
   880→  //   "tsc || rm -rf /"           →  || chains on failure
   881→  //   "echo $(cat /etc/passwd)"   →  $() command substitution
   882→  //   "echo `cat /etc/passwd`"    →  backtick command substitution
   883→  //   "echo hi > /etc/passwd"     →  >  redirect write
   884→  //   "echo hi >> /etc/passwd"    →  >> redirect append
   885→  const shellMetacharacters = /(;|&&|\|\||\||`|\$\(|\$\{|>|>>)/
   886→  const usesShellMetacharacters = shellMetacharacters.test(normalizedCmd)
   887→
   888→  if (!isAllowed || usesDangerousFlag || usesShellMetacharacters) {
   889→    return {
   890→      success: false,
   891→      error: `Command not allowed: "${normalizedCmd.slice(0, 100)}". ${!isAllowed ? 'Prefix not in whitelist.' : ''} ${usesDangerousFlag ? 'Dangerous flag detected.' : ''} ${usesShellMetacharacters ? 'Shell metacharacter (;, &&, ||, |, `, $(), >, >>) detected — command chaining is forbidden.' : ''}`,
   892→    }
   893→  }
   894→
   895→  try {
   896→    const { exec } = await import('child_process')
   897→    const { promisify } = await import('util')
   898→    const execAsync = promisify(exec)
   899→
   900→    const { stdout, stderr } = await execAsync(normalizedCmd, {
   901→      cwd: process.cwd(),
   902→      timeout: 30000,
   903→      maxBuffer: 1024 * 1024, // 1MB
   904→    })
   905→
   906→    const output = [
   907→      stdout ? `STDOUT:\n${stdout}` : '',
   908→      stderr ? `STDERR:\n${stderr}` : '',
   909→    ].filter(Boolean).join('\n')
   910→
   911→    return { success: true, output: output || '(no output)' }
   912→  } catch (err) {
   913→    const error = err as { stdout?: string; stderr?: string; message?: string }
   914→    const output = [
   915→      error.stdout ? `STDOUT:\n${error.stdout}` : '',
   916→      error.stderr ? `STDERR:\n${error.stderr}` : '',
   917→    ].filter(Boolean).join('\n')
   918→
   919→    return {
   920→      success: false,
   921→      output: output || undefined,
   922→      error: `Command failed: ${error.message}`,
   923→    }
   924→  }
   925→}

  1032→// ==================== KEY RETRY HELPER ====================
  1033→
  1034→/**
  1035→ * Retry a service call with the next available key from the pool.
  1036→ * Called when the first key fails — tries remaining keys before falling back to SDK.
  1037→ *
  1038→ * @param service - 'tavily' | 'serper' | 'jina'
  1039→ * @param param - The query (for search) or URL (for reader)
  1040→ * @returns Result if a key succeeds, null if all keys fail
  1041→ */
  1042→async function retryWithNextKey(
  1043→  service: 'tavily' | 'serper' | 'jina',
  1044→  param: string
  1045→): Promise<unknown | null> {
  1046→  const pool = service === 'tavily' ? tavilyKeyPool : service === 'serper' ? serperKeyPool : jinaKeyPool
  1047→  const maxRetries = pool.getTotalCount() - 1 // Already tried 1 key
  1048→
  1049→  for (let i = 0; i < maxRetries; i++) {
  1050→    const nextKey = pool.getNextKey()
  1051→    if (!nextKey || nextKey.isLastResort) break // No more healthy keys
  1052→
  1053→    console.log(`[ToolExecutor] ${service} retrying with key ${nextKey.index + 1}...`)
  1054→
  1055→    try {
  1056→      let res: Response
  1057→
  1058→      if (service === 'tavily') {
  1059→        res = await fetch('https://api.tavily.com/search', {
  1060→          method: 'POST',
  1061→          headers: { 'Content-Type': 'application/json' },
  1062→          body: JSON.stringify({
  1063→            api_key: nextKey.key,
  1064→            query: param,
  1065→            search_depth: 'advanced',
  1066→            include_answer: true,
  1067→            max_results: 5,
  1068→          }),
  1069→          signal: AbortSignal.timeout(30000),
  1070→        })
  1071→      } else if (service === 'serper') {
  1072→        res = await fetch('https://google.serper.dev/search', {
  1073→          method: 'POST',
  1074→          headers: {
  1075→            'X-API-KEY': nextKey.key,
  1076→            'Content-Type': 'application/json',
  1077→          },
  1078→          body: JSON.stringify({ q: param, num: 5 }),
  1079→          signal: AbortSignal.timeout(15000),
  1080→        })
  1081→      } else {
  1082→        // Jina
  1083→        const headers: Record<string, string> = { 'Accept': 'text/plain' }
  1084→        if (nextKey.key) headers['Authorization'] = `Bearer ${nextKey.key}`
  1085→        res = await fetch(`https://r.jina.ai/${param}`, {
  1086→          method: 'GET',
  1087→          headers,
  1088→          signal: AbortSignal.timeout(30000),
  1089→        })
  1090→      }
  1091→
  1092→      if (res.ok) {
  1093→        pool.reportResult(nextKey.index, true, res.status)
  1094→        console.log(`[ToolExecutor] ${service} key ${nextKey.index + 1} succeeded on retry`)
  1095→
  1096→        if (service === 'tavily') {
  1097→          const data = await res.json()
  1098→          return {
  1099→            answer: data.answer,
  1100→            results: (data.results || []).map((r: Record<string, unknown>) => ({
  1101→              title: r.title, url: r.url, content: r.content, score: r.score,
  1102→            })),
  1103→            source: `tavily-key${nextKey.index + 1}`,
  1104→          }
  1105→        } else if (service === 'serper') {
  1106→          const data = await res.json()
  1107→          return {
  1108→            organic: (data.organic || []).map((r: Record<string, unknown>) => ({
  1109→              title: r.title, link: r.link, snippet: r.snippet,
  1110→            })),
  1111→            knowledgeGraph: data.knowledgeGraph || null,
  1112→            source: `serper-key${nextKey.index + 1}`,
  1113→          }
  1114→        } else {
  1115→          // Jina
  1116→          const text = await res.text()
  1117→          const maxChars = 30000
  1118→          const truncated = text.length > maxChars
  1119→          const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
  1120→          return {
  1121→            url: param, content, length: text.length, truncated,
  1122→            source: `jina-key${nextKey.index + 1}`,
  1123→          }
  1124→        }
  1125→      }
  1126→
  1127→      // This key also failed — report and try next
  1128→      const errorBody = await res.text().catch(() => '')
  1129→      pool.reportResult(nextKey.index, false, res.status, `HTTP ${res.status}: ${errorBody.slice(0, 200)}`)

  1505→/**
  1506→ * Process a single LLM response — check for tool_calls and execute them.
  1507→ * Returns array of tool execution results for feeding back to LLM.
  1508→ *
  1509→ * This is the core of the ReAct loop:
  1510→ * 1. LLM returns response with tool_calls
  1511→ * 2. We execute each tool_call
  1512→ * 3. Results are fed back as tool messages
  1513→ * 4. LLM generates next response (may have more tool_calls or final text)
  1514→ */
  1515→export async function processToolCalls(
  1516→  toolCalls: Array<{ id: string; function: { name: string; arguments: string } }>,
  1517→): Promise<Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }>> {
  1518→  const results: Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }> = []
  1519→
  1520→  for (const tc of toolCalls) {
  1521→    let args: Record<string, unknown> = {}
  1522→    try {
  1523→      args = JSON.parse(tc.function.arguments)
  1524→    } catch {
  1525→      args = { _parseError: tc.function.arguments }
  1526→    }
  1527→
  1528→    console.log(`[ToolExecutor] Executing: ${tc.function.name}(${JSON.stringify(args).slice(0, 200)})`)
  1529→
  1530→    const result = await executeTool(tc.function.name, args)
  1531→
  1532→    results.push({
  1533→      toolCallId: tc.id,
  1534→      toolName: tc.function.name,
  1535→      result,
  1536→    })
  1537→
  1538→    console.log(`[ToolExecutor] Result: ${tc.function.name} → ${result.success ? 'OK' : 'FAIL'} (${result.duration}ms)`)
  1539→  }
  1540→
  1541→  return results
  1542→}
