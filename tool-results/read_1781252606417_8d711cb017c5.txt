   502→// EMBEDDING CACHE (Phase 6)
   503→// SQLite-backed L2 cache — survives server restarts
   504→// L1 = in-memory Map (1h TTL, fast), L2 = SQLite (7d TTL, persistent)
   505→// ============================================

   507→/// Embedding cache — stores previously computed embeddings to avoid redundant API calls
   508→/// Survives server restarts (unlike in-memory cache)
   509→/// IMPORTANT: Same text gets DIFFERENT embeddings for "query" vs "passage" (NVIDIA input_type)
   510→/// So the unique key is (hash, inputType) — NOT just hash
   511→model EmbeddingCache {
   512→  id        String   @id @default(cuid())
   513→  hash     String   /// SHA-256 hash of the input text
   514→  inputType String  @default("passage") /// "query" | "passage" — different embeddings for same text
   515→  model    String   /// Model used to generate the embedding (e.g. "nvidia/llama-nemotron-embed-1b-v2")
   516→  vector   String   /// JSON array of the embedding vector (1536 floats)
   517→  hitCount Int      @default(1) /// Number of cache hits (for LRU eviction)
   518→  createdAt DateTime @default(now())
   519→  lastHitAt DateTime @default(now()) /// Last time this cache entry was used
   520→  expiresAt DateTime /// When this cache entry expires
   521→
   522→  @@unique([hash, inputType]) /// Composite key — same text has separate query & passage embeddings
   523→  @@index([expiresAt])
   524→  @@index([lastHitAt])
   525→}

   635→/// Agent episodic memories — extracted from conversations, stored as vectors in Qdrant
   636→/// Each memory is a discrete piece of knowledge the agent learned from interaction
   637→model AgentMemory {
   638→  id             String   @id @default(cuid())
   639→  agentId        String   /// AgentProfile.id
   640→  agentName      String   /// Agent name (denormalized)
   641→  sessionId      String?  /// Source session
   642→  category       String   /// insight | fact | preference | correction | procedure | user_info
   643→  content        String   /// The actual memory content (text)
   644→  context        String?  /// Context: what triggered this memory (query, situation, etc.)
   645→  importance     Float    @default(0.5) /// 0.0-1.0 — how important this memory is
   646→  accessCount    Int      @default(0) /// How many times this memory was recalled
   647→  lastAccessedAt DateTime? /// Last time this memory was used in a response
   648→  qdrantPointId  String?  /// Qdrant point ID for vector search
   649→  embeddingModel String?  /// Embedding model used (e.g. nvidia/llama-nemotron-embed-1b-v2)
   650→  source         String   @default("auto") /// auto | manual | feedback | correction
   651→  tags           String?  /// JSON array of tags for categorization
   652→  isActive       Boolean  @default(true) /// Soft delete — inactive memories are not recalled
   653→  expiresAt      DateTime? /// Optional expiration for time-sensitive memories
   654→  createdAt      DateTime @default(now())
   655→  updatedAt      DateTime @updatedAt
   656→
   657→  @@index([agentId])
   658→  @@index([category])
   659→  @@index([importance])
   660→  @@index([accessCount])
   661→  @@index([isActive])
   662→  @@index([createdAt])
   663→  @@index([lastAccessedAt])
   664→}
