   271→   271→export interface ToolExecutionResult {
   272→   272→  success: boolean
   273→   273→  result: unknown

   276→   276→
   277→   277→/**
   278→   278→ * Execute a tool call from LLM.
   279→   279→ * Called in the ReAct loop when LLM returns a tool_call.
   280→   280→ *
   281→   281→ * @param toolName - Name of the tool to execute
   282→   282→ * @param args - Arguments from LLM's tool_call
   283→   283→ * @returns Execution result with success status
   284→   284→ */
   285→   285→export async function executeTool(
   286→   286→  toolName: string,
   287→   287→  args: Record<string, unknown>
   288→   288→): Promise<ToolExecutionResult> {
   289→   289→  const startTime = Date.now()
   290→   290→
   291→   291→  try {
   292→   292→    let result: unknown
   293→   293→
   294→   294→    switch (toolName) {
   295→   295→      case 'opencode':
   296→   296→        result = await executeOpenCode(args)
   297→   297→        break
   298→   298→      case 'knowledge_search':
   299→   299→        result = await executeKnowledgeSearch(args)
   300→   300→        break
   301→   301→      case 'knowledge_graph':
   302→   302→        result = await executeKnowledgeGraph(args)
   303→   303→        break
   304→   304→      case 'knowledge_write':
   305→   305→        result = await executeKnowledgeWrite(args)
   306→   306→        break
   307→   307→      case 'tavily':
   308→   308→        result = await executeTavily(args)
   309→   309→        break
   310→   310→      case 'serper':
   311→   311→        result = await executeSerper(args)
   312→   312→        break
   313→   313→      case 'jina':
   314→   314→        result = await executeJina(args)
   315→   315→        break
   316→   316→      default:
   317→   317→        return {
   318→   318→          success: false,
   319→   319→          result: `Unknown tool: ${toolName}`,
   320→   320→          duration: Date.now() - startTime,
   321→   321→        }
   322→   322→    }
   323→   323→
   324→   324→    return {
   325→   325→      success: true,
   326→   326→      result,
   327→   327→      duration: Date.now() - startTime,
   328→   328→    }
   329→   329→  } catch (err) {
   330→   330→    const errorMsg = err instanceof Error ? err.message : String(err)
   331→   331→    console.error(`[ToolExecutor] Error executing ${toolName}:`, errorMsg)
   332→   332→    return {
   333→   333→      success: false,
   334→   334→      result: `Tool error: ${errorMsg}`,
   335→   335→      duration: Date.now() - startTime,
   336→   336→    }
   337→   337→  }
   338→   338→}

   340→   340→// ==================== INDIVIDUAL TOOL IMPLEMENTATIONS ====================

   342→   342→/**
   343→   343→ * OpenCode — Code operations (read, write, bash, mkdir)
   344→   344→ *
   345→   345→ * Uses the OpenCode API server (configured via OPENCODE_SERVER_URL env var) if available,
   346→   346→ * otherwise falls back to direct file system operations via the project API.
   347→   347→ */
   348→   348→async function executeOpenCode(args: Record<string, unknown>): Promise<unknown> {
   349→   349→  const action = args.action as string
   350→   350→  const path = args.path as string | undefined
   351→   351→  const content = args.content as string | undefined
   352→   352→  const command = args.command as string | undefined
   353→   353→
   354→   354→  // Try OpenCode server first
   355→   355→  try {
   356→   356→    const opencodeUrl = process.env.OPENCODE_SERVER_URL || 'http://localhost:18790'
   357→   357→    const res = await fetch(`${opencodeUrl}/api/execute`, {
   358→   358→      method: 'POST',
   359→   359→      headers: { 'Content-Type': 'application/json' },
   360→   360→      body: JSON.stringify({ action, path, content, command }),
   361→   361→      signal: AbortSignal.timeout(30000),
   362→   362→    })
   363→   363→
   364→   364→    if (res.ok) {
   365→   365→      const data = await res.json()
   366→   366→      return data
   367→   367→    }
   368→   368→  } catch {
   369→   369→    // OpenCode server not available, fall through to direct implementation
   370→   370→  }
   371→   371→
   372→   372→  // Fallback: Use the project's own API for file operations
   373→   373→  switch (action) {
   374→   374→    case 'read':
   375→   375→      return await executeFileRead(path || '')
   376→   376→    case 'write':
   377→   377→      return await executeFileWrite(path || '', content || '')
   378→   378→    case 'bash':
   379→   379→      return await executeBash(command || '')
   380→   380→    case 'mkdir':
   381→   381→      return { success: true, message: `mkdir not needed — write will create parent dirs` }
   382→   382→    default:
   383→   383→      return { success: false, error: `Unknown opencode action: ${action}` }
   384→   384→  }
   385→   385→}
   386→   386→
   387→   387→/**
   388→   388→ * Read a file from the project directory.
   389→   389→ * Security: Only allows reading files within the project root.
   390→   390→ * Uses path.resolve + startsWith check to prevent path traversal.
   391→   391→ */
   392→   392→async function executeFileRead(filePath: string): Promise<{ success: boolean; content?: string; error?: string }> {
   393→   393→  if (!filePath) return { success: false, error: 'Path is required' }
   394→   394→
   395→   395→  try {
   396→   396→    const pathModule = await import('path')
   397→   397→    const projectRoot = process.cwd()
   398→   398→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   399→   399→
   400→   400→    // Security: Prevent path traversal — resolved path must be within project root
   401→   401→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   402→   402→      return { success: false, error: `Access denied: path escapes project root` }
   403→   403→    }
   404→   404→
   405→   405→    const fs = await import('fs/promises')
   406→   406→    const content = await fs.readFile(resolvedPath, 'utf-8')
   407→   407→
   408→   408→    // Limit output to prevent overwhelming LLM context
   409→   409→    const maxChars = 50000
   410→   410→    const truncated = content.length > maxChars
   411→   411→    const output = truncated ? content.slice(0, maxChars) + '\n\n... [TRUNCATED — file too large]' : content
   412→   412→
   413→   413→    return { success: true, content: output }
   414→   414→  } catch (err) {
   415→   415→    return {
   416→   416→      success: false,
   417→   417→      error: `Failed to read ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   418→   418→    }
   419→   419→  }
   420→   420→}
   421→   421→
   422→   422→/**
   423→   423→ * Write a file to the project directory.
   424→   424→ * Security: Only allows writing within the project root. Uses path.resolve to prevent traversal.
   425→   425→ * Auto-creates parent directories.
   426→   426→ */
   427→   427→async function executeFileWrite(filePath: string, content: string): Promise<{ success: boolean; message: string; error?: string }> {
   428→   428→  if (!filePath) return { success: false, message: '', error: 'Path is required' }
   429→   429→  if (!content) return { success: false, message: '', error: 'Content is required' }
   430→   430→
   431→   431→  try {
   432→   432→    const pathModule = await import('path')
   433→   433→    const projectRoot = process.cwd()
   434→   434→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   435→   435→
   436→   436→    // Security: Prevent path traversal — resolved path must be within project root
   437→   437→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   438→   438→      return { success: false, message: '', error: 'Access denied: path escapes project root' }
   439→   439→    }
   440→   440→
   441→   441→    const fs = await import('fs/promises')
   442→   442→
   443→   443→    // Auto-create parent directories
   444→   444→    const dir = pathModule.dirname(resolvedPath)
   445→   445→    await fs.mkdir(dir, { recursive: true })
   446→   446→
   447→   447→    // Write file
   448→   448→    await fs.writeFile(resolvedPath, content, 'utf-8')
   449→   449→
   450→   450→    return { success: true, message: `File written: ${filePath} (${content.length} chars)` }
   451→   451→  } catch (err) {
   452→   452→    return {
   453→   453→      success: false,
   454→   454→      message: '',
   455→   455→      error: `Failed to write ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   456→   456→    }
   457→   457→  }
   458→   458→}

   460→   460→/**
   461→   461→ * Execute a bash command safely.
   462→   462→ * Security: Only allows pre-approved commands.
   463→   463→ */
   464→   464→async function executeBash(command: string): Promise<{ success: boolean; output?: string; error?: string }> {
   465→   465→  if (!command) return { success: false, error: 'Command is required' }
   466→   466→
   467→   467→  // Security: Whitelist of allowed commands
   468→   468→  // NOTE: 'node' is excluded because `node -e "..."` can execute arbitrary code.
   469→   469→  //       'cat'/'head'/'tail'/'grep' are excluded — use opencode read instead.
   470→   470→  //       'npx'/'bunx' are excluded — can download+execute arbitrary npm packages.
   471→   471→  const allowedPrefixes = [
   472→   472→    'npm run',
   473→   473→    'npm test',
   474→   474→    'npm lint',
   475→   475→    'bun run',
   476→   476→    'bun test',
   477→   477→    'bun lint',
   478→   478→    'tsc',
   479→   479→    'eslint',
   480→   480→    'prettier',
   481→   481→    'echo',
   482→   482→    'ls',
   483→   483→    'wc',
   484→   484→    'rg',       // ripgrep — search only
   485→   485→    'which',
   486→   486→    'pwd',
   487→   487→    'prisma',
   488→   488→    'git status',
   489→   489→    'git diff',
   490→   490→    'git log',
   491→   491→  ]
   492→   492→
   493→   493→  // Block dangerous flags that could allow code execution or file access
   494→   494→  const dangerousFlags = ['-e ', '--eval ', '-r ', '--require ', '-p ', '--print ']
   495→   495→  const normalizedCmd = command.trim()
   496→   496→  const isAllowed = allowedPrefixes.some(prefix => normalizedCmd.startsWith(prefix))
   497→   497→  const usesDangerousFlag = dangerousFlags.some(flag => normalizedCmd.includes(flag))
   498→   498→
   499→   499→  // Security: Block shell metacharacters that enable command chaining/injection
   500→   500→  // These allow bypassing the whitelist by chaining additional commands:
   501→   501→  //   "npm run build; rm -rf /"   →  ;  chains commands
   502→   502→  //   "git status && cat /etc/p"  →  && chains on success
   503→   503→  //   "echo hi | bash"            →  |  pipes output
   504→   504→  //   "tsc || rm -rf /"           →  || chains on failure
   505→   505→  //   "echo $(cat /etc/passwd)"   →  $() command substitution
   506→   506→  //   "echo `cat /etc/passwd`"    →  backtick command substitution
   507→   507→  //   "echo hi > /etc/passwd"     →  >  redirect write
   508→   508→  //   "echo hi >> /etc/passwd"    →  >> redirect append
   509→   509→  const shellMetacharacters = /(;|&&|\|\||\||`|\$\(|\$\{|>|>>)/
   510→   510→  const usesShellMetacharacters = shellMetacharacters.test(normalizedCmd)
   511→   511→
   512→   512→  if (!isAllowed || usesDangerousFlag || usesShellMetacharacters) {
   513→   513→    return {
   514→   514→      success: false,
   515→   515→      error: `Command not allowed: "${normalizedCmd.slice(0, 100)}". ${!isAllowed ? 'Prefix not in whitelist.' : ''} ${usesDangerousFlag ? 'Dangerous flag detected.' : ''} ${usesShellMetacharacters ? 'Shell metacharacter (;, &&, ||, |, `, $(), >, >>) detected — command chaining is forbidden.' : ''}`,
   516→   516→    }
   517→   517→  }
   518→   518→
   519→   519→  try {
   520→   520→    const { exec } = await import('child_process')
   521→   521→    const { promisify } = await import('util')
   522→   522→    const execAsync = promisify(exec)
   523→   523→
   524→   524→    const { stdout, stderr } = await execAsync(normalizedCmd, {
   525→   525→      cwd: process.cwd(),
   526→   526→      timeout: 30000,
   527→   527→      maxBuffer: 1024 * 1024, // 1MB
   528→   528→    })
   529→   529→
   530→   530→    const output = [
   531→   531→      stdout ? `STDOUT:\n${stdout}` : '',
   532→   532→      stderr ? `STDERR:\n${stderr}` : '',
   533→   533→    ].filter(Boolean).join('\n')
   534→   534→
   535→   535→    return { success: true, output: output || '(no output)' }
   536→   536→  } catch (err) {
   537→   537→    const error = err as { stdout?: string; stderr?: string; message?: string }
   538→   538→    const output = [
   539→   539→      error.stdout ? `STDOUT:\n${error.stdout}` : '',
   540→   540→      error.stderr ? `STDERR:\n${error.stderr}` : '',
   541→   541→    ].filter(Boolean).join('\n')
   542→   542→
   543→   543→    return {
   544→   544→      success: false,
   545→   545→      output: output || undefined,
   546→   546→      error: `Command failed: ${error.message}`,
   547→   547→    }
   548→   548→  }
   549→   549→}

   658→   658→/**
   659→   659→ * Retry a service call with the next available key from the pool.
   660→   660→ * Called when the first key fails — tries remaining keys before falling back to SDK.
   661→   661→ *
   662→   662→ * @param service - 'tavily' | 'serper' | 'jina'
   663→   663→ * @param param - The query (for search) or URL (for reader)
   664→   664→ * @returns Result if a key succeeds, null if all keys fail
   665→   665→ */
   666→   666→async function retryWithNextKey(
   667→   667→  service: 'tavily' | 'serper' | 'jina',
   668→   668→  param: string
   669→   669→): Promise<unknown | null> {
   670→   670→  const pool = service === 'tavily' ? tavilyKeyPool : service === 'serper' ? serperKeyPool : jinaKeyPool

   672→   672→

   674→   674→    const nextKey = pool.getNextKey()
   675→   675→    if (!nextKey || nextKey.isLastResort) break // No more healthy keys
   676→   676→
   677→   677→    console.log(`[ToolExecutor] ${service} retrying with key ${nextKey.index + 1}...`)
   678→   678→
   679→   679→    try {
   680→   680→      let res: Response
   681→   681→
   682→   682→      if (service === 'tavily') {
   683→   683→        res = await fetch('https://api.tavily.com/search', {
   684→   684→          method: 'POST',
   685→   685→          headers: { 'Content-Type': 'application/json' },
   686→   686→          body: JSON.stringify({
   687→   687→            api_key: nextKey.key,
   688→   688→            query: param,
   689→   689→            search_depth: 'advanced',
   690→   690→            include_answer: true,
   691→   691→            max_results: 5,
   692→   692→          }),
   693→   693→          signal: AbortSignal.timeout(30000),
   694→   694→        })
   695→   695→      } else if (service === 'serper') {
   696→   696→        res = await fetch('https://google.serper.dev/search', {
   697→   697→          method: 'POST',
   698→   698→          headers: {
   699→   699→            'X-API-KEY': nextKey.key,
   700→   700→            'Content-Type': 'application/json',
   701→   701→          },
   702→   702→          body: JSON.stringify({ q: param, num: 5 }),
   703→   703→          signal: AbortSignal.timeout(15000),
   704→   704→        })
   705→   705→      } else {
   706→   706→        // Jina
   707→   707→        const headers: Record<string, string> = { 'Accept': 'text/plain' }
   708→   708→        if (nextKey.key) headers['Authorization'] = `Bearer ${nextKey.key}`
   709→   709→        res = await fetch(`https://r.jina.ai/${param}`, {
   710→   710→          method: 'GET',
   711→   711→          headers,
   712→   712→          signal: AbortSignal.timeout(30000),
   713→   713→        })
   714→   714→      }
   715→   715→
   716→   716→      if (res.ok) {
   717→   717→        pool.reportResult(nextKey.index, true, res.status)
   718→   718→        console.log(`[ToolExecutor] ${service} key ${nextKey.index + 1} succeeded on retry`)
   719→   719→
   720→   720→        if (service === 'tavily') {
   721→   721→          const data = await res.json()
   722→   722→          return {
   723→   723→            answer: data.answer,
   724→   724→            results: (data.results || []).map((r: Record<string, unknown>) => ({
   725→   725→              title: r.title, url: r.url, content: r.content, score: r.score,
   726→   726→            })),
   727→   727→            source: `tavily-key${nextKey.index + 1}`,
   728→   728→          }
   729→   729→        } else if (service === 'serper') {
   730→   730→          const data = await res.json()
   731→   731→          return {
   732→   732→            organic: (data.organic || []).map((r: Record<string, unknown>) => ({
   733→   733→              title: r.title, link: r.link, snippet: r.snippet,
   734→   734→            })),
   735→   735→            knowledgeGraph: data.knowledgeGraph || null,
   736→   736→            source: `serper-key${nextKey.index + 1}`,
   737→   737→          }
   738→   738→        } else {
   739→   739→          // Jina
   740→   740→          const text = await res.text()
   741→   741→          const maxChars = 30000
   742→   742→          const truncated = text.length > maxChars
   743→   743→          const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
   744→   744→          return {
   745→   745→            url: param, content, length: text.length, truncated,
   746→   746→            source: `jina-key${nextKey.index + 1}`,
   747→   747→          }
   748→   748→        }
   749→   749→      }
   750→   750→
   751→   751→      // This key also failed — report and try next
   752→   752→      const errorBody = await res.text().catch(() => '')
   753→   753→      pool.reportResult(nextKey.index, false, res.status, `HTTP ${res.status}: ${errorBody.slice(0, 200)}`)
   754→   754→      console.warn(`[ToolExecutor] ${service} key ${nextKey.index + 1} also failed: ${res.status}`)
   755→   755→    } catch (err) {
   756→   756→      const errMsg = err instanceof Error ? err.message : String(err)
   757→   757→      pool.reportResult(nextKey.index, false, undefined, errMsg)
   758→   758→      console.warn(`[ToolExecutor] ${service} key ${nextKey.index + 1} exception: ${errMsg}`)
   759→   759→    }
   760→   760→  }
   761→   761→
   762→   762→  return null // All keys failed
   763→   763→}

   926→   926→          organic: results.map((r: any) => ({
   927→   927→            title: r.name || r.title,
   928→   928→            link: r.url,
   929→   929→            snippet: r.snippet || '',
   930→   930→          })),
   931→   931→          knowledgeGraph: null,
   932→   932→          source: 'z-ai-sdk-web_search',
   933→   933→        }
   934→   934→      }
   935→   935→    }
   936→   936→  } catch (err) {
   937→   937→    console.warn(`[ToolExecutor] z-ai-sdk web_search fallback failed: ${err instanceof Error ? err.message : String(err)}`)
   938→   938→  }
   939→   939→
   940→   940→  return {
   941→   941→    error: 'Serper search failed and no fallback available.',
   942→   942→    hint: !serperKeyPool.hasKeys() ? 'Add SERPER_API_KEY_1..4 to .env' : `All ${serperKeyPool.getTotalCount()} keys failed (${serperKeyPool.getAvailableCount()} available)`,
   943→   943→    poolStatus: serperKeyPool.getSummary(),
   944→   944→  }

   946→   946→
   947→   947→/**
   948→   948→ * Jina — Web page reader
   949→   949→ * Uses ServiceKeyPool for multi-key rotation with automatic fallback.
   950→   950→ * Primary: Direct Jina API with rotating keys from pool
   951→   951→ * Fallback: z-ai-web-dev-sdk page_reader function (Clawhub Market skill)
   952→   952→ */
   953→   953→async function executeJina(args: Record<string, unknown>): Promise<unknown> {
   954→   954→  const url = args.url as string
   955→   955→  if (!url) return { error: 'URL is required' }
   956→   956→
   957→   957→  // Validate URL
   958→   958→  try {
   959→   959→    new URL(url)
   960→   960→  } catch {
   961→   961→    return { error: `Invalid URL: ${url}` }
   962→   962→  }
   963→   963→
   964→   964→  // Primary: Direct Jina API with key pool rotation
   965→   965→  if (jinaKeyPool.hasKeys()) {
   966→   966→    const keySelection = jinaKeyPool.getNextKey()
   967→   967→    if (keySelection) {
   968→   968→      try {
   969→   969→        const headers: Record<string, string> = {
   970→   970→          'Accept': 'text/plain',
   971→   971→        }
   972→   972→        if (keySelection.key) {
   973→   973→          headers['Authorization'] = `Bearer ${keySelection.key}`
   974→   974→        }
   975→   975→
   976→   976→        const res = await fetch(`https://r.jina.ai/${url}`, {
   977→   977→          method: 'GET',
   978→   978→          headers,
   979→   979→          signal: AbortSignal.timeout(30000),
   980→   980→        })
   981→   981→
   982→   982→        if (res.ok) {
   983→   983→          jinaKeyPool.reportResult(keySelection.index, true, res.status)
   984→   984→          const text = await res.text()
   985→   985→
   986→   986→          // Limit output to prevent overwhelming LLM context
   987→   987→          const maxChars = 30000
   988→   988→          const truncated = text.length > maxChars
   989→   989→          const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
   990→   990→
   991→   991→          return {
   992→   992→            url,
   993→   993→            content,
   994→   994→            length: text.length,
   995→   995→            truncated,
   996→   996→            source: `jina-key${keySelection.index + 1}`,
   997→   997→          }
   998→   998→        }
   999→   999→
  1000→  1000→        // API returned error status — report to pool and try next key
  1001→  1001→        const errorBody = await res.text().catch(() => '')
  1002→  1002→        jinaKeyPool.reportResult(keySelection.index, false, res.status, `HTTP ${res.status}: ${errorBody.slice(0, 200)}`)
  1003→  1003→        console.warn(`[ToolExecutor] Jina key ${keySelection.index + 1} error: ${res.status}`)
  1004→  1004→
  1005→  1005→        // Try remaining keys before falling back to SDK
  1006→  1006→        const retryResult = await retryWithNextKey('jina', url)
  1007→  1007→        if (retryResult) return retryResult
  1008→  1008→      } catch (err) {
  1009→  1009→        const errMsg = err instanceof Error ? err.message : String(err)
  1010→  1010→        jinaKeyPool.reportResult(keySelection.index, false, undefined, errMsg)
  1011→  1011→        console.warn(`[ToolExecutor] Jina key ${keySelection.index + 1} failed: ${errMsg}`)
  1012→  1012→
  1013→  1013→        // Try remaining keys before falling back to SDK
  1014→  1014→        const retryResult = await retryWithNextKey('jina', url)
  1015→  1015→        if (retryResult) return retryResult
  1016→  1016→      }
  1017→  1017→    }
  1018→  1018→  }
  1019→  1019→
  1020→  1020→  // Fallback: z-ai-web-dev-sdk page_reader
  1021→  1021→  try {
  1022→  1022→    const zai = await getZAI()
  1023→  1023→    if (zai) {
  1024→  1024→      const result = await zai.functions.invoke('page_reader', { url })
  1025→  1025→      if (result?.data?.html) {
  1026→  1026→        // Convert HTML to clean text
  1027→  1027→        const text = result.data.html
  1028→  1028→          .replace(/<script[^>]*>[\s\S]*?<\/script>/gi, '')
  1029→  1029→          .replace(/<style[^>]*>[\s\S]*?<\/style>/gi, '')
  1030→  1030→          .replace(/<[^>]*>/g, ' ')
  1031→  1031→          .replace(/&nbsp;/g, ' ')
  1032→  1032→          .replace(/&amp;/g, '&')
  1033→  1033→          .replace(/&lt;/g, '<')
  1034→  1034→          .replace(/&gt;/g, '>')
  1035→  1035→          .replace(/\s+/g, ' ')
  1036→  1036→          .trim()
  1037→  1037→
  1038→  1038→        const maxChars = 30000
  1039→  1039→        const truncated = text.length > maxChars
  1040→  1040→        const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
  1041→  1041→
  1042→  1042→        return {
  1043→  1043→          url,
  1044→  1044→          title: result.data.title || null,
  1045→  1045→          content,
  1046→  1046→          length: text.length,
  1047→  1047→          truncated,
  1048→  1048→          source: 'z-ai-sdk-page_reader',
  1049→  1049→        }
  1050→  1050→      }
  1051→  1051→    }
  1052→  1052→  } catch (err) {
  1053→  1053→    console.warn(`[ToolExecutor] z-ai-sdk page_reader fallback failed: ${err instanceof Error ? err.message : String(err)}`)
  1054→  1054→  }
  1055→  1055→
  1056→  1056→  return {
  1057→  1057→    error: 'Jina reader failed and no fallback available.',
  1058→  1058→    hint: !jinaKeyPool.hasKeys() ? 'Add JINA_API_KEY_1..2 to .env' : `All ${jinaKeyPool.getTotalCount()} keys failed (${jinaKeyPool.getAvailableCount()} available)`,
  1059→  1059→    poolStatus: jinaKeyPool.getSummary(),
  1060→  1060→  }
  1061→  1061→}

  1128→  1128→
  1129→  1129→/**
  1130→  1130→ * Process a single LLM response — check for tool_calls and execute them.
  1131→  1131→ * Returns array of tool execution results for feeding back to LLM.
  1132→  1132→ *
  1133→  1133→ * This is the core of the ReAct loop:
  1134→  1134→ * 1. LLM returns response with tool_calls
  1135→  1135→ * 2. We execute each tool_call
  1136→  1136→ * 3. Results are fed back as tool messages
  1137→  1137→ * 4. LLM generates next response (may have more tool_calls or final text)
  1138→  1138→ */
  1139→  1139→export async function processToolCalls(
  1140→  1140→  toolCalls: Array<{ id: string; function: { name: string; arguments: string } }>,
  1141→  1141→): Promise<Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }>> {
  1142→  1142→  const results: Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }> = []
  1143→  1143→
  1144→  1144→  for (const tc of toolCalls) {
  1145→  1145→    let args: Record<string, unknown> = {}
  1146→  1146→    try {
  1147→  1147→      args = JSON.parse(tc.function.arguments)
  1148→  1148→    } catch {
  1149→  1149→      args = { _parseError: tc.function.arguments }
  1150→  1150→    }
  1151→  1151→
  1152→  1152→    console.log(`[ToolExecutor] Executing: ${tc.function.name}(${JSON.stringify(args).slice(0, 200)})`)
  1153→  1153→
  1154→  1154→    const result = await executeTool(tc.function.name, args)
  1155→  1155→
  1156→  1156→    results.push({
  1157→  1157→      toolCallId: tc.id,
  1158→  1158→      toolName: tc.function.name,
  1159→  1159→      result,
  1160→  1160→    })
  1161→  1161→
  1162→  1162→    console.log(`[ToolExecutor] Result: ${tc.function.name} → ${result.success ? 'OK' : 'FAIL'} (${result.duration}ms)`)
  1163→  1163→  }
  1164→  1164→
  1165→  1165→  return results
  1166→  1166→}
