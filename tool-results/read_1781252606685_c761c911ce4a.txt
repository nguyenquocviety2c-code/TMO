    41→async function isGatewayOnlineCached(): Promise<boolean> {
    42→  const now = Date.now()
    43→  if (now - gatewayCacheTime < GATEWAY_CACHE_TTL) return gatewayOnlineCached
    44→  gatewayCacheTime = now
    45→  const result = await isGatewayOnline()

    54→function isTrivialExchange(userMsg: string, assistantMsg: string): boolean {
    55→  // Skip memory extraction for trivial exchanges (greetings, short confirmations)
    56→  if (userMsg.trim().length < 15 && assistantMsg.trim().length < 50) return true

    66→/**
    67→ * Run a ReAct (Reason-Act-Observe) loop for an agent with tool execution.
    68→ *
    69→ * This enables standalone agents like Omega to actually USE their declared tools
    70→ * (opencode, knowledge_search, tavily, etc.) instead of just having them in prompts.
    71→ *
    72→ * Flow:
    73→ * 1. Call LLM with system prompt + user message
    74→ * 2. Parse tool_call patterns from LLM output
    75→ * 3. If tool calls found → execute → feed results back → repeat
    76→ * 4. If no tool calls → return final answer
    77→ * 5. Max 5 iterations to prevent infinite loops
    78→ */
    79→async function runAgentReActLoop(params: {
    80→  systemPrompt: string
    81→  userMessage: string
    82→  chatHistory: Array<{ role: string; content: string }>
    83→  agentProvider: string
    84→  agentModel: string
    85→  agentId: string
    86→  agentName: string
    87→  agentTemperature?: number
    88→  agentMaxTokens?: number
    89→  agentTools?: string[]
    90→  maxIterations?: number
    91→}): Promise<{ content: string; iterations: number; toolCallsExecuted: number }> {
    92→  const {
    93→    systemPrompt,
    94→    userMessage,
    95→    chatHistory,
    96→    agentProvider,
    97→    agentModel,
    98→    agentId,
    99→    agentName,
   100→    agentTemperature,
   101→    agentMaxTokens,
   102→    agentTools = [],
   103→    maxIterations = 5,
   104→  } = params
   105→
   106→  // Build messages array for multi-turn conversation using ChatMessageForLLM type
   107→  // This preserves structured tool_calls with tool_call_id for proper multi-turn tool calling
   108→  const messages: ChatMessageForLLM[] = [
   109→    { role: 'system', content: systemPrompt },
   110→    // Include last 6 messages from chat history for context
   111→    ...chatHistory.slice(-6).map(m => ({
   112→      role: m.role as 'user' | 'assistant',
   113→      content: m.content,
   114→    })),
   115→    { role: 'user', content: userMessage },
   116→  ]
   117→
   118→  let finalContent = ''
   119→  let iterations = 0
   120→  let totalToolCalls = 0
   121→
   122→  while (iterations < maxIterations) {
   123→    iterations++
   124→
   125→    // Get tool definitions for this agent (for structured tool calling)
   126→    const toolDefs = agentTools.length > 0 ? getToolDefinitionsForLLM(agentTools) : []
   127→
   128→    // Call LLM with full messages array — preserves tool_calls structure (CRITICAL FIX)
   129→    // Previously used formatMessagesForLLM() which flattened tool_calls to text, losing tool_call_id
   130→    const result = await callLLMForAgentWithMessages(
   131→      messages,
   132→      { provider: agentProvider, model: agentModel },
   133→      {
   134→        agentId,
   135→        agentName,
   136→        temperature: agentTemperature,
   137→        maxTokens: agentMaxTokens,
   138→        tools: toolDefs.length > 0 ? toolDefs : undefined,
   139→      }
   140→    )
   141→
   142→    if (!result || (result.error && !result.content && !result.toolCalls?.length)) {
   143→      finalContent = result?.error || 'LLM call failed'
   144→      break
   145→    }
   146→
   147→    finalContent = result.content || ''
   148→
   149→    // Parse tool calls — prefer structured tool_calls from LLM, fallback to regex
   150→    let toolCalls: Array<{ id: string; name: string; args: Record<string, unknown> }> = []
   151→
   152→    if (result.toolCalls && result.toolCalls.length > 0) {
   153→      // ✅ Structured tool_calls from LLM (preferred — P0.4)
   154→      toolCalls = result.toolCalls.map(tc => ({
   155→        id: tc.id,
   156→        name: tc.function.name,
   157→        args: (() => { try { return JSON.parse(tc.function.arguments) } catch { return {} } })(),
   158→      }))
   159→      console.log(`[ReAct] ${agentName} structured tool_calls: ${toolCalls.map(tc => tc.name).join(', ')}`)
   160→    } else {
   161→      // ❌ Fallback regex parsing (for models that don't support tool calling)
   162→      toolCalls = parseToolCallsFromOutput(result.content)
   163→    }
   164→
   165→    if (toolCalls.length === 0) {
   166→      // No tool calls → agent is done
   167→      break
   168→    }
   169→
   170→    // Execute tool calls
   171→    console.log(`[ReAct] ${agentName} calling ${toolCalls.length} tools (iteration ${iterations})`)
   172→
   173→    // Add assistant message with tool calls to conversation
   174→    messages.push({
   175→      role: 'assistant',
   176→      content: result.content,
   177→      ...(result.toolCalls && result.toolCalls.length > 0 ? {
   178→        tool_calls: result.toolCalls,
   179→      } : {}),
   180→    })
   181→
   182→    for (const tc of toolCalls) {
   183→      totalToolCalls++
   184→
   185→      // Only execute tools that the agent has permission for
   186→      if (agentTools.length > 0 && !agentTools.includes(tc.name)) {
   187→        messages.push({
   188→          role: 'tool',
   189→          content: JSON.stringify({ tool: tc.name, success: false, result: `Tool "${tc.name}" not available for this agent. Available tools: ${agentTools.join(', ')}` }),
   190→          tool_call_id: tc.id,
   191→        })
   192→        continue
   193→      }
   194→
   195→      console.log(`[ReAct] Executing: ${tc.name}(${JSON.stringify(tc.args).slice(0, 200)})`)
   196→
   197→      const toolResult = await executeTool(tc.name, tc.args)
   198→
   199→      // Add tool result to conversation (with tool_call_id for OpenAI format)
   200→      messages.push({
   201→        role: 'tool',
   202→        content: JSON.stringify({
   203→          tool: tc.name,
   204→          success: toolResult.success,
   205→          result: toolResult.result,
   206→        }),
   207→        tool_call_id: tc.id,
   208→      })
   209→    }
   210→
   211→    // If max iterations reached, break
   212→    if (iterations >= maxIterations) {
   213→      finalContent += `\n\n⚠️ Đạt max ${maxIterations} vòng iteration.`
   214→      break
   215→    }
   216→  }
   217→
   218→  console.log(`[ReAct] ${agentName} done: ${iterations} iterations, ${totalToolCalls} tool calls`)
   219→  return { content: finalContent, iterations, toolCallsExecuted: totalToolCalls }
   220→}

   394→async function routeToOpenCode(
   395→  userMessage: string,
   396→  model?: string,
   397→  sessionId?: string,
   398→  messages?: Array<{ role: string; content: string }>,
   399→  agentProfileId?: string,
   400→  teamMode?: string,
   401→  teamName?: string,
   402→): Promise<NextResponse | null> {
   403→  try {
   404→    const online = await isOpenCodeOnline()
   405→    if (!online) return null
   406→
   407→    // OC-3.1: Enrich with Knowledge Base context before creating session
   408→    const codeConfidence = detectCodeConfidence(userMessage)
   409→    const enrichment = await enrichCodeContext(userMessage)
   410→    
   411→    // OC-3.5: Cross-Session Learning — incorporate past corrections and insights
   412→    let entityTypeCount = 0
   413→    let documentCount = 0  
   414→    let correctionCount = 0
   415→    let insightCount = 0
   416→    let filesInWorkspace = 0
   417→    try {
   418→      entityTypeCount = await db.localEntity.count()
   419→      correctionCount = await db.agentCorrection.count({ where: { applied: true } })
   420→      insightCount = await db.agentInsight.count()
   421→      documentCount = await db.localEntity.count({ where: { domain: 'document' } })
   422→    } catch {}
   423→
   424→    // Count workspace files (cross-platform: uses Node.js fs instead of `find` command)
   425→    try {
   426→      const { readdirSync, statSync } = await import('fs')
   427→      const { join } = await import('path')
   428→      const workspaceDir = process.env.OPENCODE_WORKSPACE || process.cwd()
   429→      const skipDirs = new Set(['.next', 'node_modules', '.git', 'qdrant-storage', 'db', 'upload', 'skills'])
   430→      function countFiles(dir: string): number {
   431→        let count = 0
   432→        try {
   433→          for (const entry of readdirSync(dir)) {
   434→            const full = join(dir, entry)
   435→            try {
   436→              const stat = statSync(full)
   437→              if (stat.isDirectory()) {
   438→                if (!skipDirs.has(entry)) count += countFiles(full)
   439→              } else {
   440→                count++
   441→              }
   442→            } catch {}
   443→          }
   444→        } catch {}
   445→        return count
   446→      }
   447→      filesInWorkspace = countFiles(workspaceDir)
   448→    } catch {}
   449→
   450→    // OC-3.7: Generate enriched system prompt with KB context
   451→    const enrichedSystemPrompt = generateOpenCodeSystemPrompt({
   452→      entityTypeCount,
   453→      documentCount,
   454→      correctionCount,
   455→      insightCount,
   456→      filesInWorkspace,
   457→      modelList: [],
   458→      kbEnabled: true,
   459→      mcpToolsEnabled: ['knowledge_search', 'knowledge_graph', 'knowledge_write', 'web_search'],
   460→      recentCorrections: enrichment.corrections,
   461→      recentInsights: enrichment.insights.slice(0, 5),
   462→      relatedEntities: enrichment.entities,
   463→    })
   464→
   465→    // Try to create an OpenCode session with the prompt + enriched context
   466→    const session = await createOpenCodeSession({
   467→      prompt: userMessage,
   468→      model: model || undefined,
   469→    })
   470→
   471→    if (!session) return null
   472→
   473→    // Track in SQLite
   474→    if (sessionId) {
   475→      try {
   476→        await db.agentSession.upsert({
   477→          where: { sessionId },
   478→          create: {
   479→            sessionId,
   480→            model: model || 'opencode',
   481→            provider: 'opencode',
   482→            title: userMessage.slice(0, 50) || 'Code Session',
   483→            messageCount: 1,
   484→            agentProfileId: agentProfileId || null,
   485→            teamMode: teamMode || null,
   486→            teamName: teamName || null,
   487→          },
   488→          update: {
   489→            messageCount: { increment: 1 },
   490→            model: model || undefined,
   491→          },
   492→        })
   493→      } catch {}
   494→    }
   495→
   496→    // Also track as OpenCodeSession
   497→    try {
   498→      await db.openCodeSession.create({
   499→        data: {
   500→          sessionId: session.sessionId || session.id || `oc-${Date.now()}`,
   501→          model: model || null,
   502→          provider: 'opencode',
   503→          prompt: userMessage,
   504→          status: 'active',
   505→        },
   506→      })
   507→    } catch {}
   508→
   509→    // Build enrichment info for the response
   510→    const enrichmentInfo = enrichment.enrichmentScore > 0
   511→      ? `\n\n📊 **KB Context Enrichment** (score: ${(enrichment.enrichmentScore * 100).toFixed(0)}%):\n` +
   512→        (enrichment.entities.length > 0 ? `• Entities: ${enrichment.entities.length} found (${enrichment.entities.slice(0, 3).map(e => e.name).join(', ')}${enrichment.entities.length > 3 ? '...' : ''})\n` : '') +
   513→        (enrichment.documents.length > 0 ? `• Documents: ${enrichment.documents.length} relevant\n` : '') +
   514→        (enrichment.corrections.length > 0 ? `• Corrections: ${enrichment.corrections.length} from past sessions\n` : '') +
   515→        (enrichment.insights.length > 0 ? `• Insights: ${enrichment.insights.length} available\n` : '')
   516→      : ''
   517→
   518→    // Return a response indicating the session was created with KB enrichment
   519→    return NextResponse.json({

   547→/**
   548→ * Route chat through hybrid Gateway + local architecture.
   549→ *
   550→ * Architecture (Hybrid ReAct):
   551→ *   - LLM calls → NVIDIA NIM directly (callLLMForAgent)
   552→ *   - Tool execution → Gateway /tools/invoke when available, local fallback
   553→ *
   554→ * This replaces the old approach of routing ALL through Gateway /v1/chat/completions
   555→ * which returned 404 (Gateway uses WebSocket internally, not REST for chat).
   556→ */
   557→async function routeThroughGateway(params: {
   558→  messages: Array<{ role: string; content: string }>
   559→  agentProvider: string
   560→  agentModel: string
   561→  sessionId?: string
   562→  temperature?: number
   563→  maxTokens?: number
   564→  agentProfileId?: string
   565→  agentProfileName?: string
   566→  teamMode?: string
   567→  teamName?: string
   568→  agentTools?: string[]
   569→}): Promise<NextResponse | null> {
   570→  const { messages, agentProvider, agentModel, sessionId, temperature, maxTokens,
   571→          agentProfileId, agentProfileName, teamMode, teamName, agentTools = [] } = params
   572→
   573→  try {
   574→    // Hybrid approach: Use local ReAct loop with Gateway tool proxy
   575→    // This avoids the 404 from /v1/chat/completions while still using Gateway tools
   576→    const lastUserMsg = [...messages].reverse().find(m => m.role === 'user')
   577→    const userContent = typeof lastUserMsg?.content === 'string' ? lastUserMsg.content : ''
   578→    const systemMsg = messages.find(m => m.role === 'system')
   579→    const systemPrompt = systemMsg?.content || ''
   580→    const chatHistory = messages.filter(m => m.role === 'user' || m.role === 'assistant')
   581→
   582→    // Use the existing ReAct loop which already supports Gateway tool proxy
   583→    const reactResult = await runAgentReActLoop({
   584→      systemPrompt,
   585→      userMessage: userContent,
   586→      chatHistory,
   587→      agentProvider,
   588→      agentModel,
   589→      agentId: agentProfileId || 'unknown',
   590→      agentName: agentProfileName || 'Agent',
   591→      agentTemperature: temperature,
   592→      agentMaxTokens: maxTokens,
   593→      agentTools,
   594→      maxIterations: 5,
   595→    })
   596→
   597→    // Track session
   598→    if (sessionId) {
   599→      await db.agentSession.upsert({
   600→        where: { sessionId },
   601→        create: {
   602→          sessionId,
   603→          model: agentModel,
   604→          provider: agentProvider,
   605→          title: userContent.slice(0, 50) || 'New Chat',
   606→          messageCount: messages.length + 1,
   607→          agentProfileId: agentProfileId || null,
   608→          teamMode: teamMode || null,
   609→          teamName: teamName || null,
   610→        },
   611→        update: { messageCount: { increment: 1 } },
   612→      }).catch(() => {})
   613→    }
   614→
   615→    return NextResponse.json({
   616→      content: reactResult.content,
   617→      model: agentModel,
   618→      provider: agentProvider,
   619→      sessionId: sessionId || `session-${Date.now()}`,
   620→      agentProfileId: agentProfileId || undefined,
   621→      sources: [],
   622→      confidence: 0.7,
   623→      kbResults: 'hybrid-react',
   624→      gatewayToolsAvailable: true,
   625→      reactIterations: reactResult.iterations,
   626→      toolCallsExecuted: reactResult.toolCallsExecuted,
   627→    })
   628→  } catch (err) {
   629→    console.warn('[Gateway Hybrid] Error:', err instanceof Error ? err.message : String(err))
   630→    return null
   631→  }
   632→}

  1039→            skipMetaCog: true,
  1040→            ...(agentTemperature !== undefined ? { temperature: agentTemperature } : {}),
  1041→            ...(agentMaxTokens !== undefined ? { maxTokens: agentMaxTokens } : {}),
  1042→            ...(agentProfileId ? { agentId: agentProfileId } : {}),
  1043→            ...(agentProfileName ? { agentName: agentProfileName } : {}),
  1044→            ...(agentProvider ? { agentProvider } : {}),
  1045→            ...(agentModel ? { agentModel } : {}),
  1046→          },
  1047→        }),
  1048→        signal: AbortSignal.timeout(60000),
  1049→      })
  1050→
  1051→      if (queryRes.ok) {

  1053→        const content = data.answer || 'Không thể tạo câu trả lời.'
  1054→
  1055→        // If this was a code query but OpenCode was offline, add a note
  1056→        const codeNote = isCodeQuery(lastUserContent)
  1057→          ? '\n\n⚠️ **OpenCode Server đang offline** — Không thể thực hiện code editing. Kết quả trên chỉ dựa trên Knowledge Base. Hãy bật OpenCode Server để sử dụng đầy đủ tính năng code.'
  1058→          : ''
  1059→
  1060→        // Track session (include agentProfileId if from an Agent)
  1061→        // NOTE: agentTemperature/agentMaxTokens are now passed to the local fallback via /api/query
  1062→        if (sessionId) {
  1063→          try {
  1064→            await db.agentSession.upsert({
  1065→              where: { sessionId },
  1066→              create: {
  1067→                sessionId,
  1068→                model: model || 'local-rag',
  1069→                provider: 'local-fallback',
  1070→                title: messages[0]?.content?.slice(0, 50) || 'New Chat',
  1071→                messageCount: messages.length + 1,
  1072→                agentProfileId: agentProfileId || null,
  1073→                teamMode: teamMode || null,
  1074→                teamName: teamName || null,
  1075→              },

  1077→                messageCount: { increment: 1 },

  1079→            })
  1080→          } catch {}
  1081→        }
  1082→
  1083→        // Phase 4: Auto-Learn — if agent has provider/model and confidence is in learning zone,
  1084→        // trigger background learning so the knowledge is available for future queries.
  1085→        // Non-blocking: fire-and-forget, don't delay the response.
  1086→        const queryConfidence = data.confidence ?? 0
  1087→        const hasAgent = !!(agentProfileId && agentProvider && agentModel)
  1088→        if (shouldAutoLearn(queryConfidence, hasAgent)) {

  1093→            agentId: agentProfileId!,
  1094→            agentName: agentProfileName || 'unknown',

  1098→          }).catch(err => console.error('[AutoLearn] Background error:', err instanceof Error ? err.message : String(err)))
  1099→        }
  1100→
  1101→        // Phase 5: Memory & Learning — save messages + extract memories in background
  1102→        // Non-blocking: fire-and-forget, don't delay the response.
  1103→        if (agentProfileId && sessionId) {
  1104→          // Save chat messages to DB for persistence
  1105→          saveChatMessages(sessionId, [
  1106→            { role: 'user', content: lastUserContent },
  1107→            { role: 'assistant', content, model: agentModel || model, provider: agentProvider || 'local-fallback', metadata: { confidence: queryConfidence, sources: data.sources?.length || 0 } },
  1108→          ]).catch(err => console.warn('[Memory] Failed to save chat messages:', err instanceof Error ? err.message : String(err)))
  1109→
  1110→          // Extract memories from this conversation (skip trivial exchanges to save LLM tokens)
  1111→          if (!isTrivialExchange(lastUserContent, content)) {
  1112→            extractMemoriesFromConversation({
  1113→              agentId: agentProfileId,
  1114→              agentName: agentProfileName || 'unknown',
  1115→              sessionId,
  1116→              userMessage: lastUserContent,
  1117→              assistantMessage: content,
  1118→            }).catch(err => console.warn('[Memory] Failed to extract memories:', err instanceof Error ? err.message : String(err)))
  1119→          }
  1120→
  1121→          // Periodic memory decay — every ~20 chat requests, decay old memories
  1122→          // This prevents stale memories from accumulating without requiring a cron scheduler
  1123→          memoryDecayCounter++
  1124→          if (memoryDecayCounter >= 20 && agentProfileId) {
  1125→            memoryDecayCounter = 0
  1126→            decayMemories(agentProfileId).catch(err =>
  1127→              console.warn('[Memory] Periodic decay error:', err instanceof Error ? err.message : String(err))
  1128→            )
  1129→          }
  1130→        }
  1131→
  1132→        return NextResponse.json({

  1134→          model: model || 'local-rag',
  1135→          provider: 'local-fallback',
  1136→          sessionId: sessionId || `session-${Date.now()}`,
  1137→          agentProfileId: agentProfileId || undefined,
  1138→          sources: data.sources || [],
  1139→          confidence: data.confidence,
  1140→          kbResults: 'included',
  1141→          codeQuery: isCodeQuery(lastUserContent) || undefined,
  1142→          memoriesRecalled: recalledMemories.length || undefined,
  1143→        })
  1144→      }
  1145→    } catch (err) {
  1146→      console.warn('[OpenClaw Chat] Local fallback error:', err instanceof Error ? err.message : String(err))
  1147→    }
  1148→
  1149→    // Final fallback: mock response
  1150→    const userContent = lastUserContent || ''
  1151→    const mockContent = userContent.toLowerCase().includes('hello') || userContent.toLowerCase().includes('hi')
  1152→      ? `Xin chào! Tôi là AI Agent của The Magnum Opus. Gateway hiện đang offline nên tôi hoạt động ở chế độ mock. Hãy kết nối Gateway để nhận phản hồi AI thực sự.`
  1153→      : `Tôi đã nhận tin nhắn của bạn: "${userContent.slice(0, 100)}${userContent.length > 100 ? '...' : ''}"\n\nOpenClaw Gateway đang offline và Knowledge Base cũng không khả dụng. Tôi đang ở chế độ mock. Vui lòng kiểm tra kết nối.`
  1154→
  1155→    return NextResponse.json({
  1156→      content: mockContent,
  1157→      model: model || 'mock',
  1158→      provider: 'mock',
  1159→      sessionId: sessionId || `session-${Date.now()}`,
  1160→      kbResults: 'none',
  1161→    })
  1162→  } catch (err) {
  1163→    return NextResponse.json(
  1164→      { error: 'Invalid request body', details: err instanceof Error ? err.message : String(err) },
  1165→      { status: 400 }
  1166→    )
  1167→  }
