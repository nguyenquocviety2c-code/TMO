    49→    49→// ============================================
    50→    50→
    51→    51→/**
    52→    52→ * Run a ReAct (Reason-Act-Observe) loop for an agent with tool execution.
    53→    53→ *
    54→    54→ * This enables standalone agents like Omega to actually USE their declared tools
    55→    55→ * (opencode, knowledge_search, tavily, etc.) instead of just having them in prompts.
    56→    56→ *
    57→    57→ * Flow:
    58→    58→ * 1. Call LLM with system prompt + user message
    59→    59→ * 2. Parse tool_call patterns from LLM output
    60→    60→ * 3. If tool calls found → execute → feed results back → repeat
    61→    61→ * 4. If no tool calls → return final answer
    62→    62→ * 5. Max 5 iterations to prevent infinite loops
    63→    63→ */
    64→    64→async function runAgentReActLoop(params: {
    65→    65→  systemPrompt: string
    66→    66→  userMessage: string
    67→    67→  chatHistory: Array<{ role: string; content: string }>
    68→    68→  agentProvider: string
    69→    69→  agentModel: string
    70→    70→  agentId: string
    71→    71→  agentName: string
    72→    72→  agentTemperature?: number
    73→    73→  agentMaxTokens?: number
    74→    74→  agentTools?: string[]
    75→    75→  maxIterations?: number
    76→    76→}): Promise<{ content: string; iterations: number; toolCallsExecuted: number }> {
    77→    77→  const {
    78→    78→    systemPrompt,
    79→    79→    userMessage,
    80→    80→    chatHistory,
    81→    81→    agentProvider,
    82→    82→    agentModel,
    83→    83→    agentId,
    84→    84→    agentName,
    85→    85→    agentTemperature,
    86→    86→    agentMaxTokens,
    87→    87→    agentTools = [],
    88→    88→    maxIterations = 5,
    89→    89→  } = params
    90→    90→
    91→    91→  // Build messages array for multi-turn conversation
    92→    92→  const messages: Array<{
    93→    93→    role: 'system' | 'user' | 'assistant' | 'tool'
    94→    94→    content: string
    95→    95→    tool_call_id?: string
    96→    96→  }> = [
    97→    97→    { role: 'system', content: systemPrompt },
    98→    98→    // Include last 6 messages from chat history for context
    99→    99→    ...chatHistory.slice(-6).map(m => ({
   100→   100→      role: m.role as 'user' | 'assistant',
   101→   101→      content: m.content,
   102→   102→    })),
   103→   103→    { role: 'user', content: userMessage },
   104→   104→  ]
   105→   105→
   106→   106→  let finalContent = ''
   107→   107→  let iterations = 0
   108→   108→  let totalToolCalls = 0
   109→   109→
   110→   110→  while (iterations < maxIterations) {
   111→   111→    iterations++
   112→   112→
   113→   113→    // Format messages for LLM (since callLLMForAgent takes a single prompt string)
   114→   114→    const formattedPrompt = formatMessagesForLLM(messages)
   115→   115→
   116→   116→    // Call LLM for this agent
   117→   117→    const result = await callLLMForAgent(
   118→   118→      formattedPrompt,
   119→   119→      { provider: agentProvider, model: agentModel },
   120→   120→      systemPrompt,
   121→   121→      {
   122→   122→        agentId,
   123→   123→        agentName,
   124→   124→        temperature: agentTemperature,
   125→   125→        maxTokens: agentMaxTokens,
   126→   126→      }
   127→   127→    )
   128→   128→
   129→   129→    if (!result || (result.error && !result.content)) {
   130→   130→      finalContent = result?.error || 'LLM call failed'
   131→   131→      break
   132→   132→    }
   133→   133→
   134→   134→    finalContent = result.content
   135→   135→
   136→   136→    // Check if LLM output contains tool calls
   137→   137→    const toolCalls = parseToolCallsFromOutput(result.content)
   138→   138→
   139→   139→    if (toolCalls.length === 0) {
   140→   140→      // No tool calls → agent is done
   141→   141→      break
   142→   142→    }
   143→   143→
   144→   144→    // Execute tool calls
   145→   145→    console.log(`[ReAct] ${agentName} calling ${toolCalls.length} tools (iteration ${iterations})`)
   146→   146→
   147→   147→    // Add assistant message with tool calls to conversation
   148→   148→    messages.push({
   149→   149→      role: 'assistant',
   150→   150→      content: result.content,
   151→   151→    })
   152→   152→
   153→   153→    for (const tc of toolCalls) {
   154→   154→      totalToolCalls++
   155→   155→
   156→   156→      // Only execute tools that the agent has permission for
   157→   157→      if (agentTools.length > 0 && !agentTools.includes(tc.name)) {
   158→   158→        messages.push({
   159→   159→          role: 'tool',
   160→   160→          content: JSON.stringify({ tool: tc.name, success: false, result: `Tool "${tc.name}" not available for this agent. Available tools: ${agentTools.join(', ')}` }),
   161→   161→        })
   162→   162→        continue
   163→   163→      }
   164→   164→
   165→   165→      console.log(`[ReAct] Executing: ${tc.name}(${JSON.stringify(tc.args).slice(0, 200)})`)
   166→   166→
   167→   167→      const toolResult = await executeTool(tc.name, tc.args)
   168→   168→
   169→   169→      // Add tool result to conversation
   170→   170→      messages.push({
   171→   171→        role: 'tool',
   172→   172→        content: JSON.stringify({
   173→   173→          tool: tc.name,
   174→   174→          success: toolResult.success,
   175→   175→          result: toolResult.result,
   176→   176→        }),
   177→   177→      })
   178→   178→    }
   179→   179→
   180→   180→    // If max iterations reached, break
   181→   181→    if (iterations >= maxIterations) {
   182→   182→      finalContent += `\n\n⚠️ Đạt max ${maxIterations} vòng iteration.`
   183→   183→      break
   184→   184→    }
   185→   185→  }
   186→   186→
   187→   187→  console.log(`[ReAct] ${agentName} done: ${iterations} iterations, ${totalToolCalls} tool calls`)
   188→   188→  return { content: finalContent, iterations, toolCallsExecuted: totalToolCalls }
   189→   189→}

   410→   410→    // ============================================

   421→   421→          agentId: agentProfileId,

   423→   423→          topK: 5,
   424→   424→          minImportance: 0.3,

   426→   426→          console.warn('[OpenClaw Chat] Memory recall failed:', err instanceof Error ? err.message : String(err))
   427→   427→          return []
   428→   428→        })

   432→   432→        if (profile.length > 0) {

   438→   438→    }
