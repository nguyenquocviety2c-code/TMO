    74→/**
    75→ * Execute a tool via Gateway HTTP /tools/invoke endpoint.
    76→ * Only works for HTTP-allowed tools (web_search, web_fetch, etc.).
    77→ */
    78→async function executeGatewayTool(name: string, args: Record<string, unknown>): Promise<unknown> {
    79→  if (!GATEWAY_HTTP_ALLOWED_TOOLS.has(name)) {
    80→    throw new Error(`Tool "${name}" requires Gateway session context. Try the Bridge instead.`)
    81→  }
    82→
    83→  const res = await fetch(`${GATEWAY_URL}/tools/invoke`, {
    84→    method: 'POST',
    85→    headers: { 'Content-Type': 'application/json' },
    86→    body: JSON.stringify({ name, args }),
    87→    signal: AbortSignal.timeout(30000),
    88→  })
    89→
    90→  if (!res.ok) {
    91→    const errorText = await res.text().catch(() => 'Unknown error')
    92→    throw new Error(`Gateway tool error ${res.status}: ${errorText.slice(0, 200)}`)
    93→  }
    94→
    95→  const data = await res.json()
    96→  if (!data.ok && data.error) {
    97→    throw new Error(`Tool error: ${data.error?.message || data.error || 'Unknown'}`)
    98→  }
    99→
   100→  return data.result ?? data
   101→}

   104→ * Execute a tool via the Gateway Bridge (port 18791).
   105→ * The Bridge connects to the Gateway via WebSocket with full session context,
   106→ * enabling access to ALL 36 Gateway tools including session-context ones
   107→ * (exec, read, write, edit, image_generate, etc.).

   109→async function executeBridgeTool(name: string, args: Record<string, unknown>): Promise<unknown> {
   110→  const res = await fetch(`${BRIDGE_URL}/tools/invoke`, {
   111→    method: 'POST',
   112→    headers: { 'Content-Type': 'application/json' },
   113→    body: JSON.stringify({ name, args }),
   114→    signal: AbortSignal.timeout(60000),
   115→  })
   116→
   117→  if (!res.ok) {
   118→    const errorText = await res.text().catch(() => 'Unknown error')
   119→    throw new Error(`Bridge tool error ${res.status}: ${errorText.slice(0, 200)}`)
   120→  }
   121→
   122→  const data = await res.json()
   123→  if (!data.ok && data.error) {
   124→    throw new Error(`Bridge tool error: ${typeof data.error === 'string' ? data.error : data.error?.message || 'Unknown'}`)
   125→  }
   126→
   127→  return data.result ?? data
   128→}

   161→async function getZAI(): Promise<any> {

   165→  zaiInitPromise = (async () => {

   167→      const ZAI = (await import('z-ai-web-dev-sdk')).default

   171→      zaiAvailable = true
   172→      return instance
   173→    } catch (err) {
   174→      zaiAvailable = false

   176→        '[ToolExecutor] z-ai-web-dev-sdk not available (this is normal for local deployment — direct API keys will be used).',
   177→        `Detail: ${err instanceof Error ? err.message : String(err)}`
   178→      )
   179→      return null
   180→    }
   181→  })()
   182→
   183→  return zaiInitPromise

   383→// ==================== TOOL EXECUTION ====================
   384→
   385→export interface ToolExecutionResult {
   386→  success: boolean
   387→  result: unknown
   388→  duration: number
   389→}
   390→
   391→/**
   392→ * Execute a tool call from LLM.
   393→ * Called in the ReAct loop when LLM returns a tool_call.
   394→ *
   395→ * @param toolName - Name of the tool to execute
   396→ * @param args - Arguments from LLM's tool_call
   397→ * @returns Execution result with success status
   398→ */
   399→export async function executeTool(
   400→  toolName: string,
   401→  args: Record<string, unknown>
   402→): Promise<ToolExecutionResult> {
   403→  const startTime = Date.now()
   404→
   405→  try {
   406→    let result: unknown
   407→
   408→    switch (toolName) {
   409→      case 'opencode':
   410→        result = await executeOpenCode(args)
   411→        break
   412→      case 'knowledge_search':
   413→        result = await executeKnowledgeSearch(args)
   414→        break
   415→      case 'knowledge_graph':
   416→        result = await executeKnowledgeGraph(args)
   417→        break
   418→      case 'knowledge_write':
   419→        result = await executeKnowledgeWrite(args)
   420→        break
   421→      case 'tavily':
   422→        result = await executeTavily(args)
   423→        break
   424→      case 'serper':
   425→        result = await executeSerper(args)
   426→        break
   427→      case 'jina':
   428→        result = await executeJina(args)
   429→        break
   430→      default: {
   431→        // V2: Route to Gateway HTTP, Bridge, or error
   432→        // 1. Try Gateway HTTP for HTTP-accessible tools
   433→        if (GATEWAY_HTTP_ALLOWED_TOOLS.has(toolName)) {
   434→          try {
   435→            const gwResult = await executeGatewayTool(toolName, args)
   436→            return { success: true, result: gwResult, duration: Date.now() - startTime }
   437→          } catch (gwErr) {
   438→            return {
   439→              success: false,
   440→              result: `Gateway tool "${toolName}" failed: ${gwErr instanceof Error ? gwErr.message : String(gwErr)}`,
   441→              duration: Date.now() - startTime,
   442→            }
   443→          }
   444→        }
   445→        // 2. Try Gateway Bridge for session-context tools (exec, read, write, image_generate, etc.)
   446→        if (GATEWAY_BRIDGE_TOOLS.has(toolName)) {
   447→          try {
   448→            const bridgeResult = await executeBridgeTool(toolName, args)
   449→            return { success: true, result: bridgeResult, duration: Date.now() - startTime }
   450→          } catch (bridgeErr) {
   451→            return {
   452→              success: false,
   453→              result: `Bridge tool "${toolName}" failed: ${bridgeErr instanceof Error ? bridgeErr.message : String(bridgeErr)}`,
   454→              duration: Date.now() - startTime,
   455→            }
   456→          }
   457→        }
   458→        return { success: false, result: `Unknown tool: ${toolName}. Available: local(7) + Gateway HTTP(${GATEWAY_HTTP_ALLOWED_TOOLS.size}) + Bridge(${GATEWAY_BRIDGE_TOOLS.size}) = ${7 + GATEWAY_HTTP_ALLOWED_TOOLS.size + GATEWAY_BRIDGE_TOOLS.size} tools total`, duration: Date.now() - startTime }
   459→      }
   460→    }
   461→
   462→    return {
   463→      success: true,
   464→      result,
   465→      duration: Date.now() - startTime,
   466→    }
   467→  } catch (err) {
   468→    const errorMsg = err instanceof Error ? err.message : String(err)
   469→    console.error(`[ToolExecutor] Error executing ${toolName}:`, errorMsg)
   470→    return {
   471→      success: false,
   472→      result: `Tool error: ${errorMsg}`,
   473→      duration: Date.now() - startTime,
   474→    }
   475→  }
   476→}

   480→/**
   481→ * OpenCode — Code operations (read, write, bash, mkdir)
   482→ *
   483→ * Uses the OpenCode API server (configured via OPENCODE_SERVER_URL env var) if available,
   484→ * otherwise falls back to direct file system operations via the project API.
   485→ */
   486→async function executeOpenCode(args: Record<string, unknown>): Promise<unknown> {
   487→  const action = args.action as string
   488→  const path = args.path as string | undefined
   489→  const content = args.content as string | undefined
   490→  const command = args.command as string | undefined
   491→
   492→  // Try OpenCode server first
   493→  try {
   494→    const opencodeUrl = process.env.OPENCODE_SERVER_URL || 'http://localhost:18790'
   495→    const res = await fetch(`${opencodeUrl}/api/execute`, {
   496→      method: 'POST',
   497→      headers: { 'Content-Type': 'application/json' },
   498→      body: JSON.stringify({ action, path, content, command }),
   499→      signal: AbortSignal.timeout(30000),
   500→    })
   501→
   502→    if (res.ok) {
   503→      const data = await res.json()
   504→      return data
   505→    }
   506→  } catch {
   507→    // OpenCode server not available, fall through to direct implementation
   508→  }
   509→
   510→  // Fallback: Use the project's own API for file operations
   511→  switch (action) {
   512→    case 'read':
   513→      return await executeFileRead(path || '')
   514→    case 'write':
   515→      return await executeFileWrite(path || '', content || '')
   516→    case 'bash':
   517→      return await executeBash(command || '')
   518→    case 'mkdir':
   519→      return { success: true, message: `mkdir not needed — write will create parent dirs` }
   520→    default:
   521→      return { success: false, error: `Unknown opencode action: ${action}` }
   522→  }
   523→}

   525→/**
   526→ * Read a file from the project directory.
   527→ * Security: Only allows reading files within the project root.
   528→ * Uses path.resolve + startsWith check to prevent path traversal.
   529→ */
   530→async function executeFileRead(filePath: string): Promise<{ success: boolean; content?: string; error?: string }> {
   531→  if (!filePath) return { success: false, error: 'Path is required' }
   532→
   533→  try {
   534→    const pathModule = await import('path')
   535→    const projectRoot = process.cwd()
   536→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   537→
   538→    // Security: Prevent path traversal — resolved path must be within project root
   539→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   540→      return { success: false, error: `Access denied: path escapes project root` }
   541→    }
   542→
   543→    const fs = await import('fs/promises')
   544→    const content = await fs.readFile(resolvedPath, 'utf-8')
   545→
   546→    // Limit output to prevent overwhelming LLM context
   547→    const maxChars = 50000
   548→    const truncated = content.length > maxChars
   549→    const output = truncated ? content.slice(0, maxChars) + '\n\n... [TRUNCATED — file too large]' : content
   550→
   551→    return { success: true, content: output }
   552→  } catch (err) {
   553→    return {
   554→      success: false,
   555→      error: `Failed to read ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   556→    }

   560→/**
   561→ * Write a file to the project directory.
   562→ * Security: Only allows writing within the project root. Uses path.resolve to prevent traversal.
   563→ * Auto-creates parent directories.
   564→ */
   565→async function executeFileWrite(filePath: string, content: string): Promise<{ success: boolean; message: string; error?: string }> {
   566→  if (!filePath) return { success: false, message: '', error: 'Path is required' }
   567→  if (!content) return { success: false, message: '', error: 'Content is required' }
   568→
   569→  try {
   570→    const pathModule = await import('path')
   571→    const projectRoot = process.cwd()
   572→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   573→
   574→    // Security: Prevent path traversal — resolved path must be within project root
   575→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   576→      return { success: false, message: '', error: 'Access denied: path escapes project root' }
   577→    }
   578→
   579→    const fs = await import('fs/promises')
   580→
   581→    // Auto-create parent directories
   582→    const dir = pathModule.dirname(resolvedPath)
   583→    await fs.mkdir(dir, { recursive: true })
   584→
   585→    // Write file
   586→    await fs.writeFile(resolvedPath, content, 'utf-8')
   587→
   588→    return { success: true, message: `File written: ${filePath} (${content.length} chars)` }
   589→  } catch (err) {
   590→    return {
   591→      success: false,
   592→      message: '',
   593→      error: `Failed to write ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   594→    }
   595→  }

   598→/**
   599→ * Execute a bash command safely.
   600→ * Security: Only allows pre-approved commands.
   601→ */
   602→async function executeBash(command: string): Promise<{ success: boolean; output?: string; error?: string }> {
   603→  if (!command) return { success: false, error: 'Command is required' }
   604→
   605→  // Security: Whitelist of allowed commands
   606→  // NOTE: 'node' is excluded because `node -e "..."` can execute arbitrary code.
   607→  //       'cat'/'head'/'tail'/'grep' are excluded — use opencode read instead.
   608→  //       'npx'/'bunx' are excluded — can download+execute arbitrary npm packages.
   609→  const allowedPrefixes = [
   610→    'npm run',
   611→    'npm test',
   612→    'npm lint',
   613→    'bun run',
   614→    'bun test',
   615→    'bun lint',
   616→    'tsc',
   617→    'eslint',
   618→    'prettier',
   619→    'echo',
   620→    'ls',
   621→    'wc',
   622→    'rg',       // ripgrep — search only
   623→    'which',
   624→    'pwd',
   625→    'prisma',
   626→    'git status',
   627→    'git diff',
   628→    'git log',
   629→  ]
   630→
   631→  // Block dangerous flags that could allow code execution or file access
   632→  const dangerousFlags = ['-e ', '--eval ', '-r ', '--require ', '-p ', '--print ']
   633→  const normalizedCmd = command.trim()
   634→  const isAllowed = allowedPrefixes.some(prefix => normalizedCmd.startsWith(prefix))
   635→  const usesDangerousFlag = dangerousFlags.some(flag => normalizedCmd.includes(flag))
   636→
   637→  // Security: Block shell metacharacters that enable command chaining/injection
   638→  // These allow bypassing the whitelist by chaining additional commands:
   639→  //   "npm run build; rm -rf /"   →  ;  chains commands
   640→  //   "git status && cat /etc/p"  →  && chains on success
   641→  //   "echo hi | bash"            →  |  pipes output
   642→  //   "tsc || rm -rf /"           →  || chains on failure
   643→  //   "echo $(cat /etc/passwd)"   →  $() command substitution
   644→  //   "echo `cat /etc/passwd`"    →  backtick command substitution
   645→  //   "echo hi > /etc/passwd"     →  >  redirect write
   646→  //   "echo hi >> /etc/passwd"    →  >> redirect append
   647→  const shellMetacharacters = /(;|&&|\|\||\||`|\$\(|\$\{|>|>>)/
   648→  const usesShellMetacharacters = shellMetacharacters.test(normalizedCmd)
   649→
   650→  if (!isAllowed || usesDangerousFlag || usesShellMetacharacters) {
   651→    return {
   652→      success: false,
   653→      error: `Command not allowed: "${normalizedCmd.slice(0, 100)}". ${!isAllowed ? 'Prefix not in whitelist.' : ''} ${usesDangerousFlag ? 'Dangerous flag detected.' : ''} ${usesShellMetacharacters ? 'Shell metacharacter (;, &&, ||, |, `, $(), >, >>) detected — command chaining is forbidden.' : ''}`,
   654→    }
   655→  }
   656→
   657→  try {
   658→    const { exec } = await import('child_process')
   659→    const { promisify } = await import('util')
   660→    const execAsync = promisify(exec)
   661→
   662→    const { stdout, stderr } = await execAsync(normalizedCmd, {
   663→      cwd: process.cwd(),
   664→      timeout: 30000,
   665→      maxBuffer: 1024 * 1024, // 1MB
   666→    })
   667→
   668→    const output = [
   669→      stdout ? `STDOUT:\n${stdout}` : '',
   670→      stderr ? `STDERR:\n${stderr}` : '',
   671→    ].filter(Boolean).join('\n')
   672→
   673→    return { success: true, output: output || '(no output)' }
   674→  } catch (err) {
   675→    const error = err as { stdout?: string; stderr?: string; message?: string }
   676→    const output = [
   677→      error.stdout ? `STDOUT:\n${error.stdout}` : '',
   678→      error.stderr ? `STDERR:\n${error.stderr}` : '',
   679→    ].filter(Boolean).join('\n')
   680→
   681→    return {
   682→      success: false,
   683→      output: output || undefined,
   684→      error: `Command failed: ${error.message}`,
   685→    }
   686→  }
   687→}

   796→/**
   797→ * Retry a service call with the next available key from the pool.
   798→ * Called when the first key fails — tries remaining keys before falling back to SDK.
   799→ *
   800→ * @param service - 'tavily' | 'serper' | 'jina'
   801→ * @param param - The query (for search) or URL (for reader)
   802→ * @returns Result if a key succeeds, null if all keys fail
   803→ */
   804→async function retryWithNextKey(
   805→  service: 'tavily' | 'serper' | 'jina',
   806→  param: string
   807→): Promise<unknown | null> {
   808→  const pool = service === 'tavily' ? tavilyKeyPool : service === 'serper' ? serperKeyPool : jinaKeyPool
   809→  const maxRetries = pool.getTotalCount() - 1 // Already tried 1 key
   810→
   811→  for (let i = 0; i < maxRetries; i++) {
   812→    const nextKey = pool.getNextKey()
   813→    if (!nextKey || nextKey.isLastResort) break // No more healthy keys
   814→
   815→    console.log(`[ToolExecutor] ${service} retrying with key ${nextKey.index + 1}...`)
   816→
   817→    try {
   818→      let res: Response
   819→
   820→      if (service === 'tavily') {
   821→        res = await fetch('https://api.tavily.com/search', {
   822→          method: 'POST',
   823→          headers: { 'Content-Type': 'application/json' },
   824→          body: JSON.stringify({
   825→            api_key: nextKey.key,
   826→            query: param,
   827→            search_depth: 'advanced',
   828→            include_answer: true,
   829→            max_results: 5,
   830→          }),
   831→          signal: AbortSignal.timeout(30000),
   832→        })
   833→      } else if (service === 'serper') {
   834→        res = await fetch('https://google.serper.dev/search', {
   835→          method: 'POST',
   836→          headers: {
   837→            'X-API-KEY': nextKey.key,
   838→            'Content-Type': 'application/json',
   839→          },
   840→          body: JSON.stringify({ q: param, num: 5 }),
   841→          signal: AbortSignal.timeout(15000),
   842→        })
   843→      } else {
   844→        // Jina
   845→        const headers: Record<string, string> = { 'Accept': 'text/plain' }
   846→        if (nextKey.key) headers['Authorization'] = `Bearer ${nextKey.key}`
   847→        res = await fetch(`https://r.jina.ai/${param}`, {
   848→          method: 'GET',
   849→          headers,
   850→          signal: AbortSignal.timeout(30000),
   851→        })
   852→      }
   853→
   854→      if (res.ok) {
   855→        pool.reportResult(nextKey.index, true, res.status)
   856→        console.log(`[ToolExecutor] ${service} key ${nextKey.index + 1} succeeded on retry`)
   857→
   858→        if (service === 'tavily') {
   859→          const data = await res.json()
   860→          return {
   861→            answer: data.answer,
   862→            results: (data.results || []).map((r: Record<string, unknown>) => ({
   863→              title: r.title, url: r.url, content: r.content, score: r.score,
   864→            })),
   865→            source: `tavily-key${nextKey.index + 1}`,
   866→          }
   867→        } else if (service === 'serper') {
   868→          const data = await res.json()
   869→          return {
   870→            organic: (data.organic || []).map((r: Record<string, unknown>) => ({
   871→              title: r.title, link: r.link, snippet: r.snippet,
   872→            })),
   873→            knowledgeGraph: data.knowledgeGraph || null,
   874→            source: `serper-key${nextKey.index + 1}`,
   875→          }
   876→        } else {
   877→          // Jina
   878→          const text = await res.text()
   879→          const maxChars = 30000
   880→          const truncated = text.length > maxChars
   881→          const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
   882→          return {
   883→            url: param, content, length: text.length, truncated,
   884→            source: `jina-key${nextKey.index + 1}`,
   885→          }
   886→        }
   887→      }
   888→
   889→      // This key also failed — report and try next
   890→      const errorBody = await res.text().catch(() => '')
   891→      pool.reportResult(nextKey.index, false, res.status, `HTTP ${res.status}: ${errorBody.slice(0, 200)}`)
   892→      console.warn(`[ToolExecutor] ${service} key ${nextKey.index + 1} also failed: ${res.status}`)
   893→    } catch (err) {
   894→      const errMsg = err instanceof Error ? err.message : String(err)
   895→      pool.reportResult(nextKey.index, false, undefined, errMsg)
   896→      console.warn(`[ToolExecutor] ${service} key ${nextKey.index + 1} exception: ${errMsg}`)
   897→    }
   898→  }
   899→
   900→  return null // All keys failed
   901→}

  1089→ * Fallback: z-ai-web-dev-sdk page_reader function (Clawhub Market skill)
  1090→ */
  1091→async function executeJina(args: Record<string, unknown>): Promise<unknown> {
  1092→  const url = args.url as string
  1093→  if (!url) return { error: 'URL is required' }
  1094→
  1095→  // Validate URL
  1096→  try {
  1097→    new URL(url)
  1098→  } catch {
  1099→    return { error: `Invalid URL: ${url}` }
  1100→  }
  1101→
  1102→  // Primary: Direct Jina API with key pool rotation
  1103→  if (jinaKeyPool.hasKeys()) {
  1104→    const keySelection = jinaKeyPool.getNextKey()
  1105→    if (keySelection) {
  1106→      try {
  1107→        const headers: Record<string, string> = {
  1108→          'Accept': 'text/plain',
  1109→        }
  1110→        if (keySelection.key) {
  1111→          headers['Authorization'] = `Bearer ${keySelection.key}`
  1112→        }
  1113→
  1114→        const res = await fetch(`https://r.jina.ai/${url}`, {
  1115→          method: 'GET',
  1116→          headers,
  1117→          signal: AbortSignal.timeout(30000),
  1118→        })
  1119→
  1120→        if (res.ok) {
  1121→          jinaKeyPool.reportResult(keySelection.index, true, res.status)
  1122→          const text = await res.text()
  1123→
  1124→          // Limit output to prevent overwhelming LLM context
  1125→          const maxChars = 30000
  1126→          const truncated = text.length > maxChars
  1127→          const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
  1128→
  1129→          return {
  1130→            url,
  1131→            content,
  1132→            length: text.length,
  1133→            truncated,
  1134→            source: `jina-key${keySelection.index + 1}`,
  1135→          }
  1136→        }
  1137→
  1138→        // API returned error status — report to pool and try next key
  1139→        const errorBody = await res.text().catch(() => '')
  1140→        jinaKeyPool.reportResult(keySelection.index, false, res.status, `HTTP ${res.status}: ${errorBody.slice(0, 200)}`)
  1141→        console.warn(`[ToolExecutor] Jina key ${keySelection.index + 1} error: ${res.status}`)
  1142→
  1143→        // Try remaining keys before falling back to SDK
  1144→        const retryResult = await retryWithNextKey('jina', url)
  1145→        if (retryResult) return retryResult
  1146→      } catch (err) {
  1147→        const errMsg = err instanceof Error ? err.message : String(err)
  1148→        jinaKeyPool.reportResult(keySelection.index, false, undefined, errMsg)
  1149→        console.warn(`[ToolExecutor] Jina key ${keySelection.index + 1} failed: ${errMsg}`)
  1150→
  1151→        // Try remaining keys before falling back to SDK
  1152→        const retryResult = await retryWithNextKey('jina', url)
  1153→        if (retryResult) return retryResult
  1154→      }
  1155→    }
  1156→  }
  1157→
  1158→  // Fallback: z-ai-web-dev-sdk page_reader
  1159→  try {
  1160→    const zai = await getZAI()
  1161→    if (zai) {
  1162→      const result = await zai.functions.invoke('page_reader', { url })
  1163→      if (result?.data?.html) {
  1164→        // Convert HTML to clean text
  1165→        const text = result.data.html
  1166→          .replace(/<script[^>]*>[\s\S]*?<\/script>/gi, '')
  1167→          .replace(/<style[^>]*>[\s\S]*?<\/style>/gi, '')
  1168→          .replace(/<[^>]*>/g, ' ')
  1169→          .replace(/&nbsp;/g, ' ')
  1170→          .replace(/&amp;/g, '&')
  1171→          .replace(/&lt;/g, '<')
  1172→          .replace(/&gt;/g, '>')
  1173→          .replace(/\s+/g, ' ')
  1174→          .trim()
  1175→
  1176→        const maxChars = 30000
  1177→        const truncated = text.length > maxChars
  1178→        const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
  1179→
  1180→        return {
  1181→          url,
  1182→          title: result.data.title || null,
  1183→          content,
  1184→          length: text.length,
  1185→          truncated,
  1186→          source: 'z-ai-sdk-page_reader',
  1187→        }
  1188→      }
  1189→    }
  1190→  } catch (err) {
  1191→    console.warn(`[ToolExecutor] z-ai-sdk page_reader fallback failed: ${err instanceof Error ? err.message : String(err)}`)
  1192→  }
  1193→
  1194→  return {
  1195→    error: 'Jina reader failed and no fallback available.',
  1196→    hint: !jinaKeyPool.hasKeys() ? 'Add JINA_API_KEY_1..2 to .env' : `All ${jinaKeyPool.getTotalCount()} keys failed (${jinaKeyPool.getAvailableCount()} available)`,
  1197→    poolStatus: jinaKeyPool.getSummary(),
  1198→  }
  1199→}

  1265→// ==================== REACT LOOP HELPER ====================
  1266→
  1267→/**
  1268→ * Process a single LLM response — check for tool_calls and execute them.
  1269→ * Returns array of tool execution results for feeding back to LLM.
  1270→ *
  1271→ * This is the core of the ReAct loop:
  1272→ * 1. LLM returns response with tool_calls
  1273→ * 2. We execute each tool_call
  1274→ * 3. Results are fed back as tool messages
  1275→ * 4. LLM generates next response (may have more tool_calls or final text)
  1276→ */
  1277→export async function processToolCalls(
  1278→  toolCalls: Array<{ id: string; function: { name: string; arguments: string } }>,
  1279→): Promise<Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }>> {
  1280→  const results: Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }> = []
  1281→
  1282→  for (const tc of toolCalls) {
  1283→    let args: Record<string, unknown> = {}
  1284→    try {
  1285→      args = JSON.parse(tc.function.arguments)
  1286→    } catch {
  1287→      args = { _parseError: tc.function.arguments }
  1288→    }
  1289→
  1290→    console.log(`[ToolExecutor] Executing: ${tc.function.name}(${JSON.stringify(args).slice(0, 200)})`)
  1291→
  1292→    const result = await executeTool(tc.function.name, args)
  1293→
  1294→    results.push({
  1295→      toolCallId: tc.id,
  1296→      toolName: tc.function.name,
  1297→      result,
  1298→    })
  1299→
  1300→    console.log(`[ToolExecutor] Result: ${tc.function.name} → ${result.success ? 'OK' : 'FAIL'} (${result.duration}ms)`)
  1301→  }
  1302→
  1303→  return results
  1304→}
  1305→
  1306→/**
  1307→ * Format tool results into messages for the LLM.
  1308→ * Each tool result becomes a "tool" role message.
  1309→ * Note: Duration is excluded from LLM messages to save tokens — LLM doesn't need timing info.
  1310→ */
  1311→export function formatToolResultsAsMessages(
  1312→  results: Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }>,
  1313→): Array<{ role: 'tool'; tool_call_id: string; content: string }> {
  1314→  return results.map(r => ({
  1315→    role: 'tool' as const,
  1316→    tool_call_id: r.toolCallId,
  1317→    content: JSON.stringify({
  1318→      tool: r.toolName,
  1319→      success: r.result.success,
  1320→      result: r.result.result,
  1321→    }),
  1322→  }))
  1323→}
