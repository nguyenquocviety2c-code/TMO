    64→      )
    65→      return null

    67→  })()
    68→

   269→// ==================== TOOL EXECUTION ====================
   270→
   271→export interface ToolExecutionResult {
   272→  success: boolean
   273→  result: unknown
   274→  duration: number
   275→}
   276→
   277→/**
   278→ * Execute a tool call from LLM.
   279→ * Called in the ReAct loop when LLM returns a tool_call.
   280→ *
   281→ * @param toolName - Name of the tool to execute
   282→ * @param args - Arguments from LLM's tool_call
   283→ * @returns Execution result with success status
   284→ */
   285→export async function executeTool(
   286→  toolName: string,
   287→  args: Record<string, unknown>
   288→): Promise<ToolExecutionResult> {
   289→  const startTime = Date.now()
   290→
   291→  try {
   292→    let result: unknown
   293→
   294→    switch (toolName) {
   295→      case 'opencode':
   296→        result = await executeOpenCode(args)
   297→        break
   298→      case 'knowledge_search':
   299→        result = await executeKnowledgeSearch(args)
   300→        break
   301→      case 'knowledge_graph':
   302→        result = await executeKnowledgeGraph(args)
   303→        break
   304→      case 'knowledge_write':
   305→        result = await executeKnowledgeWrite(args)
   306→        break
   307→      case 'tavily':
   308→        result = await executeTavily(args)
   309→        break
   310→      case 'serper':
   311→        result = await executeSerper(args)
   312→        break
   313→      case 'jina':
   314→        result = await executeJina(args)
   315→        break
   316→      default:
   317→        return {
   318→          success: false,
   319→          result: `Unknown tool: ${toolName}`,
   320→          duration: Date.now() - startTime,
   321→        }
   322→    }
   323→
   324→    return {
   325→      success: true,
   326→      result,
   327→      duration: Date.now() - startTime,
   328→    }
   329→  } catch (err) {
   330→    const errorMsg = err instanceof Error ? err.message : String(err)
   331→    console.error(`[ToolExecutor] Error executing ${toolName}:`, errorMsg)
   332→    return {
   333→      success: false,
   334→      result: `Tool error: ${errorMsg}`,
   335→      duration: Date.now() - startTime,
   336→    }
   337→  }
   338→}

   343→ * OpenCode — Code operations (read, write, bash, mkdir)

   345→ * Uses the OpenCode API server (configured via OPENCODE_SERVER_URL env var) if available,
   346→ * otherwise falls back to direct file system operations via the project API.

   348→async function executeOpenCode(args: Record<string, unknown>): Promise<unknown> {
   349→  const action = args.action as string
   350→  const path = args.path as string | undefined
   351→  const content = args.content as string | undefined
   352→  const command = args.command as string | undefined
   353→
   354→  // Try OpenCode server first
   355→  try {
   356→    const opencodeUrl = process.env.OPENCODE_SERVER_URL || 'http://localhost:18790'
   357→    const res = await fetch(`${opencodeUrl}/api/execute`, {
   358→      method: 'POST',
   359→      headers: { 'Content-Type': 'application/json' },
   360→      body: JSON.stringify({ action, path, content, command }),
   361→      signal: AbortSignal.timeout(30000),
   362→    })
   363→
   364→    if (res.ok) {
   365→      const data = await res.json()
   366→      return data
   367→    }
   368→  } catch {
   369→    // OpenCode server not available, fall through to direct implementation
   370→  }
   371→
   372→  // Fallback: Use the project's own API for file operations
   373→  switch (action) {
   374→    case 'read':
   375→      return await executeFileRead(path || '')
   376→    case 'write':
   377→      return await executeFileWrite(path || '', content || '')
   378→    case 'bash':
   379→      return await executeBash(command || '')
   380→    case 'mkdir':
   381→      return { success: true, message: `mkdir not needed — write will create parent dirs` }
   382→    default:
   383→      return { success: false, error: `Unknown opencode action: ${action}` }
   384→  }
   385→}

   387→/**
   388→ * Read a file from the project directory.
   389→ * Security: Only allows reading files within the project root.
   390→ * Uses path.resolve + startsWith check to prevent path traversal.
   391→ */
   392→async function executeFileRead(filePath: string): Promise<{ success: boolean; content?: string; error?: string }> {
   393→  if (!filePath) return { success: false, error: 'Path is required' }
   394→
   395→  try {
   396→    const pathModule = await import('path')
   397→    const projectRoot = process.cwd()
   398→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   399→
   400→    // Security: Prevent path traversal — resolved path must be within project root
   401→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   402→      return { success: false, error: `Access denied: path escapes project root` }
   403→    }
   404→
   405→    const fs = await import('fs/promises')
   406→    const content = await fs.readFile(resolvedPath, 'utf-8')
   407→
   408→    // Limit output to prevent overwhelming LLM context
   409→    const maxChars = 50000
   410→    const truncated = content.length > maxChars
   411→    const output = truncated ? content.slice(0, maxChars) + '\n\n... [TRUNCATED — file too large]' : content
   412→
   413→    return { success: true, content: output }
   414→  } catch (err) {
   415→    return {
   416→      success: false,
   417→      error: `Failed to read ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   418→    }
   419→  }
   420→}
   421→
   422→/**
   423→ * Write a file to the project directory.
   424→ * Security: Only allows writing within the project root. Uses path.resolve to prevent traversal.
   425→ * Auto-creates parent directories.
   426→ */
   427→async function executeFileWrite(filePath: string, content: string): Promise<{ success: boolean; message: string; error?: string }> {
   428→  if (!filePath) return { success: false, message: '', error: 'Path is required' }
   429→  if (!content) return { success: false, message: '', error: 'Content is required' }
   430→
   431→  try {
   432→    const pathModule = await import('path')
   433→    const projectRoot = process.cwd()
   434→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   435→
   436→    // Security: Prevent path traversal — resolved path must be within project root
   437→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   438→      return { success: false, message: '', error: 'Access denied: path escapes project root' }
   439→    }
   440→
   441→    const fs = await import('fs/promises')
   442→
   443→    // Auto-create parent directories
   444→    const dir = pathModule.dirname(resolvedPath)
   445→    await fs.mkdir(dir, { recursive: true })
   446→
   447→    // Write file
   448→    await fs.writeFile(resolvedPath, content, 'utf-8')
   449→
   450→    return { success: true, message: `File written: ${filePath} (${content.length} chars)` }
   451→  } catch (err) {
   452→    return {
   453→      success: false,
   454→      message: '',
   455→      error: `Failed to write ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   456→    }
   457→  }
   458→}

   460→/**
   461→ * Execute a bash command safely.
   462→ * Security: Only allows pre-approved commands.
   463→ */
   464→async function executeBash(command: string): Promise<{ success: boolean; output?: string; error?: string }> {
   465→  if (!command) return { success: false, error: 'Command is required' }
   466→
   467→  // Security: Whitelist of allowed commands
   468→  // NOTE: 'node' is excluded because `node -e "..."` can execute arbitrary code.
   469→  //       'cat'/'head'/'tail'/'grep' are excluded — use opencode read instead.
   470→  //       'npx'/'bunx' are excluded — can download+execute arbitrary npm packages.
   471→  const allowedPrefixes = [
   472→    'npm run',
   473→    'npm test',
   474→    'npm lint',
   475→    'bun run',
   476→    'bun test',
   477→    'bun lint',
   478→    'tsc',
   479→    'eslint',
   480→    'prettier',
   481→    'echo',
   482→    'ls',
   483→    'wc',
   484→    'rg',       // ripgrep — search only
   485→    'which',
   486→    'pwd',
   487→    'prisma',
   488→    'git status',
   489→    'git diff',
   490→    'git log',
   491→  ]
   492→
   493→  // Block dangerous flags that could allow code execution or file access
   494→  const dangerousFlags = ['-e ', '--eval ', '-r ', '--require ', '-p ', '--print ']
   495→  const normalizedCmd = command.trim()
   496→  const isAllowed = allowedPrefixes.some(prefix => normalizedCmd.startsWith(prefix))
   497→  const usesDangerousFlag = dangerousFlags.some(flag => normalizedCmd.includes(flag))
   498→
   499→  // Security: Block shell metacharacters that enable command chaining/injection
   500→  // These allow bypassing the whitelist by chaining additional commands:
   501→  //   "npm run build; rm -rf /"   →  ;  chains commands
   502→  //   "git status && cat /etc/p"  →  && chains on success
   503→  //   "echo hi | bash"            →  |  pipes output
   504→  //   "tsc || rm -rf /"           →  || chains on failure
   505→  //   "echo $(cat /etc/passwd)"   →  $() command substitution
   506→  //   "echo `cat /etc/passwd`"    →  backtick command substitution
   507→  //   "echo hi > /etc/passwd"     →  >  redirect write
   508→  //   "echo hi >> /etc/passwd"    →  >> redirect append
   509→  const shellMetacharacters = /(;|&&|\|\||\||`|\$\(|\$\{|>|>>)/
   510→  const usesShellMetacharacters = shellMetacharacters.test(normalizedCmd)
   511→
   512→  if (!isAllowed || usesDangerousFlag || usesShellMetacharacters) {
   513→    return {
   514→      success: false,
   515→      error: `Command not allowed: "${normalizedCmd.slice(0, 100)}". ${!isAllowed ? 'Prefix not in whitelist.' : ''} ${usesDangerousFlag ? 'Dangerous flag detected.' : ''} ${usesShellMetacharacters ? 'Shell metacharacter (;, &&, ||, |, `, $(), >, >>) detected — command chaining is forbidden.' : ''}`,
   516→    }
   517→  }
   518→
   519→  try {
   520→    const { exec } = await import('child_process')
   521→    const { promisify } = await import('util')
   522→    const execAsync = promisify(exec)
   523→
   524→    const { stdout, stderr } = await execAsync(normalizedCmd, {
   525→      cwd: process.cwd(),
   526→      timeout: 30000,
   527→      maxBuffer: 1024 * 1024, // 1MB
   528→    })
   529→
   530→    const output = [
   531→      stdout ? `STDOUT:\n${stdout}` : '',
   532→      stderr ? `STDERR:\n${stderr}` : '',
   533→    ].filter(Boolean).join('\n')
   534→
   535→    return { success: true, output: output || '(no output)' }
   536→  } catch (err) {
   537→    const error = err as { stdout?: string; stderr?: string; message?: string }
   538→    const output = [
   539→      error.stdout ? `STDOUT:\n${error.stdout}` : '',
   540→      error.stderr ? `STDERR:\n${error.stderr}` : '',
   541→    ].filter(Boolean).join('\n')
   542→
   543→    return {
   544→      success: false,
   545→      output: output || undefined,
   546→      error: `Command failed: ${error.message}`,
   547→    }
   548→  }
   549→}

   551→/**
   552→ * Knowledge Search — Semantic search in KB (Qdrant + Neo4j)
   553→ * Uses knowledge-bridge which already handles embedding + search.
   554→ */
   555→async function executeKnowledgeSearch(args: Record<string, unknown>): Promise<unknown> {
   556→  const query = args.query as string
   557→  const topK = Math.min((args.topK as number) || 5, 20)
   558→
   559→  if (!query) return { error: 'Query is required' }
   560→
   561→  try {
   562→    const { agentKnowledgeSearch } = await import('@/lib/knowledge-bridge')
   563→    const result = await agentKnowledgeSearch(query, { topK })
   564→
   565→    return {
   566→      results: result.results.slice(0, topK),
   567→      answer: result.answer,
   568→      totalFound: result.results.length,
   569→    }
   570→  } catch (err) {
   571→    return { error: `Knowledge search failed: ${err instanceof Error ? err.message : String(err)}` }
   572→  }
   573→}

   575→/**
   576→ * Knowledge Graph — Neo4j Cypher query (read-only)
   577→ * Enforces read-only at client level BEFORE sending to knowledge-bridge.
   578→ * Only MATCH, RETURN, WHERE, WITH, ORDER BY, LIMIT, SKIP, DISTINCT are allowed.
   579→ */
   580→async function executeKnowledgeGraph(args: Record<string, unknown>): Promise<unknown> {
   581→  const query = args.query as string
   582→
   583→  if (!query) return { error: 'Cypher query is required' }
   584→
   585→  // Security: Enforce read-only at client level
   586→  // Block any write/mutation Cypher keywords (case-insensitive)
   587→  const writeKeywords = /\b(CREATE|DELETE|SET|MERGE|DROP|REMOVE|DETACH|CALL\s*\{.*\}\s*IN\s+TRANSACTION)\b/i
   588→  if (writeKeywords.test(query)) {
   589→    return { error: 'Only read-only Cypher queries (MATCH) are allowed. Write operations (CREATE, DELETE, SET, MERGE, DROP, REMOVE) are forbidden.' }
   590→  }
   591→
   592→  try {
   593→    const { agentGraphQuery } = await import('@/lib/knowledge-bridge')
   594→    const result = await agentGraphQuery(query)
   595→
   596→    return {
   597→      records: result.records,
   598→      count: result.count,
   599→      error: result.error,
   600→    }
   601→  } catch (err) {
   602→    return { error: `Graph query failed: ${err instanceof Error ? err.message : String(err)}` }
   603→  }
   604→}

   606→/**
   607→ * Knowledge Write — Write entity/relationship to KB
   608→ * Uses knowledge-bridge which writes to SQLite buffer (syncs to Neo4j later).
   609→ * Only G3 (CATALYST) has this permission.
   610→ */
   611→async function executeKnowledgeWrite(args: Record<string, unknown>): Promise<unknown> {
   612→  const entityName = args.entityName as string
   613→  const entityType = args.entityType as string
   614→  const description = args.description as string
   615→  const category = args.category as string
   616→  const relationshipType = args.relationshipType as string | undefined
   617→  const targetEntityName = args.targetEntityName as string | undefined
   618→
   619→  if (!entityName || !entityType || !description || !category) {
   620→    return { error: 'entityName, entityType, description, and category are required' }
   621→  }
   622→
   623→  try {
   624→    const { agentWriteEntity, agentWriteRelationship } = await import('@/lib/knowledge-bridge')
   625→
   626→    // Write entity

   628→      entityName,
   629→      entityType,
   630→      description,
   631→      domain: category,
   632→    })
   633→
   634→    // Write relationship if specified
   635→    let relationship: { id: string; type: string } | null = null
   636→    if (relationshipType && targetEntityName) {
   637→      const rel = await agentWriteRelationship({
   638→        sourceEntityName: entityName,
   639→        targetEntityName,
   640→        relationshipType,
   641→        description: `${entityName} ${relationshipType} ${targetEntityName}`,
   642→      })
   643→      relationship = { id: rel.id, type: rel.type }
   644→    }
   645→
   646→    return {
   647→      entity: { id: entity.id, name: entity.entityName },
   648→      relationship,
   649→      message: `Đã ghi entity "${entityName}" vào KB${relationship ? ` + relationship "${relationshipType}" → ${targetEntityName}` : ''}`,
   650→    }
   651→  } catch (err) {
   652→    return { error: `Knowledge write failed: ${err instanceof Error ? err.message : String(err)}` }

   658→/**
   659→ * Retry a service call with the next available key from the pool.
   660→ * Called when the first key fails — tries remaining keys before falling back to SDK.
   661→ *
   662→ * @param service - 'tavily' | 'serper' | 'jina'
   663→ * @param param - The query (for search) or URL (for reader)
   664→ * @returns Result if a key succeeds, null if all keys fail
   665→ */
   666→async function retryWithNextKey(
   667→  service: 'tavily' | 'serper' | 'jina',
   668→  param: string
   669→): Promise<unknown | null> {
   670→  const pool = service === 'tavily' ? tavilyKeyPool : service === 'serper' ? serperKeyPool : jinaKeyPool
   671→  const maxRetries = pool.getTotalCount() - 1 // Already tried 1 key
   672→
   673→  for (let i = 0; i < maxRetries; i++) {
   674→    const nextKey = pool.getNextKey()
   675→    if (!nextKey || nextKey.isLastResort) break // No more healthy keys
   676→
   677→    console.log(`[ToolExecutor] ${service} retrying with key ${nextKey.index + 1}...`)
   678→
   679→    try {
   680→      let res: Response
   681→
   682→      if (service === 'tavily') {
   683→        res = await fetch('https://api.tavily.com/search', {
   684→          method: 'POST',
   685→          headers: { 'Content-Type': 'application/json' },
   686→          body: JSON.stringify({
   687→            api_key: nextKey.key,
   688→            query: param,
   689→            search_depth: 'advanced',
   690→            include_answer: true,
   691→            max_results: 5,
   692→          }),
   693→          signal: AbortSignal.timeout(30000),
   694→        })
   695→      } else if (service === 'serper') {
   696→        res = await fetch('https://google.serper.dev/search', {
   697→          method: 'POST',
   698→          headers: {
   699→            'X-API-KEY': nextKey.key,
   700→            'Content-Type': 'application/json',
   701→          },
   702→          body: JSON.stringify({ q: param, num: 5 }),
   703→          signal: AbortSignal.timeout(15000),
   704→        })
   705→      } else {
   706→        // Jina
   707→        const headers: Record<string, string> = { 'Accept': 'text/plain' }
   708→        if (nextKey.key) headers['Authorization'] = `Bearer ${nextKey.key}`
   709→        res = await fetch(`https://r.jina.ai/${param}`, {
   710→          method: 'GET',
   711→          headers,
   712→          signal: AbortSignal.timeout(30000),
   713→        })
   714→      }
   715→
   716→      if (res.ok) {
   717→        pool.reportResult(nextKey.index, true, res.status)
   718→        console.log(`[ToolExecutor] ${service} key ${nextKey.index + 1} succeeded on retry`)
   719→
   720→        if (service === 'tavily') {
   721→          const data = await res.json()
   722→          return {
   723→            answer: data.answer,
   724→            results: (data.results || []).map((r: Record<string, unknown>) => ({
   725→              title: r.title, url: r.url, content: r.content, score: r.score,
   726→            })),
   727→            source: `tavily-key${nextKey.index + 1}`,
   728→          }
   729→        } else if (service === 'serper') {
   730→          const data = await res.json()
   731→          return {
   732→            organic: (data.organic || []).map((r: Record<string, unknown>) => ({
   733→              title: r.title, link: r.link, snippet: r.snippet,
   734→            })),
   735→            knowledgeGraph: data.knowledgeGraph || null,
   736→            source: `serper-key${nextKey.index + 1}`,
   737→          }
   738→        } else {
   739→          // Jina
   740→          const text = await res.text()
   741→          const maxChars = 30000
   742→          const truncated = text.length > maxChars
   743→          const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
   744→          return {
   745→            url: param, content, length: text.length, truncated,
   746→            source: `jina-key${nextKey.index + 1}`,
   747→          }
   748→        }
   749→      }
   750→
   751→      // This key also failed — report and try next
   752→      const errorBody = await res.text().catch(() => '')
   753→      pool.reportResult(nextKey.index, false, res.status, `HTTP ${res.status}: ${errorBody.slice(0, 200)}`)
   754→      console.warn(`[ToolExecutor] ${service} key ${nextKey.index + 1} also failed: ${res.status}`)
   755→    } catch (err) {
   756→      const errMsg = err instanceof Error ? err.message : String(err)
   757→      pool.reportResult(nextKey.index, false, undefined, errMsg)
   758→      console.warn(`[ToolExecutor] ${service} key ${nextKey.index + 1} exception: ${errMsg}`)
   759→    }
   760→  }
   761→
   762→  return null // All keys failed
   763→}

  1129→/**
  1130→ * Process a single LLM response — check for tool_calls and execute them.
  1131→ * Returns array of tool execution results for feeding back to LLM.
  1132→ *
  1133→ * This is the core of the ReAct loop:
  1134→ * 1. LLM returns response with tool_calls
  1135→ * 2. We execute each tool_call
  1136→ * 3. Results are fed back as tool messages
  1137→ * 4. LLM generates next response (may have more tool_calls or final text)
  1138→ */
  1139→export async function processToolCalls(
  1140→  toolCalls: Array<{ id: string; function: { name: string; arguments: string } }>,
  1141→): Promise<Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }>> {
  1142→  const results: Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }> = []
  1143→
  1144→  for (const tc of toolCalls) {
  1145→    let args: Record<string, unknown> = {}
  1146→    try {
  1147→      args = JSON.parse(tc.function.arguments)
  1148→    } catch {
  1149→      args = { _parseError: tc.function.arguments }
  1150→    }
  1151→
  1152→    console.log(`[ToolExecutor] Executing: ${tc.function.name}(${JSON.stringify(args).slice(0, 200)})`)
  1153→
  1154→    const result = await executeTool(tc.function.name, args)
  1155→
  1156→    results.push({
  1157→      toolCallId: tc.id,
  1158→      toolName: tc.function.name,
  1159→      result,
  1160→    })
  1161→
  1162→    console.log(`[ToolExecutor] Result: ${tc.function.name} → ${result.success ? 'OK' : 'FAIL'} (${result.duration}ms)`)
  1163→  }
  1164→
  1165→  return results
  1166→}

  1168→/**
  1169→ * Format tool results into messages for the LLM.
  1170→ * Each tool result becomes a "tool" role message.
  1171→ * Note: Duration is excluded from LLM messages to save tokens — LLM doesn't need timing info.
  1172→ */
  1173→export function formatToolResultsAsMessages(
  1174→  results: Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }>,
  1175→): Array<{ role: 'tool'; tool_call_id: string; content: string }> {
  1176→  return results.map(r => ({
  1177→    role: 'tool' as const,
  1178→    tool_call_id: r.toolCallId,
  1179→    content: JSON.stringify({
  1180→      tool: r.toolName,
  1181→      success: r.result.success,
  1182→      result: r.result.result,
  1183→    }),
  1184→  }))
  1185→}
