     1→/**
     2→ * Code Team — Tool Execution Layer
     3→ *
     4→ * Central place to execute all tools that agents can call via LLM function calling.
     5→ * Supports ReAct loop: LLM returns tool_call → execute → feed result back → repeat.
     6→ *
     7→ * Tools available (per-agent permissions from agents.ts):
     8→ *   - opencode:         Code operations (read/write/bash) — via internal API
     9→ *   - knowledge_search:  Semantic search in KB (Qdrant) — via knowledge-bridge
    10→ *   - knowledge_graph:   Graph query (Neo4j Cypher) — via knowledge-bridge
    11→ *   - knowledge_write:   Write entity/relationship to KB — via knowledge-bridge
    12→ *   - tavily:           AI-optimized web search — via Tavily API + z-ai-sdk fallback
    13→ *   - serper:           Google search — via Serper API + z-ai-sdk fallback
    14→ *   - jina:             Web page reader — via Jina API + z-ai-sdk fallback
    15→ *
    16→ * Architecture:
    17→ *   Workflow Engine → callLLMForAgent() with tool definitions
    18→ *   → LLM returns tool_call → executeTool() → feed result back
    19→ *   → Repeat until LLM outputs text (no more tool_calls)
    20→ *
    21→ * Phase 5 Enhancement:
    22→ *   - Tavily/Serper/Jina use direct API keys (from .env)
    23→ *   - z-ai-web-dev-sdk provides automatic fallback when API keys fail

   296→/**
   297→ * Execute a tool via Gateway HTTP /tools/invoke endpoint.
   298→ * Only works for HTTP-allowed tools (web_search, web_fetch, etc.).
   299→ * File operations and exec require session context and are NOT available.
   300→ */
   301→async function executeGatewayTool(name: string, args: Record<string, unknown>): Promise<unknown> {
   302→  if (!GATEWAY_HTTP_ALLOWED_TOOLS.has(name)) {
   303→    throw new Error(`Tool "${name}" requires Gateway session context — not available via HTTP proxy. Use /v1/chat/completions instead.`)
   304→  }
   305→
   306→  const res = await fetch(`${GATEWAY_URL}/tools/invoke`, {
   307→    method: 'POST',
   308→    headers: { 'Content-Type': 'application/json' },
   309→    body: JSON.stringify({ name, args }),
   310→    signal: AbortSignal.timeout(30000),
   311→  })
   312→
   313→  if (!res.ok) {
   314→    const errorText = await res.text().catch(() => 'Unknown error')
   315→    throw new Error(`Gateway tool error ${res.status}: ${errorText.slice(0, 200)}`)
   316→  }
   317→
   318→  const data = await res.json()
   319→  if (!data.ok && data.error) {
   320→    throw new Error(`Tool error: ${data.error?.message || data.error || 'Unknown'}`)
   321→  }
   322→
   323→  return data.result ?? data
   324→}

   326→// ==================== Z-AI-SDK LAZY INITIALIZATION ====================
   327→//
   328→// ⚠️  IMPORTANT: z-ai-web-dev-sdk is an OPTIONAL dependency!
   329→//
   330→// This SDK ONLY works inside the z.ai sandbox environment where backend
   331→// services (web_search, page_reader) are available. If you clone this
   332→// project and run locally, the SDK will not initialize — this is expected
   333→// and NOT an error. Direct API keys (TAVILY_API_KEY, SERPER_API_KEY,
   334→// JINA_API_KEY) in .env will be used instead.
   335→//
   336→// Fallback chain: Direct API key → z-ai-sdk → error with helpful hint
   337→//
   338→
   339→let zaiInstance: any = null
   340→let zaiInitPromise: Promise<any> | null = null
   341→let zaiAvailable: boolean | null = null // null = not yet checked
   342→
   343→async function getZAI(): Promise<any> {
   344→  if (zaiInstance) return zaiInstance
   345→  if (zaiInitPromise) return zaiInitPromise
   346→
   347→  zaiInitPromise = (async () => {
   348→    try {
   349→      const ZAI = (await import('z-ai-web-dev-sdk')).default
   350→      const instance = await ZAI.create()
   351→      console.log('[ToolExecutor] ✅ z-ai-web-dev-sdk initialized — web_search + page_reader available as fallback')
   352→      zaiInstance = instance
   353→      zaiAvailable = true
   354→      return instance
   355→    } catch (err) {
   356→      zaiAvailable = false
   357→      console.warn(
   358→        '[ToolExecutor] z-ai-web-dev-sdk not available (this is normal for local deployment — direct API keys will be used).',
   359→        `Detail: ${err instanceof Error ? err.message : String(err)}`
   360→      )
   361→      return null
   362→    }
   363→  })()
   364→
   365→  return zaiInitPromise
   366→}

   548→export function getToolDefinitionsForPosition(position: string): ToolDefinition[] {
   549→  const tools = getAgentTools(position)
   550→  return tools
   551→    .map(t => ALL_TOOL_DEFINITIONS[t])
   552→    .filter((def): def is ToolDefinition => def !== undefined)
   553→}

   567→export interface ToolExecutionResult {
   568→  success: boolean
   569→  result: unknown
   570→  duration: number
   571→}
   572→
   573→/**
   574→ * Execute a tool call from LLM.
   575→ * Called in the ReAct loop when LLM returns a tool_call.
   576→ *
   577→ * @param toolName - Name of the tool to execute
   578→ * @param args - Arguments from LLM's tool_call
   579→ * @returns Execution result with success status
   580→ */
   581→export async function executeTool(
   582→  toolName: string,
   583→  args: Record<string, unknown>
   584→): Promise<ToolExecutionResult> {
   585→  const startTime = Date.now()
   586→
   587→  try {
   588→    let result: unknown
   589→
   590→    switch (toolName) {
   591→      case 'opencode':
   592→        result = await executeOpenCode(args)
   593→        break
   594→      case 'knowledge_search':
   595→        result = await executeKnowledgeSearch(args)
   596→        break
   597→      case 'knowledge_graph':
   598→        result = await executeKnowledgeGraph(args)
   599→        break
   600→      case 'knowledge_write':
   601→        result = await executeKnowledgeWrite(args)
   602→        break
   603→      case 'tavily':
   604→        result = await executeTavily(args)
   605→        break
   606→      case 'serper':
   607→        result = await executeSerper(args)
   608→        break
   609→      case 'jina':
   610→        result = await executeJina(args)
   611→        break
   612→      default: {
   613→        // P0.5: Try Gateway proxy for tools not in local set
   614→        if (GATEWAY_HTTP_ALLOWED_TOOLS.has(toolName)) {
   615→          try {
   616→            const gwResult = await executeGatewayTool(toolName, args)
   617→            return { success: true, result: gwResult, duration: Date.now() - startTime }
   618→          } catch (gwErr) {
   619→            return {
   620→              success: false,
   621→              result: `Gateway tool "${toolName}" failed: ${gwErr instanceof Error ? gwErr.message : String(gwErr)}`,
   622→              duration: Date.now() - startTime,
   623→            }
   624→          }
   625→        }
   626→        return { success: false, result: `Unknown tool: ${toolName}. Available local tools: opencode, knowledge_search, knowledge_graph, knowledge_write, tavily, serper, jina. Gateway tools: ${Array.from(GATEWAY_HTTP_ALLOWED_TOOLS).join(', ')}`, duration: Date.now() - startTime }
   627→      }
   628→    }
   629→
   630→    return {
   631→      success: true,
   632→      result,
   633→      duration: Date.now() - startTime,
   634→    }
   635→  } catch (err) {
   636→    const errorMsg = err instanceof Error ? err.message : String(err)
   637→    console.error(`[ToolExecutor] Error executing ${toolName}:`, errorMsg)
   638→    return {
   639→      success: false,
   640→      result: `Tool error: ${errorMsg}`,
   641→      duration: Date.now() - startTime,
   642→    }
   643→  }

   693→/**
   694→ * Read a file from the project directory.
   695→ * Security: Only allows reading files within the project root.
   696→ * Uses path.resolve + startsWith check to prevent path traversal.
   697→ */
   698→async function executeFileRead(filePath: string): Promise<{ success: boolean; content?: string; error?: string }> {
   699→  if (!filePath) return { success: false, error: 'Path is required' }
   700→
   701→  try {
   702→    const pathModule = await import('path')
   703→    const projectRoot = process.cwd()
   704→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   705→
   706→    // Security: Prevent path traversal — resolved path must be within project root
   707→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   708→      return { success: false, error: `Access denied: path escapes project root` }
   709→    }
   710→
   711→    const fs = await import('fs/promises')
   712→    const content = await fs.readFile(resolvedPath, 'utf-8')
   713→
   714→    // Limit output to prevent overwhelming LLM context
   715→    const maxChars = 50000
   716→    const truncated = content.length > maxChars
   717→    const output = truncated ? content.slice(0, maxChars) + '\n\n... [TRUNCATED — file too large]' : content
   718→
   719→    return { success: true, content: output }
   720→  } catch (err) {
   721→    return {
   722→      success: false,
   723→      error: `Failed to read ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   724→    }
   725→  }
   726→}

   733→async function executeFileWrite(filePath: string, content: string): Promise<{ success: boolean; message: string; error?: string }> {
   734→  if (!filePath) return { success: false, message: '', error: 'Path is required' }
   735→  if (!content) return { success: false, message: '', error: 'Content is required' }
   736→
   737→  try {
   738→    const pathModule = await import('path')
   739→    const projectRoot = process.cwd()
   740→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   741→
   742→    // Security: Prevent path traversal — resolved path must be within project root

   760→      message: '',
   761→      error: `Failed to write ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   762→    }
   763→  }

   766→/**
   767→ * Execute a bash command safely.
   768→ * Security: Only allows pre-approved commands.
   769→ */
   770→async function executeBash(command: string): Promise<{ success: boolean; output?: string; error?: string }> {
   771→  if (!command) return { success: false, error: 'Command is required' }
   772→
   773→  // Security: Whitelist of allowed commands
   774→  // NOTE: 'node' is excluded because `node -e "..."` can execute arbitrary code.
   775→  //       'cat'/'head'/'tail'/'grep' are excluded — use opencode read instead.
   776→  //       'npx'/'bunx' are excluded — can download+execute arbitrary npm packages.
   777→  const allowedPrefixes = [
   778→    'npm run',
   779→    'npm test',
   780→    'npm lint',
   781→    'bun run',
   782→    'bun test',
   783→    'bun lint',
   784→    'tsc',
   785→    'eslint',
   786→    'prettier',
   787→    'echo',
   788→    'ls',
   789→    'wc',
   790→    'rg',       // ripgrep — search only
   791→    'which',
   792→    'pwd',
   793→    'prisma',
   794→    'git status',
   795→    'git diff',
   796→    'git log',
   797→  ]
   798→
   799→  // Block dangerous flags that could allow code execution or file access
   800→  const dangerousFlags = ['-e ', '--eval ', '-r ', '--require ', '-p ', '--print ']
   801→  const normalizedCmd = command.trim()
   802→  const isAllowed = allowedPrefixes.some(prefix => normalizedCmd.startsWith(prefix))
   803→  const usesDangerousFlag = dangerousFlags.some(flag => normalizedCmd.includes(flag))
   804→
   805→  // Security: Block shell metacharacters that enable command chaining/injection
   806→  // These allow bypassing the whitelist by chaining additional commands:
   807→  //   "npm run build; rm -rf /"   →  ;  chains commands
   808→  //   "git status && cat /etc/p"  →  && chains on success
   809→  //   "echo hi | bash"            →  |  pipes output
   810→  //   "tsc || rm -rf /"           →  || chains on failure
   811→  //   "echo $(cat /etc/passwd)"   →  $() command substitution
   812→  //   "echo `cat /etc/passwd`"    →  backtick command substitution
   813→  //   "echo hi > /etc/passwd"     →  >  redirect write
   814→  //   "echo hi >> /etc/passwd"    →  >> redirect append
   815→  const shellMetacharacters = /(;|&&|\|\||\||`|\$\(|\$\{|>|>>)/
   816→  const usesShellMetacharacters = shellMetacharacters.test(normalizedCmd)
   817→
   818→  if (!isAllowed || usesDangerousFlag || usesShellMetacharacters) {
   819→    return {
   820→      success: false,
   821→      error: `Command not allowed: "${normalizedCmd.slice(0, 100)}". ${!isAllowed ? 'Prefix not in whitelist.' : ''} ${usesDangerousFlag ? 'Dangerous flag detected.' : ''} ${usesShellMetacharacters ? 'Shell metacharacter (;, &&, ||, |, `, $(), >, >>) detected — command chaining is forbidden.' : ''}`,
   822→    }
   823→  }
   824→
   825→  try {
   826→    const { exec } = await import('child_process')
   827→    const { promisify } = await import('util')
   828→    const execAsync = promisify(exec)
   829→
   830→    const { stdout, stderr } = await execAsync(normalizedCmd, {
   831→      cwd: process.cwd(),
   832→      timeout: 30000,
   833→      maxBuffer: 1024 * 1024, // 1MB
   834→    })
   835→
   836→    const output = [
   837→      stdout ? `STDOUT:\n${stdout}` : '',
   838→      stderr ? `STDERR:\n${stderr}` : '',
   839→    ].filter(Boolean).join('\n')
   840→
   841→    return { success: true, output: output || '(no output)' }
   842→  } catch (err) {
   843→    const error = err as { stdout?: string; stderr?: string; message?: string }
   844→    const output = [
   845→      error.stdout ? `STDOUT:\n${error.stdout}` : '',
   846→      error.stderr ? `STDERR:\n${error.stderr}` : '',
   847→    ].filter(Boolean).join('\n')
   848→
   849→    return {
   850→      success: false,
   851→      output: output || undefined,
   852→      error: `Command failed: ${error.message}`,
   853→    }
   854→  }
   855→}

   964→/**
   965→ * Retry a service call with the next available key from the pool.
   966→ * Called when the first key fails — tries remaining keys before falling back to SDK.
   967→ *
   968→ * @param service - 'tavily' | 'serper' | 'jina'
   969→ * @param param - The query (for search) or URL (for reader)
   970→ * @returns Result if a key succeeds, null if all keys fail
   971→ */
   972→async function retryWithNextKey(
   973→  service: 'tavily' | 'serper' | 'jina',
   974→  param: string
   975→): Promise<unknown | null> {
   976→  const pool = service === 'tavily' ? tavilyKeyPool : service === 'serper' ? serperKeyPool : jinaKeyPool
   977→  const maxRetries = pool.getTotalCount() - 1 // Already tried 1 key
   978→
   979→  for (let i = 0; i < maxRetries; i++) {
   980→    const nextKey = pool.getNextKey()
   981→    if (!nextKey || nextKey.isLastResort) break // No more healthy keys
   982→
   983→    console.log(`[ToolExecutor] ${service} retrying with key ${nextKey.index + 1}...`)
   984→
   985→    try {
   986→      let res: Response
   987→
   988→      if (service === 'tavily') {
   989→        res = await fetch('https://api.tavily.com/search', {
   990→          method: 'POST',
   991→          headers: { 'Content-Type': 'application/json' },
   992→          body: JSON.stringify({
   993→            api_key: nextKey.key,
   994→            query: param,
   995→            search_depth: 'advanced',
   996→            include_answer: true,
   997→            max_results: 5,
   998→          }),
   999→          signal: AbortSignal.timeout(30000),
  1000→        })
  1001→      } else if (service === 'serper') {
  1002→        res = await fetch('https://google.serper.dev/search', {
  1003→          method: 'POST',
  1004→          headers: {
  1005→            'X-API-KEY': nextKey.key,
  1006→            'Content-Type': 'application/json',
  1007→          },
  1008→          body: JSON.stringify({ q: param, num: 5 }),
  1009→          signal: AbortSignal.timeout(15000),
  1010→        })
  1011→      } else {
  1012→        // Jina
  1013→        const headers: Record<string, string> = { 'Accept': 'text/plain' }
  1014→        if (nextKey.key) headers['Authorization'] = `Bearer ${nextKey.key}`
  1015→        res = await fetch(`https://r.jina.ai/${param}`, {
  1016→          method: 'GET',
  1017→          headers,
  1018→          signal: AbortSignal.timeout(30000),
  1019→        })
  1020→      }
  1021→
  1022→      if (res.ok) {
  1023→        pool.reportResult(nextKey.index, true, res.status)
  1024→        console.log(`[ToolExecutor] ${service} key ${nextKey.index + 1} succeeded on retry`)
  1025→
  1026→        if (service === 'tavily') {
  1027→          const data = await res.json()
  1028→          return {
  1029→            answer: data.answer,
  1030→            results: (data.results || []).map((r: Record<string, unknown>) => ({
  1031→              title: r.title, url: r.url, content: r.content, score: r.score,
  1032→            })),
  1033→            source: `tavily-key${nextKey.index + 1}`,
  1034→          }
  1035→        } else if (service === 'serper') {
  1036→          const data = await res.json()
  1037→          return {
  1038→            organic: (data.organic || []).map((r: Record<string, unknown>) => ({
  1039→              title: r.title, link: r.link, snippet: r.snippet,
  1040→            })),
  1041→            knowledgeGraph: data.knowledgeGraph || null,
  1042→            source: `serper-key${nextKey.index + 1}`,
  1043→          }
  1044→        } else {
  1045→          // Jina
  1046→          const text = await res.text()
  1047→          const maxChars = 30000
  1048→          const truncated = text.length > maxChars
  1049→          const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
  1050→          return {
  1051→            url: param, content, length: text.length, truncated,
  1052→            source: `jina-key${nextKey.index + 1}`,
  1053→          }
  1054→        }
  1055→      }
  1056→
  1057→      // This key also failed — report and try next
  1058→      const errorBody = await res.text().catch(() => '')
  1059→      pool.reportResult(nextKey.index, false, res.status, `HTTP ${res.status}: ${errorBody.slice(0, 200)}`)
  1060→      console.warn(`[ToolExecutor] ${service} key ${nextKey.index + 1} also failed: ${res.status}`)
  1061→    } catch (err) {
  1062→      const errMsg = err instanceof Error ? err.message : String(err)
  1063→      pool.reportResult(nextKey.index, false, undefined, errMsg)
  1064→      console.warn(`[ToolExecutor] ${service} key ${nextKey.index + 1} exception: ${errMsg}`)
  1065→    }
  1066→  }
  1067→
  1068→  return null // All keys failed
  1069→}

  1253→/**
  1254→ * Jina — Web page reader
  1255→ * Uses ServiceKeyPool for multi-key rotation with automatic fallback.
  1256→ * Primary: Direct Jina API with rotating keys from pool
  1257→ * Fallback: z-ai-web-dev-sdk page_reader function (Clawhub Market skill)
  1258→ */
  1259→async function executeJina(args: Record<string, unknown>): Promise<unknown> {
  1260→  const url = args.url as string
  1261→  if (!url) return { error: 'URL is required' }
  1262→
  1263→  // Validate URL
  1264→  try {
  1265→    new URL(url)
  1266→  } catch {
  1267→    return { error: `Invalid URL: ${url}` }
  1268→  }
  1269→
  1270→  // Primary: Direct Jina API with key pool rotation
  1271→  if (jinaKeyPool.hasKeys()) {
  1272→    const keySelection = jinaKeyPool.getNextKey()
  1273→    if (keySelection) {
  1274→      try {
  1275→        const headers: Record<string, string> = {
  1276→          'Accept': 'text/plain',
  1277→        }
  1278→        if (keySelection.key) {
  1279→          headers['Authorization'] = `Bearer ${keySelection.key}`
  1280→        }
  1281→
  1282→        const res = await fetch(`https://r.jina.ai/${url}`, {
  1283→          method: 'GET',
  1284→          headers,
  1285→          signal: AbortSignal.timeout(30000),
  1286→        })
  1287→
  1288→        if (res.ok) {
  1289→          jinaKeyPool.reportResult(keySelection.index, true, res.status)
  1290→          const text = await res.text()
  1291→
  1292→          // Limit output to prevent overwhelming LLM context
  1293→          const maxChars = 30000
  1294→          const truncated = text.length > maxChars
  1295→          const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
  1296→
  1297→          return {
  1298→            url,
  1299→            content,
  1300→            length: text.length,
  1301→            truncated,
  1302→            source: `jina-key${keySelection.index + 1}`,
  1303→          }
  1304→        }
  1305→
  1306→        // API returned error status — report to pool and try next key
  1307→        const errorBody = await res.text().catch(() => '')
  1308→        jinaKeyPool.reportResult(keySelection.index, false, res.status, `HTTP ${res.status}: ${errorBody.slice(0, 200)}`)
  1309→        console.warn(`[ToolExecutor] Jina key ${keySelection.index + 1} error: ${res.status}`)
  1310→
  1311→        // Try remaining keys before falling back to SDK
  1312→        const retryResult = await retryWithNextKey('jina', url)
  1313→        if (retryResult) return retryResult
  1314→      } catch (err) {
  1315→        const errMsg = err instanceof Error ? err.message : String(err)
  1316→        jinaKeyPool.reportResult(keySelection.index, false, undefined, errMsg)
  1317→        console.warn(`[ToolExecutor] Jina key ${keySelection.index + 1} failed: ${errMsg}`)
  1318→
  1319→        // Try remaining keys before falling back to SDK
  1320→        const retryResult = await retryWithNextKey('jina', url)
  1321→        if (retryResult) return retryResult
  1322→      }
  1323→    }
  1324→  }
  1325→
  1326→  // Fallback: z-ai-web-dev-sdk page_reader
  1327→  try {
  1328→    const zai = await getZAI()
  1329→    if (zai) {
  1330→      const result = await zai.functions.invoke('page_reader', { url })
  1331→      if (result?.data?.html) {
  1332→        // Convert HTML to clean text
  1333→        const text = result.data.html
  1334→          .replace(/<script[^>]*>[\s\S]*?<\/script>/gi, '')
  1335→          .replace(/<style[^>]*>[\s\S]*?<\/style>/gi, '')
  1336→          .replace(/<[^>]*>/g, ' ')
  1337→          .replace(/&nbsp;/g, ' ')
  1338→          .replace(/&amp;/g, '&')
  1339→          .replace(/&lt;/g, '<')
  1340→          .replace(/&gt;/g, '>')
  1341→          .replace(/\s+/g, ' ')
  1342→          .trim()
  1343→
  1344→        const maxChars = 30000
  1345→        const truncated = text.length > maxChars
  1346→        const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
  1347→
  1348→        return {
  1349→          url,
  1350→          title: result.data.title || null,
  1351→          content,
  1352→          length: text.length,
  1353→          truncated,
  1354→          source: 'z-ai-sdk-page_reader',
  1355→        }
  1356→      }
  1357→    }
  1358→  } catch (err) {
  1359→    console.warn(`[ToolExecutor] z-ai-sdk page_reader fallback failed: ${err instanceof Error ? err.message : String(err)}`)
  1360→  }
  1361→
  1362→  return {

  1364→    hint: !jinaKeyPool.hasKeys() ? 'Add JINA_API_KEY_1..2 to .env' : `All ${jinaKeyPool.getTotalCount()} keys failed (${jinaKeyPool.getAvailableCount()} available)`,

  1435→/**
  1436→ * Process a single LLM response — check for tool_calls and execute them.
  1437→ * Returns array of tool execution results for feeding back to LLM.
  1438→ *
  1439→ * This is the core of the ReAct loop:
  1440→ * 1. LLM returns response with tool_calls
  1441→ * 2. We execute each tool_call
  1442→ * 3. Results are fed back as tool messages
  1443→ * 4. LLM generates next response (may have more tool_calls or final text)
  1444→ */
  1445→export async function processToolCalls(
  1446→  toolCalls: Array<{ id: string; function: { name: string; arguments: string } }>,
  1447→): Promise<Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }>> {
  1448→  const results: Array<{ toolCallId: string; toolName: string; result: ToolExecutionResult }> = []
  1449→
  1450→  for (const tc of toolCalls) {
  1451→    let args: Record<string, unknown> = {}
  1452→    try {
  1453→      args = JSON.parse(tc.function.arguments)
  1454→    } catch {
  1455→      args = { _parseError: tc.function.arguments }
  1456→    }
  1457→
  1458→    console.log(`[ToolExecutor] Executing: ${tc.function.name}(${JSON.stringify(args).slice(0, 200)})`)
  1459→
  1460→    const result = await executeTool(tc.function.name, args)
  1461→
  1462→    results.push({
  1463→      toolCallId: tc.id,
  1464→      toolName: tc.function.name,
  1465→      result,
  1466→    })
  1467→
  1468→    console.log(`[ToolExecutor] Result: ${tc.function.name} → ${result.success ? 'OK' : 'FAIL'} (${result.duration}ms)`)
  1469→  }
  1470→
  1471→  return results
  1472→}

  1479→export function formatToolResultsAsMessages(
