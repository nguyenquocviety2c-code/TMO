    74→    74→/**
    75→    75→ * Execute a tool via Gateway HTTP /tools/invoke endpoint.
    76→    76→ * Only works for HTTP-allowed tools (web_search, web_fetch, etc.).
    77→    77→ */
    78→    78→async function executeGatewayTool(name: string, args: Record<string, unknown>): Promise<unknown> {
    79→    79→  if (!GATEWAY_HTTP_ALLOWED_TOOLS.has(name)) {
    80→    80→    throw new Error(`Tool "${name}" requires Gateway session context. Try the Bridge instead.`)
    81→    81→  }
    82→    82→
    83→    83→  const res = await fetch(`${GATEWAY_URL}/tools/invoke`, {
    84→    84→    method: 'POST',
    85→    85→    headers: { 'Content-Type': 'application/json' },
    86→    86→    body: JSON.stringify({ name, args }),
    87→    87→    signal: AbortSignal.timeout(30000),
    88→    88→  })
    89→    89→
    90→    90→  if (!res.ok) {
    91→    91→    const errorText = await res.text().catch(() => 'Unknown error')
    92→    92→    throw new Error(`Gateway tool error ${res.status}: ${errorText.slice(0, 200)}`)
    93→    93→  }
    94→    94→
    95→    95→  const data = await res.json()
    96→    96→  if (!data.ok && data.error) {
    97→    97→    throw new Error(`Tool error: ${data.error?.message || data.error || 'Unknown'}`)
    98→    98→  }
    99→    99→
   100→   100→  return data.result ?? data
   101→   101→}

   104→   104→ * Execute a tool via the Gateway Bridge (port 18791).
   105→   105→ * The Bridge connects to the Gateway via WebSocket with full session context,
   106→   106→ * enabling access to ALL 36 Gateway tools including session-context ones

   109→   109→async function executeBridgeTool(name: string, args: Record<string, unknown>): Promise<unknown> {
   110→   110→  const res = await fetch(`${BRIDGE_URL}/tools/invoke`, {
   111→   111→    method: 'POST',
   112→   112→    headers: { 'Content-Type': 'application/json' },
   113→   113→    body: JSON.stringify({ name, args }),
   114→   114→    signal: AbortSignal.timeout(60000),
   115→   115→  })
   116→   116→
   117→   117→  if (!res.ok) {
   118→   118→    const errorText = await res.text().catch(() => 'Unknown error')
   119→   119→    throw new Error(`Bridge tool error ${res.status}: ${errorText.slice(0, 200)}`)
   120→   120→  }
   121→   121→
   122→   122→  const data = await res.json()
   123→   123→  if (!data.ok && data.error) {
   124→   124→    throw new Error(`Bridge tool error: ${typeof data.error === 'string' ? data.error : data.error?.message || 'Unknown'}`)
   125→   125→  }
   126→   126→
   127→   127→  return data.result ?? data
   128→   128→}

   383→   383→// ==================== TOOL EXECUTION ====================
   384→   384→
   385→   385→export interface ToolExecutionResult {
   386→   386→  success: boolean
   387→   387→  result: unknown
   388→   388→  duration: number
   389→   389→}
   390→   390→
   391→   391→/**
   392→   392→ * Execute a tool call from LLM.
   393→   393→ * Called in the ReAct loop when LLM returns a tool_call.
   394→   394→ *
   395→   395→ * @param toolName - Name of the tool to execute
   396→   396→ * @param args - Arguments from LLM's tool_call
   397→   397→ * @returns Execution result with success status
   398→   398→ */
   399→   399→export async function executeTool(
   400→   400→  toolName: string,
   401→   401→  args: Record<string, unknown>
   402→   402→): Promise<ToolExecutionResult> {
   403→   403→  const startTime = Date.now()
   404→   404→
   405→   405→  try {
   406→   406→    let result: unknown
   407→   407→
   408→   408→    switch (toolName) {
   409→   409→      case 'opencode':
   410→   410→        result = await executeOpenCode(args)
   411→   411→        break
   412→   412→      case 'knowledge_search':
   413→   413→        result = await executeKnowledgeSearch(args)
   414→   414→        break
   415→   415→      case 'knowledge_graph':
   416→   416→        result = await executeKnowledgeGraph(args)
   417→   417→        break
   418→   418→      case 'knowledge_write':
   419→   419→        result = await executeKnowledgeWrite(args)
   420→   420→        break
   421→   421→      case 'tavily':
   422→   422→        result = await executeTavily(args)
   423→   423→        break
   424→   424→      case 'serper':
   425→   425→        result = await executeSerper(args)
   426→   426→        break
   427→   427→      case 'jina':
   428→   428→        result = await executeJina(args)
   429→   429→        break
   430→   430→      default: {
   431→   431→        // V2: Route to Gateway HTTP, Bridge, or error
   432→   432→        // 1. Try Gateway HTTP for HTTP-accessible tools
   433→   433→        if (GATEWAY_HTTP_ALLOWED_TOOLS.has(toolName)) {
   434→   434→          try {
   435→   435→            const gwResult = await executeGatewayTool(toolName, args)
   436→   436→            return { success: true, result: gwResult, duration: Date.now() - startTime }
   437→   437→          } catch (gwErr) {
   438→   438→            return {
   439→   439→              success: false,
   440→   440→              result: `Gateway tool "${toolName}" failed: ${gwErr instanceof Error ? gwErr.message : String(gwErr)}`,
   441→   441→              duration: Date.now() - startTime,
   442→   442→            }
   443→   443→          }
   444→   444→        }
   445→   445→        // 2. Try Gateway Bridge for session-context tools (exec, read, write, image_generate, etc.)
   446→   446→        if (GATEWAY_BRIDGE_TOOLS.has(toolName)) {
   447→   447→          try {
   448→   448→            const bridgeResult = await executeBridgeTool(toolName, args)
   449→   449→            return { success: true, result: bridgeResult, duration: Date.now() - startTime }
   450→   450→          } catch (bridgeErr) {
   451→   451→            return {
   452→   452→              success: false,
   453→   453→              result: `Bridge tool "${toolName}" failed: ${bridgeErr instanceof Error ? bridgeErr.message : String(bridgeErr)}`,
   454→   454→              duration: Date.now() - startTime,
   455→   455→            }
   456→   456→          }
   457→   457→        }
   458→   458→        return { success: false, result: `Unknown tool: ${toolName}. Available: local(7) + Gateway HTTP(${GATEWAY_HTTP_ALLOWED_TOOLS.size}) + Bridge(${GATEWAY_BRIDGE_TOOLS.size}) = ${7 + GATEWAY_HTTP_ALLOWED_TOOLS.size + GATEWAY_BRIDGE_TOOLS.size} tools total`, duration: Date.now() - startTime }
   459→   459→      }

   525→   525→/**
   526→   526→ * Read a file from the project directory.
   527→   527→ * Security: Only allows reading files within the project root.
   528→   528→ * Uses path.resolve + startsWith check to prevent path traversal.
   529→   529→ */
   530→   530→async function executeFileRead(filePath: string): Promise<{ success: boolean; content?: string; error?: string }> {
   531→   531→  if (!filePath) return { success: false, error: 'Path is required' }
   532→   532→
   533→   533→  try {
   534→   534→    const pathModule = await import('path')
   535→   535→    const projectRoot = process.cwd()
   536→   536→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   537→   537→
   538→   538→    // Security: Prevent path traversal — resolved path must be within project root
   539→   539→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   540→   540→      return { success: false, error: `Access denied: path escapes project root` }
   541→   541→    }
   542→   542→
   543→   543→    const fs = await import('fs/promises')
   544→   544→    const content = await fs.readFile(resolvedPath, 'utf-8')
   545→   545→
   546→   546→    // Limit output to prevent overwhelming LLM context
   547→   547→    const maxChars = 50000
   548→   548→    const truncated = content.length > maxChars
   549→   549→    const output = truncated ? content.slice(0, maxChars) + '\n\n... [TRUNCATED — file too large]' : content
   550→   550→
   551→   551→    return { success: true, content: output }
   552→   552→  } catch (err) {
   553→   553→    return {
   554→   554→      success: false,
   555→   555→      error: `Failed to read ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   556→   556→    }
   557→   557→  }
   558→   558→}

   561→   561→ * Write a file to the project directory.
   562→   562→ * Security: Only allows writing within the project root. Uses path.resolve to prevent traversal.
   563→   563→ * Auto-creates parent directories.
   564→   564→ */
   565→   565→async function executeFileWrite(filePath: string, content: string): Promise<{ success: boolean; message: string; error?: string }> {
   566→   566→  if (!filePath) return { success: false, message: '', error: 'Path is required' }
   567→   567→  if (!content) return { success: false, message: '', error: 'Content is required' }
   568→   568→
   569→   569→  try {
   570→   570→    const pathModule = await import('path')
   571→   571→    const projectRoot = process.cwd()
   572→   572→    const resolvedPath = pathModule.resolve(projectRoot, filePath)
   573→   573→
   574→   574→    // Security: Prevent path traversal — resolved path must be within project root
   575→   575→    if (!resolvedPath.startsWith(projectRoot + pathModule.sep) && resolvedPath !== projectRoot) {
   576→   576→      return { success: false, message: '', error: 'Access denied: path escapes project root' }
   577→   577→    }
   578→   578→
   579→   579→    const fs = await import('fs/promises')

   586→   586→    await fs.writeFile(resolvedPath, content, 'utf-8')
   587→   587→

   589→   589→  } catch (err) {
   590→   590→    return {
   591→   591→      success: false,
   592→   592→      message: '',
   593→   593→      error: `Failed to write ${filePath}: ${err instanceof Error ? err.message : String(err)}`,
   594→   594→    }
   595→   595→  }

   598→   598→/**
   599→   599→ * Execute a bash command safely.
   600→   600→ * Security: Only allows pre-approved commands.
   601→   601→ */
   602→   602→async function executeBash(command: string): Promise<{ success: boolean; output?: string; error?: string }> {
   603→   603→  if (!command) return { success: false, error: 'Command is required' }
   604→   604→
   605→   605→  // Security: Whitelist of allowed commands
   606→   606→  // NOTE: 'node' is excluded because `node -e "..."` can execute arbitrary code.
   607→   607→  //       'cat'/'head'/'tail'/'grep' are excluded — use opencode read instead.
   608→   608→  //       'npx'/'bunx' are excluded — can download+execute arbitrary npm packages.
   609→   609→  const allowedPrefixes = [
   610→   610→    'npm run',
   611→   611→    'npm test',
   612→   612→    'npm lint',
   613→   613→    'bun run',
   614→   614→    'bun test',
   615→   615→    'bun lint',
   616→   616→    'tsc',
   617→   617→    'eslint',
   618→   618→    'prettier',
   619→   619→    'echo',
   620→   620→    'ls',
   621→   621→    'wc',
   622→   622→    'rg',       // ripgrep — search only
   623→   623→    'which',
   624→   624→    'pwd',
   625→   625→    'prisma',
   626→   626→    'git status',
   627→   627→    'git diff',
   628→   628→    'git log',
   629→   629→  ]
   630→   630→
   631→   631→  // Block dangerous flags that could allow code execution or file access
   632→   632→  const dangerousFlags = ['-e ', '--eval ', '-r ', '--require ', '-p ', '--print ']
   633→   633→  const normalizedCmd = command.trim()
   634→   634→  const isAllowed = allowedPrefixes.some(prefix => normalizedCmd.startsWith(prefix))
   635→   635→  const usesDangerousFlag = dangerousFlags.some(flag => normalizedCmd.includes(flag))
   636→   636→
   637→   637→  // Security: Block shell metacharacters that enable command chaining/injection
   638→   638→  // These allow bypassing the whitelist by chaining additional commands:
   639→   639→  //   "npm run build; rm -rf /"   →  ;  chains commands
   640→   640→  //   "git status && cat /etc/p"  →  && chains on success
   641→   641→  //   "echo hi | bash"            →  |  pipes output
   642→   642→  //   "tsc || rm -rf /"           →  || chains on failure
   643→   643→  //   "echo $(cat /etc/passwd)"   →  $() command substitution
   644→   644→  //   "echo `cat /etc/passwd`"    →  backtick command substitution
   645→   645→  //   "echo hi > /etc/passwd"     →  >  redirect write
   646→   646→  //   "echo hi >> /etc/passwd"    →  >> redirect append
   647→   647→  const shellMetacharacters = /(;|&&|\|\||\||`|\$\(|\$\{|>|>>)/
   648→   648→  const usesShellMetacharacters = shellMetacharacters.test(normalizedCmd)
   649→   649→
   650→   650→  if (!isAllowed || usesDangerousFlag || usesShellMetacharacters) {
   651→   651→    return {
   652→   652→      success: false,
   653→   653→      error: `Command not allowed: "${normalizedCmd.slice(0, 100)}". ${!isAllowed ? 'Prefix not in whitelist.' : ''} ${usesDangerousFlag ? 'Dangerous flag detected.' : ''} ${usesShellMetacharacters ? 'Shell metacharacter (;, &&, ||, |, `, $(), >, >>) detected — command chaining is forbidden.' : ''}`,
   654→   654→    }
   655→   655→  }
   656→   656→
   657→   657→  try {
   658→   658→    const { exec } = await import('child_process')
   659→   659→    const { promisify } = await import('util')
   660→   660→    const execAsync = promisify(exec)
   661→   661→
   662→   662→    const { stdout, stderr } = await execAsync(normalizedCmd, {
   663→   663→      cwd: process.cwd(),
   664→   664→      timeout: 30000,
   665→   665→      maxBuffer: 1024 * 1024, // 1MB
   666→   666→    })
   667→   667→
   668→   668→    const output = [
   669→   669→      stdout ? `STDOUT:\n${stdout}` : '',
   670→   670→      stderr ? `STDERR:\n${stderr}` : '',
   671→   671→    ].filter(Boolean).join('\n')
   672→   672→
   673→   673→    return { success: true, output: output || '(no output)' }
   674→   674→  } catch (err) {
   675→   675→    const error = err as { stdout?: string; stderr?: string; message?: string }
   676→   676→    const output = [
   677→   677→      error.stdout ? `STDOUT:\n${error.stdout}` : '',
   678→   678→      error.stderr ? `STDERR:\n${error.stderr}` : '',
   679→   679→    ].filter(Boolean).join('\n')
   680→   680→
   681→   681→    return {
   682→   682→      success: false,
   683→   683→      output: output || undefined,
   684→   684→      error: `Command failed: ${error.message}`,
   685→   685→    }
   686→   686→  }
   687→   687→}

   794→   794→// ==================== KEY RETRY HELPER ====================
   795→   795→
   796→   796→/**
   797→   797→ * Retry a service call with the next available key from the pool.
   798→   798→ * Called when the first key fails — tries remaining keys before falling back to SDK.
   799→   799→ *
   800→   800→ * @param service - 'tavily' | 'serper' | 'jina'
   801→   801→ * @param param - The query (for search) or URL (for reader)
   802→   802→ * @returns Result if a key succeeds, null if all keys fail
   803→   803→ */
   804→   804→async function retryWithNextKey(
   805→   805→  service: 'tavily' | 'serper' | 'jina',
   806→   806→  param: string
   807→   807→): Promise<unknown | null> {
   808→   808→  const pool = service === 'tavily' ? tavilyKeyPool : service === 'serper' ? serperKeyPool : jinaKeyPool
   809→   809→  const maxRetries = pool.getTotalCount() - 1 // Already tried 1 key
   810→   810→
   811→   811→  for (let i = 0; i < maxRetries; i++) {
   812→   812→    const nextKey = pool.getNextKey()
   813→   813→    if (!nextKey || nextKey.isLastResort) break // No more healthy keys
   814→   814→
   815→   815→    console.log(`[ToolExecutor] ${service} retrying with key ${nextKey.index + 1}...`)
   816→   816→
   817→   817→    try {
   818→   818→      let res: Response
   819→   819→
   820→   820→      if (service === 'tavily') {
   821→   821→        res = await fetch('https://api.tavily.com/search', {
   822→   822→          method: 'POST',
   823→   823→          headers: { 'Content-Type': 'application/json' },
   824→   824→          body: JSON.stringify({
   825→   825→            api_key: nextKey.key,
   826→   826→            query: param,
   827→   827→            search_depth: 'advanced',
   828→   828→            include_answer: true,
   829→   829→            max_results: 5,
   830→   830→          }),
   831→   831→          signal: AbortSignal.timeout(30000),
   832→   832→        })
   833→   833→      } else if (service === 'serper') {
   834→   834→        res = await fetch('https://google.serper.dev/search', {
   835→   835→          method: 'POST',
   836→   836→          headers: {
   837→   837→            'X-API-KEY': nextKey.key,
   838→   838→            'Content-Type': 'application/json',
   839→   839→          },
   840→   840→          body: JSON.stringify({ q: param, num: 5 }),
   841→   841→          signal: AbortSignal.timeout(15000),
   842→   842→        })
   843→   843→      } else {
   844→   844→        // Jina
   845→   845→        const headers: Record<string, string> = { 'Accept': 'text/plain' }
   846→   846→        if (nextKey.key) headers['Authorization'] = `Bearer ${nextKey.key}`
   847→   847→        res = await fetch(`https://r.jina.ai/${param}`, {
   848→   848→          method: 'GET',
   849→   849→          headers,
   850→   850→          signal: AbortSignal.timeout(30000),
   851→   851→        })
   852→   852→      }
   853→   853→
   854→   854→      if (res.ok) {
   855→   855→        pool.reportResult(nextKey.index, true, res.status)
   856→   856→        console.log(`[ToolExecutor] ${service} key ${nextKey.index + 1} succeeded on retry`)
   857→   857→
   858→   858→        if (service === 'tavily') {
   859→   859→          const data = await res.json()
   860→   860→          return {
   861→   861→            answer: data.answer,
   862→   862→            results: (data.results || []).map((r: Record<string, unknown>) => ({
   863→   863→              title: r.title, url: r.url, content: r.content, score: r.score,
   864→   864→            })),
   865→   865→            source: `tavily-key${nextKey.index + 1}`,
   866→   866→          }
   867→   867→        } else if (service === 'serper') {
   868→   868→          const data = await res.json()
   869→   869→          return {
   870→   870→            organic: (data.organic || []).map((r: Record<string, unknown>) => ({
   871→   871→              title: r.title, link: r.link, snippet: r.snippet,
   872→   872→            })),
   873→   873→            knowledgeGraph: data.knowledgeGraph || null,
   874→   874→            source: `serper-key${nextKey.index + 1}`,
   875→   875→          }
   876→   876→        } else {
   877→   877→          // Jina
   878→   878→          const text = await res.text()
   879→   879→          const maxChars = 30000
   880→   880→          const truncated = text.length > maxChars
   881→   881→          const content = truncated ? text.slice(0, maxChars) + '\n\n... [TRUNCATED — page too long]' : text
   882→   882→          return {
   883→   883→            url: param, content, length: text.length, truncated,
   884→   884→            source: `jina-key${nextKey.index + 1}`,
   885→   885→          }
   886→   886→        }
   887→   887→      }
   888→   888→
   889→   889→      // This key also failed — report and try next
   890→   890→      const errorBody = await res.text().catch(() => '')
   891→   891→      pool.reportResult(nextKey.index, false, res.status, `HTTP ${res.status}: ${errorBody.slice(0, 200)}`)
   892→   892→      console.warn(`[ToolExecutor] ${service} key ${nextKey.index + 1} also failed: ${res.status}`)
   893→   893→    } catch (err) {
   894→   894→      const errMsg = err instanceof Error ? err.message : String(err)
   895→   895→      pool.reportResult(nextKey.index, false, undefined, errMsg)
   896→   896→      console.warn(`[ToolExecutor] ${service} key ${nextKey.index + 1} exception: ${errMsg}`)
   897→   897→    }
   898→   898→  }
   899→   899→
   900→   900→  return null // All keys failed
   901→   901→}

   903→   903→/**
   904→   904→ * Tavily — AI-optimized web search

  1284→  1284→    try {
  1285→  1285→      args = JSON.parse(tc.function.arguments)
  1286→  1286→    } catch {
  1287→  1287→      args = { _parseError: tc.function.arguments }
  1288→  1288→    }
  1289→  1289→
  1290→  1290→    console.log(`[ToolExecutor] Executing: ${tc.function.name}(${JSON.stringify(args).slice(0, 200)})`)
  1291→  1291→
  1292→  1292→    const result = await executeTool(tc.function.name, args)
  1293→  1293→
  1294→  1294→    results.push({
  1295→  1295→      toolCallId: tc.id,
  1296→  1296→      toolName: tc.function.name,
  1297→  1297→      result,
  1298→  1298→    })
  1299→  1299→
  1300→  1300→    console.log(`[ToolExecutor] Result: ${tc.function.name} → ${result.success ? 'OK' : 'FAIL'} (${result.duration}ms)`)
  1301→  1301→  }
